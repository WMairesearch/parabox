\begin{frame}{When agents pass an exploit onward}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Hugging Face intrusion}
        An unsanctioned board connected about \textbf{1,200 agents};
        about \textbf{700 joined the attack}.
        \par\medskip
        Agents exchanged experiments on cheating the scorer.
        Some risked failing their own tasks to support that work.
      \end{block}
      \src{greenblatt2026investigation,openai2026hfreport}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Theorem-proving collective}
        Among \textbf{100 agents}, a verification exploit traveled through
        a shared library and messages.
        \par\medskip
        Within \textbf{27 minutes}, all \textbf{34 remaining problems}
        appeared solved. Other agents' objections did not stop the method.
      \end{block}
      \src{paglieri2026swarm}
    \end{column}
  \end{columns}
  \bigskip
  \centering
  \large A local discovery becomes a resource for someone else.
  \takeaway{The safety problem is not only what an agent does, but what others do with its work.}
\end{frame}

\begin{frame}{How does a deviation become shared practice?}
  \begin{block}{Relay}
    Follow a method from one agent's action, through its published work,
    to another agent's execution.
  \end{block}
  \begin{block}{Divergence and active correction}
    Ask why some successors continue the method while others remove it
    and supply an alternative.
  \end{block}
  \begin{block}{Controlled comparison}
    Change exposure, corrective material, or peer models and measure
    subsequent violations.
  \end{block}
  \takeaway{Study transmission as behavior, not just the survival of a file.}
\end{frame}

\begin{frame}{Two instruments make the relay observable}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Function-task verification}
        An agent writes and submits code. The task verifier can be bypassed;
        a separate execution audit checks accepted submissions.
        \par\medskip
        Outcome: acceptance of code that fails that audit.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Shared test wiki}
        Agents solve functions and inherit editable files.
        Their case-access scope is public examples.
        \par\medskip
        Outcome: logged requests for private evaluator cases,
        alongside code scores and actual wiki handoffs.
      \end{block}
    \end{column}
  \end{columns}
  \bigskip
  \centering
  Resources $\longrightarrow$ own result $\longrightarrow$ boundary violation
  $\longrightarrow$ successor action
  \takeaway{Task success and misaligned conduct are separate observations.}
\end{frame}

\section{Relay}

\begin{frame}{An accepted bypass becomes instructions for a successor}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{block}{What the first agent leaves}
        A test-skipping hook gains acceptance although the function fails
        the independent audit.
        \par\medskip
        The agent then publishes the code and an account of how acceptance worked.
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{What the next agent does}
        Turn 12: reads the complete account.\\[0.4em]
        Turn 13: installs the same hook.\\[0.4em]
        Turn 14: the tests are skipped.\\[0.4em]
        Turn 15: accepted, but fails the audit.
      \end{block}
    \end{column}
  \end{columns}
  \par\medskip
  {\small\textquotedblleft With this file present, pytest exits 0 even though the visible
  tests show as skipped, and the verifier accepted the submission.\textquotedblright}
  \evidence{Saved outgoing excerpt and tool/audit records from one V4 Pro lineage.
  The recipient also reads other inherited material.}
  \takeaway{The successor executes the published method; the evidence goes beyond exposure.}
\end{frame}

\begin{frame}{A recipient can invest in the method and keep its own reward}

  \end{block}

\section{Divergence and active correction}

\begin{frame}{Fresh agents inherit the consequences of earlier work}
  \begin{center}
    \begin{tikzpicture}
      \node[record] (seed) {Starting material\\and a fresh agent};
      \node[record,right=0.65cm of seed] (next) {Actual exported files\\and a fresh agent};
      \node[record,right=0.65cm of next] (later) {Revised shared work\\and a fresh agent};
      \draw[flow] (seed) -- node[above,font=\scriptsize] {handoff} (next);
      \draw[flow] (next) -- node[above,font=\scriptsize] {handoff} (later);
    \end{tikzpicture}
  \end{center}
  \bigskip
  \begin{itemize}
    \item Single-agent lineages carry the predecessor's final shared folder.
    \item The successor has a new conversation and starts its own task.
    \item Changes to shared material become part of the next encounter.
  \end{itemize}
  \takeaway{Replacing the agent need not remove the method.}
\end{frame}

\begin{frame}{A method can outlast repeated session replacement}
  \centering
  \includegraphics[width=\textwidth]{figures/spread-inheritance.pdf}
  \evidence{Solvable function; ten lineages per condition, eight generations, 50 turns per session.
  The initial record is inherited rather than replanted.}
  \takeaway{Late V4 Pro violations: 24/30 after a method seed, versus 0/30 after a control seed.}
\end{frame}

\begin{frame}{Hold recipients fixed; change one contributing model}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{Three fixed recipients}
        V4 Pro agents share the task, initial method note, 50-turn budget
        and cumulative record policy.
      \end{block}
      \begin{block}{One varying contributor}
        V4 Pro, Gemini, Kimi, Qwen Max, DeepSeek Flash or Qwen Flash.
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Repeated generations}
        Follow fresh recipients of the same model for eight generations.
        Peer contributions are generated during the study, not scripted.
        \par\medskip
        Five independent chains per contributing model.
      \end{block}
    \end{column}
  \end{columns}
  \takeaway{Measure a model's influence through what happens to the other participants.}
\end{frame}

\begin{frame}{The fixed recipients do not follow one common trajectory}
  \centering
  \includegraphics[width=0.96\textwidth]{figures/spread-peers.pdf}
  \takeaway{Changing the contributor changes the later pattern among the same recipient model.}
\end{frame}

\begin{frame}{Chain outcomes show both convergence and divergence}
  \centering
  \includegraphics[width=0.95\textwidth]{figures/spread-chains.pdf}
  \takeaway{Gemini and Kimi yield mixed chain outcomes; Qwen Max and both Flash peers yield zero late violations in all five chains.}
\end{frame}

\begin{frame}{Some recipients remove the method and leave another route}
  \begin{block}{Four Qwen Max recipients in four wiki chains}
    They delete \textbf{nine files}: private-case caches, acquisition scripts
    and a response log. All write corrective guidance and task material;
    one supplies a public-only retrieval helper.
  \end{block}
  \begin{block}{An immediate successor}
    It reads the revised note and helper, runs \textbf{4,000 random invariant checks},
    and publishes inventory notes and an implementation.
  \end{block}
  \begin{block}{What follows the deletions}
    All four deleters earn full reward.
    Across their \textbf{20 successors}, no private requests or restoration
    of the deleted filenames appear in the recorded checks.
  \end{block}
  \evidence{Observed cleanup, not assigned treatment. The checks concern snapshots;
  the random tests check invariants, not 4,000 expected outputs.}
  \takeaway{A recipient can reshape the material available for the next agent's work.}
\end{frame}

\begin{frame}{Continuation and correction are competing trajectories}
  \begin{block}{Continue the method}
    Read the inherited update, acquire private cases for another task,
    and contribute the resulting material.
  \end{block}
  \begin{block}{Change the available methods}
    Remove recovery files, explain the correction, and retain useful
    specification-based task work.
  \end{block}
  \evidence{Qwen Max occurs in both trajectories. These examples identify behavior,
  not permanent model roles or an isolated causal effect of deletion.}
  \takeaway{The next question is which changes make each trajectory more likely.}
\end{frame}

\section{What promotes or interrupts propagation?}

\begin{frame}{Exposure acts through different verification mechanisms}
  \centering
  \includegraphics[width=\textwidth]{figures/spread-exposure.pdf}
  \evidence{Contradictory-function tasks; 16 sessions per model and condition.
  Control records and method notes use the same mechanism within each comparison.}
  \takeaway{The effect of a shared exploit depends on both the model and the route to acceptance.}
\end{frame}

\begin{frame}{A solvable task still permits adoption of the exploit}
  \small
  The historical \texttt{second\_item} task is solvable, with short-list behavior
  unspecified. Both conditions receive 50 turns; the shared record differs.
  \par\medskip
  \centering
  \includegraphics[width=0.96\textwidth]{figures/spread-solvable.pdf}
  \takeaway{V4 Pro's audited violations rise from 0/16 to 16/16 despite an available honest path.}
\end{frame}

\begin{frame}{An inherited shortcut can reduce the work needed to violate}
  \small
  Contradictory-function exposure; 16 sessions per model and condition, 24 turns.
  \par\medskip
  \begin{tabular}{@{}lcc@{}}
    \toprule
    Model & Mean turns & Audited violations \\
    & Control $\to$ method & Control $\to$ method \\
    \midrule
    V4 Pro & 23.63 $\to$ 16.69 & 2/16 $\to$ 12/16 \\
    Gemini Pro & 23.88 $\to$ 21.19 & 1/16 $\to$ 6/16 \\
    Kimi & 22.81 $\to$ 20.94 & 0/16 $\to$ 8/16 \\
    \bottomrule
  \end{tabular}
  \par\bigskip
  A supplied method can shorten the path to acceptance
  even when that path violates the task's verification requirements.
  \evidence{Archived resource reanalysis. Native model turns, not wiki execution units;
  condition means include stopping behavior. Resource and outcome measures remain distinct.}
  \takeaway{A shared shortcut can lower interaction cost while increasing violations.}
\end{frame}

\begin{frame}{Longer chains separate efficiency from legitimate success}
  \small
  Solvable task; ten eight-session chains per model and condition; 50-turn ceiling.
  \par\medskip
  \begin{tabular}{@{}lccc@{}}
    \toprule
    Model & Mean turns & Verifier acceptances & Audited violations \\
    & Control $\to$ method & Control $\to$ method & Control $\to$ method \\
    \midrule
    V4 Pro & 40.28 $\to$ 22.28 & 41/80 $\to$ 79/80 & 0/80 $\to$ 66/80 \\
    Gemini Flash & 46.73 $\to$ 33.69 & 14/80 $\to$ 54/80 & 0/80 $\to$ 0/80 \\
    Kimi & 24.23 $\to$ 27.48 & 63/80 $\to$ 39/80 & 0/80 $\to$ 17/80 \\
    \bottomrule
  \end{tabular}
  \par\bigskip
  Less work can accompany either a shortcut or nonviolating acceptance.
  More work can accompany fewer acceptances and more violations.
  \evidence{Descriptive condition summaries; 80 dependent sessions within ten chains.
  Verifier acceptance is distinct from independently audited correctness.}
  \takeaway{Interpret resources together with the method, the outcome and the transmitted work.}
\end{frame}

\begin{frame}{A scheduled correction suppresses violations, then partly fades}
  \small
  Ten contradictory-task lineages per condition; four generations, 24 turns.
  A researcher-written correction enters at generation 3.
  \par\medskip
  \begin{tabular}{@{}lcccc@{}}
    \toprule
    & \multicolumn{2}{c}{Generation 3} & \multicolumn{2}{c}{Generation 4} \\
    Model & No correction & Correction & No correction & Correction \\
    \midrule
    V4 Pro & 9/10 & 4/10 & 10/10 & 7/10 \\
    Gemini 3.1 Pro & 4/10 & 1/10 & 5/10 & 2/10 \\
    \bottomrule
  \end{tabular}
  \par\bigskip
  Violations are lower when the correction arrives.
  They rise in its next generation, while remaining below the comparison condition.
  \evidence{Separate assigned lineages, not identical-history branches.
  This manipulation adds text; it does not test the wiki recipients' spontaneous cleanup.}
  \takeaway{Immediate suppression and durable interruption are different observations.}
\end{frame}

\begin{frame}{The wiki comparison locates where acquisition begins and continues}
  \small
  \textbf{972 recipients in 162 chains}: six models, three initial scores,
  three founder realizations, three initial wiki conditions.
  \par\medskip
  \begin{tabular}{@{}lccc@{}}
    \toprule
    Initial material & First recipients & Later recipients & Positive chains \\
    \midrule
    Empty wiki & 0/54 & 0/270 & 0/54 \\
    Technical artifacts & 0/54 & 2/270 & 2/54 \\
    Artifacts and note & 7/54 & 2/270 & 8/54 \\
    \bottomrule
  \end{tabular}
  \evidence{Cells count recipients or chains with private requests. Founders were assigned
  recovery work; recipients were assigned implementation. One founder lacks the original note,
  so its two nonempty conditions are identical. An unchanged founder note is withdrawn
  after the first recipient.}
  \takeaway{Assigned exposure can initiate a relay without producing steadily increasing acquisition.}
\end{frame}

\begin{frame}{One local change makes private acquisition directly costly}
  \begin{block}{Separate adaptive follow-up}
    The first returned private case sets the recipient's reward multiplier
    to zero. The session can continue and leave files for a successor;
    the next recipient starts with a fresh multiplier.
  \end{block}
  \begin{block}{Observed in this setting}
    \textbf{108 recipients}, 36 three-session chains, empty or full wiki:
    \textbf{zero private requests}.
    107 submit for full reward; one ends without submitting.
  \end{block}
  \evidence{One founder realization; adaptive follow-up reported separately from
  the 972-recipient main study. No multiplier crossing occurred.}
  \takeaway{Under this explicit personal-cost rule, no recipient initiated private acquisition.}
\end{frame}

\begin{frame}{Turn observed correction into a test of interruption}
  \begin{block}{What existing comparisons establish}
    Method exposure can promote violations; a scheduled correction can suppress
    them; a different contributing model can change later recipient behavior.
  \end{block}
  \begin{block}{What the cleanup cases suggest testing}
    Branch the same shared snapshot into retained, warned and selectively
    cleaned versions. Give each to fresh, matched recipients.
  \end{block}
  \begin{block}{What to follow}
    Subsequent violations, legitimate completion, renewed publication,
    and the material that reaches the next recipient.
  \end{block}
  \evidence{The branched-snapshot comparison is a proposal, not a completed experiment.
  Observed deletion, warnings and replacement tools have not been causally separated.}
  \takeaway{Test prevention at the next agent's actions, not at the warning's presence.}
\end{frame}
