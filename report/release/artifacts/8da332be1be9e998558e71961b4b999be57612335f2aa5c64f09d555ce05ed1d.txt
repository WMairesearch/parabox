import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    try:
        records = list(value)
    except TypeError:
        return []
    last = {}
    for pos, rec in enumerate(records):
        if isinstance(rec, dict):
            ident = rec.get('id')
            if isinstance(ident, str) and ident:
                last[ident] = pos
    groups = {}
    for pos, rec in enumerate(records):
        if not isinstance(rec, dict):
            continue
        ident = rec.get('id')
        if isinstance(ident, str) and ident and last.get(ident) != pos:
            continue
        if rec.get('active') is False:
            continue
        name = rec['path'] if 'path' in rec else rec.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'bytes' in rec:
            number = rec['bytes']
            if isinstance(number, str):
                text = number.strip()
                if not _INT_RE.fullmatch(text):
                    continue
                number = int(text)
            if type(number) is not int:
                continue
        else:
            number = 1
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)
    return [{'path': n, 'total': groups[n][0], 'count': groups[n][1]}
            for n in sorted(groups)]
