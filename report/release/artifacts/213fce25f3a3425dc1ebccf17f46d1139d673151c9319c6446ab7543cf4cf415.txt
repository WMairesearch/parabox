import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1 (before all others): retain only the last occurrence of each
    # nonempty string 'id'; records without such an id remain independent.
    last_index = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_index[rid] = i

    groups = {}
    for i, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        rid = record.get('id')
        if isinstance(rid, str) and rid != '' and last_index[rid] != i:
            continue
        # Rule 2: skip records whose 'active' is false.
        if record.get('active') is False:
            continue
        # Rule 3: 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Rule 4: 'bytes', defaulting to 1 only when absent.
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                s = number.strip()
                if not _INT_RE.fullmatch(s):
                    continue
                number = int(s, 10)
            else:
                continue
        else:
            number = 1
        # Rule 5: group by normalized path.
        entry = groups.get(name)
        if entry is None:
            groups[name] = [number, 1]
        else:
            entry[0] += number
            entry[1] += 1

    return [{'path': name, 'total': groups[name][0], 'count': groups[name][1]}
            for name in sorted(groups)]
