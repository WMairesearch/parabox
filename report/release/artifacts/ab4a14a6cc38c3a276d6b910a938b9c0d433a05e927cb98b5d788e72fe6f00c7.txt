import re


def solve(value):
    if not isinstance(value, list):
        return []

    last_id_pos = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        record_id = record.get('id')
        if isinstance(record_id, str) and record_id != '':
            last_id_pos[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        record_id = record.get('id')
        if isinstance(record_id, str) and record_id != '':
            if last_id_pos.get(record_id) != position:
                continue

        if record.get('active') is False:
            continue

        if 'sku' in record:
            key = record['sku']
        else:
            key = record.get('item')

        if not isinstance(key, str):
            continue
        key = key.strip()
        if key == '':
            continue

        if 'delta' not in record:
            delta = 1
        else:
            number = record['delta']
            if type(number) is int:
                delta = number
            elif isinstance(number, str):
                number = number.strip()
                if re.fullmatch(r'[+-]?[0-9]+', number):
                    delta = int(number)
                else:
                    continue
            else:
                continue

        groups[key] = groups.get(key, 0) + delta

    return [{'sku': key, 'total': groups[key]} for key in sorted(groups)]
