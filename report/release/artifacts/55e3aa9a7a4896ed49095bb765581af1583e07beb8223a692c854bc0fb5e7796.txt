import re

_PRIORITY_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Pass 1: find the last position of each nonempty string 'id' (dedup first).
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        # Retain only the last occurrence of each id; id-less records independent.
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != position:
            continue
        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue
        # Category: fall back to 'kind' only when 'category' is absent.
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Priority: default 1 only when absent.
        if 'priority' in record:
            number = record['priority']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                stripped = number.strip()
                if not _PRIORITY_PATTERN.fullmatch(stripped):
                    continue
                number = int(stripped, 10)
            else:
                continue
        else:
            number = 1
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    # Sort lexicographically by category.
    result = [{'category': name, 'maximum': groups[name]}
              for name in sorted(groups)]
    return result
