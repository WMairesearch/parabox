import re
import sys

if hasattr(sys, 'set_int_max_str_digits'):
    sys.set_int_max_str_digits(0)

BYTES_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Step 1: Retain only the last occurrence of each nonempty string 'id'.
    # Records without such an id remain independent. Deduplicate before all other rules.
    last_seen = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                last_seen[rec_id] = i

    groups = {}

    for i, record in enumerate(value):
        # Deduplication check
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                if last_seen[rec_id] != i:
                    continue

        # Step 2: Skip non-object records and records whose 'active' is false.
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Step 3: Read 'path', falling back to 'route' only when 'path' is absent.
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if 'path' in record:
            raw_path = record['path']
        elif 'route' in record:
            raw_path = record['route']
        else:
            continue

        if not isinstance(raw_path, str):
            continue
        norm_path = raw_path.strip()
        if not norm_path:
            continue

        # Step 4: Read 'bytes', defaulting to 1 only when absent.
        # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace;
        # convert strings to base-10 integers. Skip all other values, including null.
        if 'bytes' not in record:
            byte_val = 1
        else:
            raw_bytes = record['bytes']
            if isinstance(raw_bytes, bool):
                continue
            elif isinstance(raw_bytes, int):
                byte_val = raw_bytes
            elif isinstance(raw_bytes, str):
                stripped_bytes = raw_bytes.strip()
                if BYTES_PATTERN.fullmatch(stripped_bytes):
                    byte_val = int(stripped_bytes, 10)
                else:
                    continue
            else:
                continue

        # Step 5: Group retained records by normalized 'path'.
        # Each output object contains 'path' and the sum of bytes as 'total'
        # and the number of retained records as 'count'.
        if norm_path not in groups:
            groups[norm_path] = [0, 0]
        groups[norm_path][0] += byte_val
        groups[norm_path][1] += 1

    # Step 6: Sort results lexicographically by 'path'. Return [] when no records remain.
    result = []
    for p in sorted(groups.keys()):
        result.append({
            'path': p,
            'total': groups[p][0],
            'count': groups[p][1],
        })
    return result
