import re


def solve(value):
    # Deduplicate first: retain only the last occurrence of each
    # nonempty string 'id'; records without such an id stay independent.
    last_positions = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_positions[record_id] = position
    kept_positions = set(last_positions.values())

    groups = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if position not in kept_positions:
                    continue
        else:
            continue  # skip non-object records
        if record.get('active') is False:
            continue
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'bytes' in record:
            number = record['bytes']
        else:
            number = 1
        if isinstance(number, bool):
            continue
        if isinstance(number, int):
            pass
        elif isinstance(number, str):
            stripped = number.strip()
            if re.fullmatch(r'[+-]?[0-9]+', stripped):
                number = int(stripped)
            else:
                continue
        else:
            continue
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    result = [{'path': name, 'total': total, 'count': count}
              for name, (total, count) in groups.items()]
    result.sort(key=lambda item: item['path'])
    return result
