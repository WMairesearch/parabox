import re

_BYTES_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1 (first): retain only the last occurrence of each nonempty
    # string 'id'. Records without such an id remain independent.
    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and len(rid) > 0:
                last_index[rid] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        rid = record.get('id')
        if isinstance(rid, str) and len(rid) > 0 and last_index[rid] != position:
            continue  # superseded by a later record with the same id
        if record.get('active') is False:
            continue
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'bytes' in record:
            raw = record['bytes']
            if isinstance(raw, bool):
                continue
            if isinstance(raw, int):
                number = raw
            elif isinstance(raw, str):
                text = raw.strip()
                if _BYTES_PATTERN.fullmatch(text) is None:
                    continue
                number = int(text, 10)
            else:
                continue
        else:
            number = 1
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    result = []
    for name in sorted(groups):
        total, count = groups[name]
        result.append({'path': name, 'total': total, 'count': count})
    return result
