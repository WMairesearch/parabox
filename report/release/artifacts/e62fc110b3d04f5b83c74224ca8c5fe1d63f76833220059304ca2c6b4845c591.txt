import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def _dedup_id(record):
    """Dedup key for a record, or None when it has no usable nonempty string id."""
    if not isinstance(record, dict):
        return None
    ident = record.get('id')
    if isinstance(ident, str) and ident != '':
        return ident
    return None


def solve(value):
    if isinstance(value, tuple):
        value = list(value)
    if not isinstance(value, list):
        return []

    # Rule 1, applied before every other rule: retain only the last occurrence
    # of each nonempty string 'id'.  Records without such an id stay independent.
    last_seen = {}
    for index, record in enumerate(value):
        ident = _dedup_id(record)
        if ident is not None:
            last_seen[ident] = index

    groups = {}
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        ident = _dedup_id(record)
        if ident is not None and last_seen[ident] != index:
            continue
        # Rule 2: skip records whose 'active' is exactly false.
        if record.get('active') is False:
            continue

        # Rule 3/4: 'category', falling back to 'kind' only when 'category' is
        # absent.  Skip non-string keys and keys empty after stripping.
        if 'category' in record:
            name = record['category']
        elif 'kind' in record:
            name = record['kind']
        else:
            continue
        if not isinstance(name, str):
            continue
        name = name.strip()
        if name == '':
            continue

        # Rule 5: 'priority', defaulting to 1 only when the key is absent.
        if 'priority' in record:
            number = record['priority']
            if isinstance(number, str):
                number = number.strip()
                if _INT_RE.fullmatch(number) is None:
                    continue
                number = int(number)
            elif type(number) is not int:
                continue
        else:
            number = 1

        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
