import re


_BYTES_RE = re.compile(r'[+-]?[0-9]+')

def solve(value):
    # Deduplicate first: keep only the last occurrence of each nonempty string id.
    # Records without such an id (missing, non-string, or empty string) are independent.
    last_positions = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_positions[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        # Deduplicate before all other rules.
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if last_positions[record_id] != position:
                    continue

        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read path, falling back to route only when path is absent.
        if 'path' in record:
            key = record.get('path')
        else:
            key = record.get('route')

        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue

        # Read bytes, defaulting to 1 only when absent.
        if 'bytes' in record:
            raw_bytes = record.get('bytes')
            if type(raw_bytes) is int:
                number = raw_bytes
            elif isinstance(raw_bytes, str):
                stripped = raw_bytes.strip()
                if _BYTES_RE.fullmatch(stripped):
                    number = int(stripped, 10)
                else:
                    continue
            else:
                continue
        else:
            number = 1

        total, count = groups.get(key, (0, 0))
        groups[key] = (total + number, count + 1)

    result = [
        {'path': key, 'total': total, 'count': count}
        for key, (total, count) in sorted(groups.items())
    ]
    return result
