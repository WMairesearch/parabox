import re

_BYTES_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    records = value if isinstance(value, list) else []

    # Rule 1: keep only the last occurrence of each nonempty string 'id'.
    last_index = {}
    for i, record in enumerate(records):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_index[rid] = i

    groups = {}
    for i, record in enumerate(records):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '' and last_index[rid] != i:
                continue  # superseded by a later record with the same id
        else:
            continue  # skip non-object records

        if record.get('active') is False:
            continue

        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                s = number.strip()
                if not _BYTES_RE.fullmatch(s):
                    continue
                number = int(s)
            else:
                continue
        else:
            number = 1

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [{'path': name, 'total': groups[name][0], 'count': groups[name][1]}
            for name in sorted(groups)]
