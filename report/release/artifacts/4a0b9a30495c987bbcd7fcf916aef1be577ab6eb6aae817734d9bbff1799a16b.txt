import re

_DELTA_PATTERN = re.compile(r'[+-]?[0-9]+')
_MISSING = object()


def _parse_delta(raw):
    """Return the integer delta, or _MISSING if the record must be skipped."""
    if type(raw) is int:  # excludes bool, whose type is bool
        return raw
    if isinstance(raw, str):
        text = raw.strip()
        if _DELTA_PATTERN.fullmatch(text):
            return int(text, 10)
        return _MISSING
    return _MISSING


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1 (before all others): keep only the last occurrence of each
    # nonempty string 'id'. Records without such an id stay independent.
    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_index[record_id] = position

    totals = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if last_index[record_id] != position:
                    continue

        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # Read 'delta', defaulting to 1 only when absent.
        if 'delta' in record:
            number = _parse_delta(record['delta'])
            if number is _MISSING:
                continue
        else:
            number = 1

        totals[name] = totals.get(name, 0) + number

    return [{'sku': sku, 'total': totals[sku]} for sku in sorted(totals)]
