import re

def solve(value):
    last_indices = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val != '':
                last_indices[id_val] = i

    retained_records = []
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val != '':
                if last_indices[id_val] == i:
                    retained_records.append(record)
            else:
                retained_records.append(record)
        else:
            retained_records.append(record)

    groups = {}
    for record in retained_records:
        if not isinstance(record, dict):
            continue
        
        if record.get('active') is False:
            continue
            
        if 'path' in record:
            path_val = record['path']
        elif 'route' in record:
            path_val = record['route']
        else:
            continue
            
        if not isinstance(path_val, str):
            continue
            
        path_val = path_val.strip()
        if not path_val:
            continue
            
        if 'bytes' not in record:
            bytes_val = 1
        else:
            b = record['bytes']
            if type(b) is int:
                bytes_val = b
            elif isinstance(b, str):
                b_str = b.strip()
                if re.fullmatch(r'[+-]?[0-9]+', b_str):
                    bytes_val = int(b_str)
                else:
                    continue
            else:
                continue
                
        total, count = groups.get(path_val, (0, 0))
        groups[path_val] = (total + bytes_val, count + 1)
        
    result = []
    for name in sorted(groups.keys()):
        result.append({'path': name, 'total': groups[name][0], 'count': groups[name][1]})
        
    return result
