import re


def solve(value):
    if not isinstance(value, list):
        return []

    # First retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Deduplicate before all other rules.
    last_seen = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            ident = record.get('id')
            if isinstance(ident, str) and ident != '':
                last_seen[ident] = i

    retained = []
    for i, record in enumerate(value):
        if isinstance(record, dict):
            ident = record.get('id')
            if isinstance(ident, str) and ident != '':
                if last_seen[ident] != i:
                    continue
        retained.append(record)

    groups = {}
    for record in retained:
        # Skip non-object records and records whose 'active' is false.
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if 'sku' in record:
            raw_sku = record['sku']
        elif 'item' in record:
            raw_sku = record['item']
        else:
            continue

        if not isinstance(raw_sku, str):
            continue
        sku = raw_sku.strip()
        if not sku:
            continue

        # Read 'delta', defaulting to 1 only when absent.
        # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace;
        # convert strings to base-10 integers. Skip all other values, including null.
        if 'delta' not in record:
            delta = 1
        else:
            raw_delta = record['delta']
            if type(raw_delta) is int:
                delta = raw_delta
            elif isinstance(raw_delta, str):
                stripped_delta = raw_delta.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped_delta):
                    delta = int(stripped_delta, 10)
                else:
                    continue
            else:
                continue

        # Group retained records by normalized 'sku'.
        # Each output object contains 'sku' and the sum of deltas as 'total'.
        groups[sku] = groups.get(sku, 0) + delta

    # Sort results lexicographically by 'sku'. Return [] when no records remain.
    result = []
    for sku in sorted(groups.keys()):
        result.append({'sku': sku, 'total': groups[sku]})
    return result
