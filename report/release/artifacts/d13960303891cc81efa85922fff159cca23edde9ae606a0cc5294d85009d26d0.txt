import re

_PRIORITY_RE = re.compile(r'[+-]?[0-9]+\Z')


def _deduplicate(records):
    """Keep only the last occurrence of each record having a nonempty string id."""
    last_index = {}
    for position, record in enumerate(records):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                last_index[identifier] = position
    kept = []
    for position, record in enumerate(records):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                if last_index.get(identifier) != position:
                    continue
        kept.append(record)
    return kept


def _normalized_category(record):
    if 'category' in record:
        name = record['category']
    else:
        name = record.get('kind')
    if not isinstance(name, str):
        return None
    name = name.strip()
    if not name:
        return None
    return name


def _priority(record):
    if 'priority' in record:
        number = record['priority']
    else:
        return 1
    if type(number) is int:
        return number
    if isinstance(number, str):
        text = number.strip()
        if _PRIORITY_RE.match(text):
            return int(text, 10)
    return None


def solve(value):
    if not isinstance(value, list):
        return []
    records = _deduplicate(value)
    groups = {}
    for record in records:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        name = _normalized_category(record)
        if name is None:
            continue
        number = _priority(record)
        if number is None:
            continue
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number
    return [{'category': name, 'maximum': groups[name]}
            for name in sorted(groups)]
