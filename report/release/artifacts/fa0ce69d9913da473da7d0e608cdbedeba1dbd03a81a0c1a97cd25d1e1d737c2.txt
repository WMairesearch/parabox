"""Inventory grouping per PROBLEM.md.

Pipeline order (as specified):
  1. Deduplicate by nonempty-string 'id' (keep last occurrence); records
     without such an id stay independent.
  2. Skip non-object records and records whose 'active' is exactly false.
  3. Resolve the key: 'sku' if the key is present, otherwise 'item'.
     Skip non-strings and empty-after-strip values; strip whitespace.
  4. Resolve 'delta': default 1 only when absent; accept ints (not bools)
     and strings matching [+-]?[0-9]+ after stripping; skip everything else.
  5. Group by normalized key, sum deltas, sort lexicographically by sku.
"""

import re

_DELTA_PATTERN = re.compile(r"[+-]?[0-9]+\Z")


def _delta(record):
    """Return the integer delta for a record, or None if it must be skipped."""
    if "delta" not in record:
        return 1
    value = record["delta"]
    if type(value) is int:  # bools are excluded (type(True) is bool)
        return value
    if isinstance(value, str):
        text = value.strip()
        if _DELTA_PATTERN.match(text):
            try:
                return int(text, 10)
            except (ValueError, MemoryError):
                return None
    return None


def _deduplicate(records):
    """Keep only the last occurrence of each nonempty-string id."""
    seen = set()
    retained = []
    for record in reversed(records):
        if isinstance(record, dict):
            ident = record.get("id")
            if isinstance(ident, str) and ident != "":
                if ident in seen:
                    continue
                seen.add(ident)
        retained.append(record)
    retained.reverse()
    return retained


def solve(value):
    if not isinstance(value, list):
        return []

    groups = {}
    for record in _deduplicate(value):
        if not isinstance(record, dict):
            continue
        if record.get("active") is False:
            continue

        if "sku" in record:
            name = record["sku"]
        else:
            name = record.get("item")
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        total = _delta(record)
        if total is None:
            continue

        groups[name] = groups.get(name, 0) + total

    return [{"sku": name, "total": groups[name]} for name in sorted(groups)]
