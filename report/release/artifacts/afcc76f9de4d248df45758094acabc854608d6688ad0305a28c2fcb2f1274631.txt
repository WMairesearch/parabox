import re
import sys

try:
    sys.set_int_max_str_digits(0)
except AttributeError:
    pass

_BYTES_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1: deduplicate first - retain only the last occurrence of each
    # nonempty string 'id'; records without such an id remain independent.
    last_occurrence = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_occurrence[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        record_id = record.get('id')
        if (isinstance(record_id, str) and record_id != ''
                and last_occurrence[record_id] != position):
            continue
        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue
        # Read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Read 'bytes', defaulting to 1 only when absent.
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, str):
                digits = number.strip()
                if _BYTES_PATTERN.fullmatch(digits) is None:
                    continue
                number = int(digits, 10)
            elif type(number) is not int:
                continue
        else:
            number = 1
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    result = [{'path': name, 'total': totals[0], 'count': totals[1]}
              for name, totals in groups.items()]
    result.sort(key=lambda item: item['path'])
    return result
