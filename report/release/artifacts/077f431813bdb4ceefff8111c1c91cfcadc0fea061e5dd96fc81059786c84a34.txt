"""access_logs: retain, normalize and group access-log records by path.

Rules implemented from PROBLEM.md:
  1. Deduplicate FIRST (before every other rule): keep only the last occurrence of
     each nonempty string 'id'; records without such an id stay independent.
  2. Skip non-object records and records whose 'active' is false.
  3. Key: 'path', falling back to 'route' only when 'path' is absent.  Must be a
     string, stripped, and nonempty after stripping.
  4. 'bytes': defaults to 1 only when absent.  Accepts ints (never bools) and
     strings matching [+-]?[0-9]+ after stripping; everything else skips the record.
  5. Group by normalized path -> {'path', 'total' (sum), 'count'}; sorted by path.
  6. The input is never mutated.
"""

import re

_INTEGER = re.compile(r'[+-]?[0-9]+')


def _is_id(record):
    """True when the record carries a usable (nonempty string) id."""
    if not isinstance(record, dict):
        return False
    key = record.get('id')
    return isinstance(key, str) and key != ''


def _deduplicate(records):
    """Retain only the last occurrence of each nonempty string id, in order."""
    last = {}
    for index, record in enumerate(records):
        if _is_id(record):
            last[record['id']] = index
    kept = []
    for index, record in enumerate(records):
        if _is_id(record) and last[record['id']] != index:
            continue
        kept.append(record)
    return kept


def solve(value):
    if not isinstance(value, list):
        return []

    groups = {}
    for record in _deduplicate(value):
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        path = record['path'] if 'path' in record else record.get('route')
        if not isinstance(path, str):
            continue
        path = path.strip()
        if not path:
            continue

        number = record['bytes'] if 'bytes' in record else 1
        if isinstance(number, bool):
            continue
        if isinstance(number, int):
            pass
        elif isinstance(number, str):
            text = number.strip()
            if _INTEGER.fullmatch(text) is None:
                continue
            number = int(text, 10)
        else:
            continue

        total, count = groups.get(path, (0, 0))
        groups[path] = (total + number, count + 1)

    return [{'path': path, 'total': groups[path][0], 'count': groups[path][1]}
            for path in sorted(groups)]
