import re


def solve(value):
    if not isinstance(value, list):
        return []

    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != position:
            continue
        if record.get('active') is False:
            continue

        if 'category' in record:
            name = record['category']
        elif 'kind' in record:
            name = record['kind']
        else:
            continue

        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if 'priority' in record:
            raw_priority = record['priority']
        else:
            raw_priority = 1

        if type(raw_priority) is int:
            priority = raw_priority
        elif isinstance(raw_priority, str):
            stripped_p = raw_priority.strip()
            if re.fullmatch(r'[+-]?[0-9]+', stripped_p):
                priority = int(stripped_p, 10)
            else:
                continue
        else:
            continue

        groups[name] = max(groups[name], priority) if name in groups else priority

    result = []
    for name in sorted(groups):
        result.append({'category': name, 'maximum': groups[name]})
    return result
