import re


_DIGITS = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: keep only the last occurrence of each nonempty string id.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if type(identifier) is str and identifier != '':
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if type(identifier) is str and identifier != '' and latest[identifier] != position:
            continue
        if record.get('active') is False:
            continue
        # 'path' when present (even if unusable), otherwise 'route'.
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if type(name) is not str:
            continue
        name = name.strip()
        if not name:
            continue
        number = record['bytes'] if 'bytes' in record else 1
        if type(number) is str:
            stripped = number.strip()
            if not _DIGITS.fullmatch(stripped):
                continue
            number = int(stripped, 10)
        elif type(number) is not int:
            continue
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': groups[name][0], 'count': groups[name][1]}
        for name in sorted(groups)
    ]
