import re


def solve(value):
    # Deduplicate before all other rules: keep only the last occurrence of
    # each nonempty string 'id'.  Records without such an id stay independent.
    last_by_id = {}
    records = []
    for record in value:
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_by_id[record_id] = record
            else:
                records.append(record)
        else:
            records.append(record)
    records.extend(last_by_id.values())

    groups = {}
    for record in records:
        if not isinstance(record, dict):
            continue

        # Skip records whose 'active' member is JSON false.
        if record.get('active') is False:
            continue

        # Read sku, falling back to item only when sku is absent.
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')

        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # Read delta, defaulting to 1 only when absent.
        if 'delta' in record:
            number = record['delta']
        else:
            number = 1

        if isinstance(number, bool):
            continue
        if isinstance(number, int):
            delta = number
        elif isinstance(number, str):
            stripped = number.strip()
            if re.fullmatch(r'[+-]?[0-9]+', stripped):
                delta = int(stripped, 10)
            else:
                continue
        else:
            continue

        groups[name] = groups.get(name, 0) + delta

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
