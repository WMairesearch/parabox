"""Inventory grouping per PROBLEM.md."""

import re

_DELTA_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1 (first): keep only the last occurrence of each nonempty string id.
    # Records without such an id stay independent.
    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                last_index[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier != '':
            if last_index[identifier] != position:
                continue

        if record.get('active') is False:
            continue

        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if name == '':
            continue

        if 'delta' in record:
            number = record['delta']
            if isinstance(number, str):
                candidate = number.strip()
                if _DELTA_RE.fullmatch(candidate):
                    number = int(candidate, 10)
            elif type(number) is not int:
                continue
        else:
            number = 1

        if type(number) is not int:
            continue

        groups[name] = groups.get(name, 0) + number

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
