import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Deduplicate first: keep only the last occurrence of each nonempty
    # string 'id'. Records without such an id stay independent.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier:
            if latest[identifier] != position:
                continue
        if record.get('active') is False:
            continue

        # 'path' is used whenever present; 'route' only when 'path' is absent.
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # 'bytes' defaults to 1 only when absent.
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                text = number.strip()
                if not _INT_RE.fullmatch(text):
                    continue
                number = int(text, 10)
            else:
                continue
        else:
            number = 1

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': groups[name][0], 'count': groups[name][1]}
        for name in sorted(groups)
    ]
