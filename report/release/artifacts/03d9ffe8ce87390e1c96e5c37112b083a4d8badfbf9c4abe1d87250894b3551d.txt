import re


def solve(value):
    # Deduplicate first: retain only the last occurrence of each nonempty
    # string 'id'; records without such an id remain independent.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != position:
            continue
        if record.get('active') is False:
            continue
        # Fall back to 'kind' only when 'category' is absent.
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')
        if not isinstance(name, str) or not name.strip():
            continue
        name = name.strip()
        number = record.get('priority', 1)
        if isinstance(number, str):
            stripped = number.strip()
            if re.fullmatch(r'[+-]?[0-9]+', stripped):
                number = int(stripped, 10)
        if type(number) is not int:  # excludes booleans and everything else
            continue
        groups[name] = max(groups[name], number) if name in groups else number

    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
