import re


def solve(value):
    # Deduplicate by nonempty string 'id', keeping only the last occurrence.
    last_id_index = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            id_value = record.get('id')
            if isinstance(id_value, str) and id_value != "":
                last_id_index[id_value] = index

    groups = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            id_value = record.get('id')
            if isinstance(id_value, str) and id_value != "":
                if last_id_index.get(id_value) != index:
                    continue

        # Skip non-objects and records whose 'active' is false.
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Normalize the grouping key.
        if 'sku' in record:
            key = record['sku']
        elif 'item' in record:
            key = record['item']
        else:
            continue
        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue

        # Read and validate delta.
        if 'delta' not in record:
            delta = 1
        else:
            delta_value = record['delta']
            if type(delta_value) is int:
                delta = delta_value
            elif isinstance(delta_value, str):
                delta_value = delta_value.strip()
                if re.fullmatch(r'[+-]?[0-9]+', delta_value):
                    delta = int(delta_value)
                else:
                    continue
            else:
                continue

        groups[key] = groups.get(key, 0) + delta

    return [{'sku': key, 'total': total} for key, total in sorted(groups.items())]
