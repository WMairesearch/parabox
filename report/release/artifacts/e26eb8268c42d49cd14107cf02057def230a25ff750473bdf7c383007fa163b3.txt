import re

_BYTES_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Pass 1: find the last position of each record with a nonempty string id.
    last_for_id = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_for_id[rid] = index

    groups = {}
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        # Deduplication: keep only the last occurrence of each nonempty string id.
        rid = record.get('id')
        if isinstance(rid, str) and rid != '' and last_for_id[rid] != index:
            continue
        # Skip inactive records.
        if record.get('active') is False:
            continue
        # Read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Read 'bytes', defaulting to 1 only when absent.
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                amount = number
            elif isinstance(number, str):
                text = number.strip()
                if not _BYTES_RE.fullmatch(text):
                    continue
                amount = int(text, 10)
            else:
                continue
        else:
            amount = 1
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + amount, count + 1)

    return [
        {'path': name, 'total': total, 'count': count}
        for name, (total, count) in sorted(groups.items())
    ]
