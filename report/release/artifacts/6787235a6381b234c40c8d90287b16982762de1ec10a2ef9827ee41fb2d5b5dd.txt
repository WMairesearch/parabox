import re


def solve(value):
    # First, deduplicate by 'id': keep only the last occurrence of each
    # nonempty string id.  Records without such an id are kept independently.
    removed = set()
    last_pos = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                previous = last_pos.get(record_id)
                if previous is not None:
                    removed.add(previous)
                last_pos[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        if position in removed:
            continue
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str) or not name.strip():
            continue
        name = name.strip()

        # Read 'bytes', defaulting to 1 only when absent.
        if 'bytes' not in record:
            number = 1
        else:
            number = record['bytes']
            if isinstance(number, str):
                stripped = number.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    number = int(stripped)
                else:
                    continue
            if type(number) is not int:
                continue

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': groups[name][0], 'count': groups[name][1]}
        for name in sorted(groups)
    ]
