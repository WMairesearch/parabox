import json
import re


_PRIORITY_RE = re.compile(r'[+-]?[0-9]+')


def _last_occurrence_indices(value):
    """Return the indices to retain after id deduplication."""
    last_id_index = {}
    independent_indices = []
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            independent_indices.append(index)
            continue
        record_id = record.get('id')
        if isinstance(record_id, str) and record_id != '':
            last_id_index[record_id] = index
        else:
            independent_indices.append(index)
    return set(independent_indices) | set(last_id_index.values())


def solve(value):
    if isinstance(value, str):
        value = json.loads(value)
    if not isinstance(value, list):
        return []

    retained_indices = _last_occurrence_indices(value)
    groups = {}

    for index in sorted(retained_indices):
        record = value[index]

        if not isinstance(record, dict):
            continue

        if record.get('active') is False:
            continue

        if 'category' in record:
            category_value = record['category']
        else:
            category_value = record.get('kind')

        if not isinstance(category_value, str):
            continue
        category = category_value.strip()
        if not category:
            continue

        if 'priority' not in record:
            priority = 1
        else:
            priority_value = record['priority']
            if type(priority_value) is int:
                priority = priority_value
            elif isinstance(priority_value, str):
                stripped = priority_value.strip()
                if _PRIORITY_RE.fullmatch(stripped):
                    priority = int(stripped)
                else:
                    continue
            else:
                continue

        if category not in groups or priority > groups[category]:
            groups[category] = priority

    return [
        {'category': category, 'maximum': groups[category]}
        for category in sorted(groups)
    ]
