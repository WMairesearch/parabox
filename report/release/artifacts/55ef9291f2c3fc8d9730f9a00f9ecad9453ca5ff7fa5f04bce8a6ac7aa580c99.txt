import re


def solve(value):
    if not isinstance(value, list):
        return []

    # First rule: keep only the last occurrence of each nonempty string 'id'.
    # Records without such an id remain independent and keep their positions.
    last_id_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            id_value = record.get('id')
            if isinstance(id_value, str) and len(id_value) > 0:
                last_id_position[id_value] = position

    deduplicated = []
    for position, record in enumerate(value):
        if isinstance(record, dict):
            id_value = record.get('id')
            if isinstance(id_value, str) and len(id_value) > 0:
                if last_id_position[id_value] != position:
                    continue
        deduplicated.append(record)

    groups = {}
    for record in deduplicated:
        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records whose 'active' is JSON false.
        if record.get('active') is False:
            continue

        # Read category, falling back to kind only when category is absent.
        if 'category' in record:
            raw_key = record['category']
        else:
            raw_key = record.get('kind')

        if not isinstance(raw_key, str):
            continue
        category = raw_key.strip()
        if not category:
            continue

        # Read priority, defaulting to 1 only when absent.
        if 'priority' not in record:
            priority = 1
        else:
            raw_priority = record['priority']
            if type(raw_priority) is int:
                priority = raw_priority
            elif isinstance(raw_priority, str):
                stripped = raw_priority.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    priority = int(stripped)
                else:
                    continue
            else:
                continue

        if category not in groups or priority > groups[category]:
            groups[category] = priority

    return [{'category': category, 'maximum': groups[category]}
            for category in sorted(groups)]
