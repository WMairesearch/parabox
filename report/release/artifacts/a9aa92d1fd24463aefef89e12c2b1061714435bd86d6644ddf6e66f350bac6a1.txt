import re

_BYTES_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1 (first): retain only the last occurrence of each nonempty
    # string 'id'; records without such an id remain independent.
    last_for_id = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_for_id[rid] = index

    groups = {}
    for index, record in enumerate(value):
        # Deduplication happens before all other rules.
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '' and last_for_id[rid] != index:
                continue
        # Skip non-object records.
        if not isinstance(record, dict):
            continue
        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue
        # Read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        # Skip non-string or empty-after-stripping keys; strip whitespace.
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Read 'bytes', defaulting to 1 only when absent.
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                text = number.strip()
                if _BYTES_RE.fullmatch(text) is None:
                    continue
                number = int(text, 10)
            else:
                # Skip all other values, including null.
                continue
        else:
            number = 1
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    result = [{'path': name, 'total': totals[0], 'count': totals[1]}
              for name, totals in groups.items()]
    # Sort results lexicographically by 'path'.
    result.sort(key=lambda item: item['path'])
    return result
