import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def _dedupe(value):
    """Keep only the last record for each nonempty string 'id'.

    Records without such an id are independent and always kept.
    """
    last = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            ident = record.get('id')
            if isinstance(ident, str) and ident != '':
                last[ident] = position
    if not last:
        return list(value)
    keep = set(last.values())
    out = []
    for position, record in enumerate(value):
        ident = record.get('id') if isinstance(record, dict) else None
        if isinstance(ident, str) and ident != '':
            if position not in keep:
                continue
        out.append(record)
    return out


def _pick_path(record):
    if 'path' in record:
        name = record['path']
    else:
        name = record.get('route')
    if not isinstance(name, str):
        return None
    name = name.strip()
    if not name:
        return None
    return name


def _pick_bytes(record):
    """Return (ok, amount)."""
    if 'bytes' not in record:
        return True, 1
    number = record['bytes']
    if isinstance(number, bool):
        return False, 0
    if isinstance(number, int):
        return True, int(number)
    if isinstance(number, str):
        text = number.strip()
        if _INT_RE.fullmatch(text):
            return True, int(text, 10)
        return False, 0
    return False, 0


def solve(value):
    if not isinstance(value, list):
        return []
    records = _dedupe(value)
    groups = {}
    for record in records:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        name = _pick_path(record)
        if name is None:
            continue
        ok, amount = _pick_bytes(record)
        if not ok:
            continue
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + amount, count + 1)
    result = []
    for name in sorted(groups):
        total, count = groups[name]
        result.append({'path': name, 'total': total, 'count': count})
    return result
