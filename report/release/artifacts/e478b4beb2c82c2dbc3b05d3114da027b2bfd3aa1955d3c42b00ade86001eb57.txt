import re

def solve(value):
    if not isinstance(value, list):
        return []

    # 1. Deduplicate
    last_occurrence_idx = {}
    for idx, r in enumerate(value):
        if isinstance(r, dict) and 'id' in r:
            id_val = r['id']
            if isinstance(id_val, str) and id_val != '':
                last_occurrence_idx[id_val] = idx
                
    retained_records = []
    for idx, r in enumerate(value):
        if isinstance(r, dict) and 'id' in r:
            id_val = r['id']
            if isinstance(id_val, str) and id_val != '':
                if last_occurrence_idx[id_val] != idx:
                    continue
        retained_records.append(r)
            
    # 2. Process records
    groups = {}
    for r in retained_records:
        if not isinstance(r, dict):
            continue
        if r.get('active') is False:
            continue
            
        if 'path' in r:
            key_val = r['path']
        elif 'route' in r:
            key_val = r['route']
        else:
            continue
            
        if not isinstance(key_val, str):
            continue
            
        key_val = key_val.strip()
        if key_val == '':
            continue
            
        if 'bytes' not in r:
            bytes_val = 1
        else:
            bytes_raw = r['bytes']
            if type(bytes_raw) is int:
                bytes_val = bytes_raw
            elif isinstance(bytes_raw, str):
                b_str = bytes_raw.strip()
                if re.fullmatch(r'[+-]?[0-9]+', b_str):
                    bytes_val = int(b_str)
                else:
                    continue
            else:
                continue
                
        if key_val not in groups:
            groups[key_val] = {'total': 0, 'count': 0}
        groups[key_val]['total'] += bytes_val
        groups[key_val]['count'] += 1
        
    # 3. Format output
    result = []
    for k in sorted(groups.keys()):
        result.append({
            'path': k,
            'total': groups[k]['total'],
            'count': groups[k]['count']
        })
        
    return result
