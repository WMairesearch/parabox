import re

def solve(value):
    seen_ids = set()
    retained = []
    
    # Deduplicate before all other rules.
    # Scan backwards to retain only the last occurrence of each nonempty string 'id'.
    for rec in reversed(value):
        if isinstance(rec, dict):
            id_val = rec.get('id')
            if isinstance(id_val, str) and id_val != '':
                if id_val in seen_ids:
                    continue
                seen_ids.add(id_val)
        retained.append(rec)
        
    groups = {}
    for rec in retained:
        # Skip non-object records
        if not isinstance(rec, dict):
            continue
            
        # Skip records whose 'active' is false
        if rec.get('active') is False:
            continue
            
        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in rec:
            key_val = rec['sku']
        elif 'item' in rec:
            key_val = rec['item']
        else:
            continue
            
        # Skip non-string or empty-after-stripping keys
        if not isinstance(key_val, str):
            continue
        sku = key_val.strip()
        if not sku:
            continue
            
        # Read 'delta', defaulting to 1 only when absent.
        if 'delta' not in rec:
            delta = 1
        else:
            delta_val = rec['delta']
            if type(delta_val) is int:
                delta = delta_val
            elif isinstance(delta_val, str):
                d_str = delta_val.strip()
                if re.match(r'^[+-]?[0-9]+$', d_str):
                    delta = int(d_str)
                else:
                    continue
            else:
                continue
                
        groups[sku] = groups.get(sku, 0) + delta

    # Sort results lexicographically by 'sku'
    result = [{'sku': k, 'total': v} for k, v in sorted(groups.items())]
    return result
