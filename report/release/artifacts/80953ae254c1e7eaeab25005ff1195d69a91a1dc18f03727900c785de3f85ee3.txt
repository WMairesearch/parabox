import re

_INTEGER_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Pass 1: deduplicate - keep only the last occurrence of each nonempty
    # string 'id'. Records without such an id remain independent.
    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_index[record_id] = position

    groups = {}
    # Pass 2: filter and group.
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        record_id = record.get('id')
        if (isinstance(record_id, str) and record_id != ''
                and last_index[record_id] != position):
            continue
        if record.get('active') is False:
            continue
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'priority' in record:
            number = record['priority']
            if type(number) is int:
                pass
            elif isinstance(number, str):
                stripped = number.strip()
                if not _INTEGER_RE.fullmatch(stripped):
                    continue
                number = int(stripped, 10)
            else:
                continue
        else:
            number = 1
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    result = []
    for name in sorted(groups):
        result.append({'category': name, 'maximum': groups[name]})
    return result
