import re

def solve(value):
    if not isinstance(value, list):
        return []

    valid_ids = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val != "":
                valid_ids[id_val] = i

    groups = {}

    for i, record in enumerate(value):
        if not isinstance(record, dict):
            continue
            
        id_val = record.get('id')
        if isinstance(id_val, str) and id_val != "":
            if valid_ids[id_val] != i:
                continue

        if record.get('active') is False:
            continue

        if 'path' in record:
            path_val = record['path']
        elif 'route' in record:
            path_val = record['route']
        else:
            continue

        if not isinstance(path_val, str):
            continue

        path_val = path_val.strip()
        if not path_val:
            continue

        if 'bytes' not in record:
            bytes_val = 1
        else:
            b = record['bytes']
            if type(b) is int:
                bytes_val = b
            elif isinstance(b, str):
                b_str = b.strip()
                if re.match(r'^[+-]?[0-9]+$', b_str):
                    bytes_val = int(b_str)
                else:
                    continue
            else:
                continue

        if path_val not in groups:
            groups[path_val] = {'total': 0, 'count': 0}
        
        groups[path_val]['total'] += bytes_val
        groups[path_val]['count'] += 1

    result = []
    for path in sorted(groups.keys()):
        result.append({
            'path': path,
            'total': groups[path]['total'],
            'count': groups[path]['count']
        })
        
    return result
