import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def _dedup_id(record):
    """Return the dedup key for a record, or None when it has no usable id.

    Only a nonempty *string* 'id' participates in deduplication.  Ids are used
    verbatim (not stripped), so ' x ' and 'x' are different ids.
    """
    if not isinstance(record, dict):
        return None
    ident = record.get('id')
    if isinstance(ident, str) and ident != '':
        return ident
    return None


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []

    # Rule 1, applied before every other rule: retain only the last occurrence
    # of each nonempty string 'id'.  Records without such an id stay
    # independent.  Because dedup happens first, a later dropped copy (e.g. an
    # inactive or otherwise invalid record) still removes the earlier one.
    last_seen = {}
    for index, record in enumerate(value):
        ident = _dedup_id(record)
        if ident is not None:
            last_seen[ident] = index

    totals = {}
    for index, record in enumerate(value):
        # Rule 2a: skip non-object records.
        if not isinstance(record, dict):
            continue
        ident = _dedup_id(record)
        if ident is not None and last_seen[ident] != index:
            continue
        # Rule 2b: skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Rule 3: key is 'sku', falling back to 'item' only when 'sku' is
        # absent (present but invalid means skip, with no fallback).
        if 'sku' in record:
            name = record['sku']
        elif 'item' in record:
            name = record['item']
        else:
            continue
        if not isinstance(name, str):
            continue
        name = name.strip()
        if name == '':
            continue

        # Rule 4: 'delta', defaulting to 1 only when the key is absent.
        if 'delta' in record:
            number = record['delta']
            if isinstance(number, str):
                number = number.strip()
                if _INT_RE.fullmatch(number) is None:
                    continue
                number = int(number, 10)
            elif type(number) is not int:
                # Booleans, floats, null, lists, dicts, ... drop the record.
                continue
        else:
            number = 1

        totals[name] = totals.get(name, 0) + number

    # Rule 5: sorted lexicographically by 'sku'; [] when nothing remains.
    return [{'sku': name, 'total': totals[name]} for name in sorted(totals)]
