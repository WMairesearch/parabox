"""access_logs: group retained records by normalized path.

Implemented strictly from PROBLEM.md:
  * dedupe by last occurrence of each nonempty string 'id' FIRST
  * skip non-object records and records whose 'active' is false
  * key = 'path', falling back to 'route' only when 'path' is ABSENT
  * 'bytes' defaults to 1 only when ABSENT; ints (not bools) and
    [+-]?[0-9]+ strings (after stripping) accepted; everything else skipped
  * emit {'path', 'total', 'count'} sorted lexicographically by path
  * never mutate the input
"""

import re

_BYTES_RE = re.compile(r'[+-]?[0-9]+')


def _dedupe(records):
    """Keep only the last occurrence of each nonempty string id."""
    last = {}
    for index, record in enumerate(records):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last[rid] = index
    kept = []
    for index, record in enumerate(records):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '' and last[rid] != index:
                continue
        kept.append(record)
    return kept


def _bytes(record):
    """Return the byte value, or None when the record must be skipped."""
    if 'bytes' not in record:
        return 1
    number = record['bytes']
    if isinstance(number, bool):
        return None
    if isinstance(number, int):
        return number
    if isinstance(number, str):
        text = number.strip()
        if _BYTES_RE.fullmatch(text) is None:
            return None
        return int(text, 10)
    return None


def solve(value):
    if not isinstance(value, list):
        return []
    groups = {}
    for record in _dedupe(value):
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        name = record['path'] if 'path' in record else record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if name == '':
            continue
        number = _bytes(record)
        if number is None:
            continue
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)
    return [{'path': key, 'total': groups[key][0], 'count': groups[key][1]}
            for key in sorted(groups)]
