import re

_INTEGER = re.compile(r'[+-]?[0-9]+')


def _valid_id(record):
    """Return the record's id when it is a nonempty string, else None."""
    if not isinstance(record, dict):
        return None
    ident = record.get('id')
    if isinstance(ident, str) and ident != '':
        return ident
    return None


def _delta(record):
    """Return the integer delta for a record, or None when it must be skipped."""
    if 'delta' not in record:
        return 1
    value = record['delta']
    if type(value) is int:
        return value
    if isinstance(value, bool):
        return None
    if isinstance(value, str):
        text = value.strip()
        if _INTEGER.fullmatch(text):
            return int(text, 10)
        return None
    return None


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: keep only the last occurrence of each nonempty string id.
    last = {}
    for position, record in enumerate(value):
        ident = _valid_id(record)
        if ident is not None:
            last[ident] = position

    totals = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        ident = _valid_id(record)
        if ident is not None and last[ident] != position:
            continue
        if record.get('active') is False:
            continue
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        number = _delta(record)
        if number is None:
            continue
        totals[name] = totals.get(name, 0) + number

    return [{'sku': name, 'total': totals[name]} for name in sorted(totals)]
