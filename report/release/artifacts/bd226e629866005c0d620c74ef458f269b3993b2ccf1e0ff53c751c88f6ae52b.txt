import re

_DELTA_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Pass 1: last index of each nonempty-string 'id'
    last = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last[rid] = i

    groups = {}
    for i, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        rid = record.get('id')
        if isinstance(rid, str) and rid != '' and last.get(rid) != i:
            continue  # superseded by a later record with the same id
        if record.get('active') is False:
            continue
        name = record['sku'] if 'sku' in record else record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        number = record.get('delta', 1)
        if isinstance(number, bool):
            continue
        if isinstance(number, int):
            pass
        elif isinstance(number, str):
            text = number.strip()
            if _DELTA_RE.fullmatch(text) is None:
                continue
            number = int(text, 10)
        else:
            continue
        groups[name] = groups.get(name, 0) + number
    return [{'sku': sku, 'total': groups[sku]} for sku in sorted(groups)]
