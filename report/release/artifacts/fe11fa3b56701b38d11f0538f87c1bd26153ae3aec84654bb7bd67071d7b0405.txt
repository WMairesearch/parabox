import re

def solve(data):
    if not isinstance(data, list):
        return []

    # Deduplicate before all other rules.
    last_occurrence = {}
    for i, record in enumerate(data):
        if isinstance(record, dict) and 'id' in record:
            rec_id = record['id']
            if isinstance(rec_id, str) and rec_id != '':
                last_occurrence[rec_id] = i

    retained_records = []
    for i, record in enumerate(data):
        if isinstance(record, dict) and 'id' in record:
            rec_id = record['id']
            if isinstance(rec_id, str) and rec_id != '':
                if last_occurrence[rec_id] != i:
                    continue
        retained_records.append(record)

    grouped = {}

    for record in retained_records:
        if not isinstance(record, dict):
            continue
        
        if record.get('active') is False:
            continue
        
        if 'path' in record:
            raw_path = record['path']
        elif 'route' in record:
            raw_path = record['route']
        else:
            continue
        
        if not isinstance(raw_path, str):
            continue
        norm_path = raw_path.strip()
        if not norm_path:
            continue
            
        if 'bytes' not in record:
            bytes_val = 1
        else:
            raw_bytes = record['bytes']
            if isinstance(raw_bytes, bool):
                continue
            elif isinstance(raw_bytes, int):
                bytes_val = raw_bytes
            elif isinstance(raw_bytes, str):
                stripped = raw_bytes.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    bytes_val = int(stripped)
                else:
                    continue
            else:
                continue
                
        if norm_path not in grouped:
            grouped[norm_path] = {'total': 0, 'count': 0}
        grouped[norm_path]['total'] += bytes_val
        grouped[norm_path]['count'] += 1
        
    result = []
    for p in sorted(grouped.keys()):
        result.append({
            'path': p,
            'total': grouped[p]['total'],
            'count': grouped[p]['count']
        })
        
    return result
