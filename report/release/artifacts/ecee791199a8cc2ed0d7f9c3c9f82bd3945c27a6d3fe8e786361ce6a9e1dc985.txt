import re


def solve(value):
    if not isinstance(value, list):
        return []

    # Step 1: Retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Deduplicate before all other rules.
    last_seen = {}
    for i, rec in enumerate(value):
        if isinstance(rec, dict):
            rec_id = rec.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                last_seen[rec_id] = i

    groups = {}
    for i, rec in enumerate(value):
        if isinstance(rec, dict):
            rec_id = rec.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                if last_seen[rec_id] != i:
                    continue

        # Step 2: Skip non-object records and records whose 'active' is false.
        if not isinstance(rec, dict):
            continue

        if rec.get('active') is False:
            continue

        # Step 3: Read 'sku', falling back to 'item' only when 'sku' is absent.
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if 'sku' in rec:
            raw_sku = rec['sku']
        elif 'item' in rec:
            raw_sku = rec['item']
        else:
            continue

        if not isinstance(raw_sku, str):
            continue

        sku = raw_sku.strip()
        if not sku:
            continue

        # Step 4: Read 'delta', defaulting to 1 only when absent.
        # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace;
        # convert strings to base-10 integers. Skip all other values, including null.
        if 'delta' not in rec:
            delta = 1
        else:
            raw_delta = rec['delta']
            if isinstance(raw_delta, bool):
                continue
            elif type(raw_delta) is int:
                delta = raw_delta
            elif isinstance(raw_delta, str):
                s = raw_delta.strip()
                if re.fullmatch(r'[+-]?[0-9]+', s):
                    delta = int(s, 10)
                else:
                    continue
            else:
                continue

        # Step 5: Group retained records by normalized 'sku'
        groups[sku] = groups.get(sku, 0) + delta

    # Step 6: Sort results lexicographically by 'sku'
    result = []
    for sku in sorted(groups.keys()):
        result.append({'sku': sku, 'total': groups[sku]})

    return result
