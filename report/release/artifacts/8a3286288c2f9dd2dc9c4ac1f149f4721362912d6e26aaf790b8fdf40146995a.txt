import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def _dedup_id(record):
    """Return the dedup key for a record, or None when it has no usable id."""
    if not isinstance(record, dict):
        return None
    ident = record.get('id')
    if isinstance(ident, str) and ident != '':
        return ident
    return None


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []

    # Rule 1 (before everything else): keep only the last occurrence of each
    # nonempty string 'id'.  Records without such an id stay independent.
    last_seen = {}
    for index, record in enumerate(value):
        ident = _dedup_id(record)
        if ident is not None:
            last_seen[ident] = index

    groups = {}
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        ident = _dedup_id(record)
        if ident is not None and last_seen[ident] != index:
            continue
        if record.get('active') is False:
            continue

        # Key: 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            name = record['path']
        elif 'route' in record:
            name = record['route']
        else:
            continue
        if not isinstance(name, str):
            continue
        name = name.strip()
        if name == '':
            continue

        # Value: 'bytes', defaulting to 1 only when the key is absent.
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, str):
                number = number.strip()
                if _INT_RE.fullmatch(number) is None:
                    continue
                number = int(number)
            elif type(number) is not int:
                continue
        else:
            number = 1

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': total, 'count': count}
        for name, (total, count) in sorted(groups.items(), key=lambda item: item[0])
    ]
