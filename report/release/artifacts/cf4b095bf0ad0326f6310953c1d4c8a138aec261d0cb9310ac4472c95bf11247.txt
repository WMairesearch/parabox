import re


def solve(value):
    if not isinstance(value, list):
        return []

    # First rule: keep only the last occurrence of each nonempty string id.
    # Records without such an id are independent and all pass this step.
    last_id_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and len(record_id) > 0:
                last_id_position[record_id] = position

    deduplicated = []
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and len(record_id) > 0:
                if position != last_id_position[record_id]:
                    continue
        deduplicated.append(record)

    groups = {}
    for record in deduplicated:
        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records explicitly marked active=false.
        if record.get('active') is False:
            continue

        # Read sku, falling back to item only when sku is absent.
        if 'sku' in record:
            key = record['sku']
        else:
            key = record.get('item')

        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue

        # Read delta, defaulting to 1 only when absent.
        if 'delta' in record:
            number = record['delta']
            if type(number) is int:
                delta = number
            elif isinstance(number, str):
                number_string = number.strip()
                if re.fullmatch(r'[+-]?\d+', number_string):
                    delta = int(number_string)
                else:
                    continue
            else:
                continue
        else:
            delta = 1

        groups[key] = groups.get(key, 0) + delta

    return [{'sku': key, 'total': groups[key]} for key in sorted(groups)]
