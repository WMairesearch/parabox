import re

def solve(value):
    # Step 1: Deduplicate by nonempty string 'id'
    last_occurrence = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                last_occurrence[rec_id] = i
    
    groups = {}
    for i, record in enumerate(value):
        # Check deduplication
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                if last_occurrence[rec_id] != i:
                    continue
        
        # Step 2: Skip non-object records and 'active' is false
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
            
        # Step 3: Read 'sku'
        if 'sku' in record:
            sku = record['sku']
        elif 'item' in record:
            sku = record['item']
        else:
            continue
            
        if not isinstance(sku, str):
            continue
        sku = sku.strip()
        if not sku:
            continue
            
        # Step 4: Read 'delta'
        if 'delta' not in record:
            delta = 1
        else:
            delta_raw = record['delta']
            if type(delta_raw) is int:
                delta = delta_raw
            elif isinstance(delta_raw, str):
                delta_str = delta_raw.strip()
                if re.fullmatch(r'[+-]?[0-9]+', delta_str):
                    delta = int(delta_str)
                else:
                    continue
            else:
                continue
                
        # Group
        groups[sku] = groups.get(sku, 0) + delta
        
    return [{'sku': k, 'total': v} for k, v in sorted(groups.items())]
