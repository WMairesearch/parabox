import re

_INT_RE = re.compile(r'[+-]?[0-9]+\Z')


def _has_id(record):
    if not isinstance(record, dict):
        return False
    ident = record.get('id')
    return isinstance(ident, str) and ident != ''


def _dedupe(value):
    """Keep only the last occurrence of each nonempty string id."""
    seen = set()
    kept = []
    for record in reversed(value):
        if _has_id(record):
            ident = record['id']
            if ident in seen:
                continue
            seen.add(ident)
        kept.append(record)
    kept.reverse()
    return kept


def _delta(record):
    if 'delta' not in record:
        return 1
    number = record['delta']
    if isinstance(number, bool):
        return None
    if isinstance(number, int):
        return number
    if isinstance(number, str):
        text = number.strip()
        if _INT_RE.match(text):
            try:
                return int(text, 10)
            except ValueError:
                return None
    return None


def solve(value):
    if not isinstance(value, list):
        return []
    groups = {}
    for record in _dedupe(value):
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        name = record['sku'] if 'sku' in record else record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        number = _delta(record)
        if number is None:
            continue
        groups[name] = groups.get(name, 0) + number
    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
