import re


def solve(value):
    # Deduplicate records by the last occurrence of each nonempty string id.
    # Records without such an id are not deduplicated.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        identifier = record.get('id')
        if isinstance(identifier, str) and identifier:
            # Keep only the last occurrence for this id.  This rule runs
            # before every other skip/validation rule below.
            if latest[identifier] != position:
                continue

        if record.get('active') is False:
            continue

        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')

        if not isinstance(name, str) or not name.strip():
            continue
        name = name.strip()

        if 'bytes' in record:
            number = record['bytes']
        else:
            number = 1

        if isinstance(number, str):
            stripped = number.strip()
            if re.fullmatch(r'[+-]?[0-9]+', stripped):
                number = int(stripped)
        if type(number) is not int:
            continue

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': groups[name][0], 'count': groups[name][1]}
        for name in sorted(groups)
    ]
