import re

_PRIORITY_RE = re.compile(r'^[+-]?[0-9]+$')


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1: deduplicate by nonempty string 'id', retaining the last
    # occurrence. Records without such an id remain independent.
    last_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_position[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        record_id = record.get('id')
        if isinstance(record_id, str) and record_id != '':
            if last_position.get(record_id) != position:
                continue

        # Skip records whose 'active' is false.
        if record.get('active', True) is False:
            continue

        # Read 'category', falling back to 'kind' only when absent.
        if 'category' in record:
            name = record.get('category')
        else:
            name = record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # Read 'priority', defaulting to 1 only when absent.
        if 'priority' in record:
            number = record.get('priority')
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                text = number.strip()
                if not _PRIORITY_RE.match(text):
                    continue
                number = int(text, 10)
            else:
                continue
        else:
            number = 1

        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    result = []
    for name in sorted(groups):
        result.append({'category': name, 'maximum': groups[name]})
    return result
