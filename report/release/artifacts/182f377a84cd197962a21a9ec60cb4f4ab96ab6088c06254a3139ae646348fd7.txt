import re


def solve(value):
    if not isinstance(value, list):
        return []

    # Step 1: Deduplicate before all other rules.
    # First retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent.
    last_id_indices = {}
    for idx, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                last_id_indices[rec_id] = idx

    retained_records = []
    for idx, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                if last_id_indices[rec_id] != idx:
                    continue
        retained_records.append(record)

    # Step 2: Process retained records and aggregate.
    groups = {}
    for record in retained_records:
        # Skip non-object records and records whose 'active' is false.
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Read 'path', falling back to 'route' only when 'path' is absent.
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if 'path' in record:
            raw_path = record['path']
        elif 'route' in record:
            raw_path = record['route']
        else:
            continue

        if not isinstance(raw_path, str):
            continue
        norm_path = raw_path.strip()
        if not norm_path:
            continue

        # Read 'bytes', defaulting to 1 only when absent.
        # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace;
        # convert strings to base-10 integers. Skip all other values, including null.
        if 'bytes' not in record:
            bytes_val = 1
        else:
            raw_bytes = record['bytes']
            if type(raw_bytes) is int:
                bytes_val = raw_bytes
            elif isinstance(raw_bytes, str):
                stripped_bytes = raw_bytes.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped_bytes):
                    bytes_val = int(stripped_bytes, 10)
                else:
                    continue
            else:
                continue

        # Group retained records by normalized 'path'.
        if norm_path not in groups:
            groups[norm_path] = {'total': 0, 'count': 0}
        groups[norm_path]['total'] += bytes_val
        groups[norm_path]['count'] += 1

    # Sort results lexicographically by 'path'. Return [] when no records remain.
    result = []
    for p in sorted(groups.keys()):
        result.append({
            'path': p,
            'total': groups[p]['total'],
            'count': groups[p]['count']
        })
    return result
