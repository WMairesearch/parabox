import re

def solve(value):
    last_indices = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val != "":
                last_indices[id_val] = i

    retained = []
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val != "":
                if last_indices[id_val] != i:
                    continue
        retained.append(record)

    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        
        if 'sku' in record:
            sku_val = record['sku']
        elif 'item' in record:
            sku_val = record['item']
        else:
            continue
            
        if not isinstance(sku_val, str):
            continue
            
        sku_val = sku_val.strip()
        if not sku_val:
            continue
            
        if 'delta' not in record:
            delta_val = 1
        else:
            raw_delta = record['delta']
            if isinstance(raw_delta, bool):
                continue
            elif isinstance(raw_delta, int):
                delta_val = raw_delta
            elif isinstance(raw_delta, str):
                raw_delta = raw_delta.strip()
                if re.fullmatch(r'[+-]?[0-9]+', raw_delta):
                    delta_val = int(raw_delta)
                else:
                    continue
            else:
                continue
                
        groups[sku_val] = groups.get(sku_val, 0) + delta_val

    result = []
    for sku in sorted(groups.keys()):
        result.append({'sku': sku, 'total': groups[sku]})
        
    return result
