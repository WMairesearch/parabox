import re

_DELTA_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1 (before all others): retain only the last occurrence of each
    # nonempty string 'id'; records without such an id remain independent.
    last_positions = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_positions[rid] = position

    groups = {}
    for position, record in enumerate(value):
        # Skip non-object records.
        if not isinstance(record, dict):
            continue
        # Apply dedup: keep only the last record for each nonempty string id.
        rid = record.get('id')
        if isinstance(rid, str) and rid != '':
            if last_positions[rid] != position:
                continue
        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue
        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        # Skip non-string or empty-after-stripping keys.
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Read 'delta', defaulting to 1 only when absent.
        if 'delta' in record:
            number = record['delta']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                stripped = number.strip()
                if _DELTA_PATTERN.fullmatch(stripped) is None:
                    continue
                number = int(stripped, 10)
            else:
                continue
        else:
            number = 1
        groups[name] = groups.get(name, 0) + number

    # Sort results lexicographically by 'sku'.
    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
