import re
import sys

try:
    sys.set_int_max_str_digits(0)
except AttributeError:
    pass

PRIORITY_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    last = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last[rid] = index
    groups = {}
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        rid = record.get('id')
        if isinstance(rid, str) and rid != '' and last[rid] != index:
            continue
        if record.get('active') is False:
            continue
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'priority' in record:
            number = record['priority']
            if type(number) is int:
                pass
            elif isinstance(number, str):
                text = number.strip()
                if not PRIORITY_RE.fullmatch(text):
                    continue
                number = int(text)
            else:
                continue
        else:
            number = 1
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number
    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
