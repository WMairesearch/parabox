import re


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1: retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent.
    last_positions = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_positions[record_id] = position

    groups = {}

    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if position != last_positions[record_id]:
                    continue

        # Rule 2: skip non-object records and records whose 'active' is false.
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Rule 3: read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            path = record['path']
        else:
            path = record.get('route')
        if not isinstance(path, str):
            continue
        path = path.strip()
        if not path:
            continue

        # Rule 4: read 'bytes', defaulting to 1 only when absent.
        if 'bytes' in record:
            number = record['bytes']
        else:
            number = 1

        if isinstance(number, str):
            stripped = number.strip()
            if re.fullmatch(r'[+-]?[0-9]+', stripped):
                number = int(stripped)
            else:
                continue
        elif type(number) is not int:
            continue

        # Rule 5: group by normalized path.
        total, count = groups.get(path, (0, 0))
        groups[path] = (total + number, count + 1)

    return [
        {'path': path, 'total': groups[path][0], 'count': groups[path][1]}
        for path in sorted(groups)
    ]
