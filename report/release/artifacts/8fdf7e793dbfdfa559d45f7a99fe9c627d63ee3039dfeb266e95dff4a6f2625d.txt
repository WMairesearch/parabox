"""events: group records by normalized category and report the maximum priority."""

import re

_INTEGER = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []

    # 1) Deduplicate first: keep only the LAST occurrence of each nonempty
    #    string 'id'.  Records without such an id stay independent.
    latest = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = index

    groups = {}
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != index:
            continue

        # 2) Skip inactive records ('active' exactly false).
        if record.get('active') is False:
            continue

        # 3) Category, falling back to 'kind' only when 'category' is absent.
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # 4) Priority defaults to 1 only when the key is absent.
        if 'priority' in record:
            number = record['priority']
            if isinstance(number, str):
                text = number.strip()
                if not _INTEGER.fullmatch(text):
                    continue
                number = int(text)
            elif type(number) is not int:
                continue
        else:
            number = 1

        current = groups.get(name)
        if current is None or number > current:
            groups[name] = number

    return [{'category': key, 'maximum': groups[key]} for key in sorted(groups)]
