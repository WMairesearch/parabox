import re


def solve(value):
    # First pass: deduplicate by nonempty string 'id', keeping only the
    # last occurrence of each id.  Records without such an id are unaffected.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        # Deduplicate before all other rules.
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier:
            if latest[identifier] != position:
                continue

        if record.get('active') is False:
            continue

        name = record.get('path', record.get('route'))
        if not isinstance(name, str) or not name.strip():
            continue
        name = name.strip()

        number = record.get('bytes', 1)
        if isinstance(number, str) and re.fullmatch(r'[+-]?[0-9]+', number.strip()):
            number = int(number.strip())
        if type(number) is not int:
            continue

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': total, 'count': count}
        for name, (total, count) in sorted(groups.items())
    ]
