import re


def solve(value):
    if not isinstance(value, list):
        return []

    last_index = {}
    for idx, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                last_index[rec_id] = idx

    groups = {}
    for idx, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        rec_id = record.get('id')
        if isinstance(rec_id, str) and rec_id:
            if last_index[rec_id] != idx:
                continue

        if record.get('active') is False:
            continue

        if 'sku' in record:
            raw_key = record['sku']
        elif 'item' in record:
            raw_key = record['item']
        else:
            continue

        if not isinstance(raw_key, str):
            continue
        sku = raw_key.strip()
        if not sku:
            continue

        if 'delta' not in record:
            delta = 1
        else:
            raw_delta = record['delta']
            if isinstance(raw_delta, bool):
                continue
            elif isinstance(raw_delta, int):
                delta = raw_delta
            elif isinstance(raw_delta, str):
                s = raw_delta.strip()
                if re.fullmatch(r'[+-]?[0-9]+', s):
                    delta = int(s, 10)
                else:
                    continue
            else:
                continue

        groups[sku] = groups.get(sku, 0) + delta

    result = [{'sku': k, 'total': groups[k]} for k in sorted(groups.keys())]
    return result
