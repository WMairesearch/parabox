import re


def solve(value):
    # Deduplicate first: keep only the last occurrence of each nonempty string id.
    last_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                last_position[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                if last_position[identifier] != position:
                    continue

        # Skip non-object records and records whose active is false.
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Read category, falling back to kind only when category is absent.
        if 'category' in record:
            key = record['category']
        else:
            key = record.get('kind')

        if not isinstance(key, str) or not key.strip():
            continue
        key = key.strip()

        # Read priority, defaulting to 1 only when absent.
        if 'priority' not in record:
            priority = 1
        else:
            priority_value = record['priority']
            if isinstance(priority_value, bool):
                continue
            if isinstance(priority_value, int):
                priority = priority_value
            elif isinstance(priority_value, str):
                priority_value = priority_value.strip()
                if re.fullmatch(r'[+-]?[0-9]+', priority_value):
                    priority = int(priority_value)
                else:
                    continue
            else:
                continue

        if key in groups:
            if priority > groups[key]:
                groups[key] = priority
        else:
            groups[key] = priority

    return [
        {'category': key, 'maximum': groups[key]}
        for key in sorted(groups)
    ]
