import re

_INTEGER = re.compile(r'[+-]?[0-9]+')
_MISSING = object()


def _read_priority(record):
    """Return the record priority, or _MISSING when it must be skipped."""
    if 'priority' not in record:
        return 1
    value = record['priority']
    if type(value) is int:          # bools are rejected on purpose
        return value
    if isinstance(value, str):
        text = value.strip()
        if _INTEGER.fullmatch(text):
            return int(text, 10)
    return _MISSING


def _read_category(record):
    """Return the normalized category, or None when the record must be skipped."""
    if 'category' in record:
        raw = record['category']
    else:
        raw = record.get('kind')
    if not isinstance(raw, str):
        return None
    name = raw.strip()
    if not name:
        return None
    return name


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: keep only the last occurrence of every nonempty string id.
    keep = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        ident = record.get('id')
        if isinstance(ident, str) and ident != '':
            keep[ident] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        ident = record.get('id')
        if isinstance(ident, str) and ident != '' and keep.get(ident) != position:
            continue
        if record.get('active') is False:
            continue
        name = _read_category(record)
        if name is None:
            continue
        number = _read_priority(record)
        if number is _MISSING:
            continue
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [{'category': name, 'maximum': groups[name]}
            for name in sorted(groups)]
