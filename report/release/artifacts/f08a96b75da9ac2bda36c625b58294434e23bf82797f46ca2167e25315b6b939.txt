import re


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate before all other rules: for each nonempty string 'id',
    # keep only the last occurrence. Records without such an id remain
    # independent.
    last_id_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_id_position[record_id] = position

    retained = []
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if position != last_id_position[record_id]:
                    continue
        retained.append(record)

    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # 'path' wins whenever it is present; only fall back to 'route'
        # when 'path' is absent.
        if 'path' in record:
            path = record['path']
        else:
            path = record.get('route')

        if not isinstance(path, str) or not path.strip():
            continue
        path = path.strip()

        # 'bytes' defaults to 1 only when the key is absent.
        if 'bytes' not in record:
            number = 1
        else:
            raw = record['bytes']
            if type(raw) is int:
                number = raw
            elif isinstance(raw, str):
                raw = raw.strip()
                if not re.fullmatch(r'[+-]?[0-9]+', raw):
                    continue
                number = int(raw)
            else:
                continue

        total, count = groups.get(path, (0, 0))
        groups[path] = (total + number, count + 1)

    return [
        {'path': path, 'total': total, 'count': count}
        for path, (total, count) in sorted(groups.items())
    ]
