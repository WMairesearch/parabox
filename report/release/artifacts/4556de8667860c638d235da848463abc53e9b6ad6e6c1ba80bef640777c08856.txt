import re


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate before all other rules: keep only the last occurrence of each
    # record that has a nonempty string 'id'.  Records without such an id remain
    # independent (every one is kept).
    last_id_positions = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        if 'id' not in record:
            continue
        record_id = record['id']
        if isinstance(record_id, str) and record_id != '':
            last_id_positions[record_id] = position

    retained = []
    for position, record in enumerate(value):
        if isinstance(record, dict) and 'id' in record:
            record_id = record['id']
            if isinstance(record_id, str) and record_id != '':
                if last_id_positions.get(record_id) != position:
                    continue
        retained.append(record)

    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue

        if record.get('active') is False:
            continue

        # Read 'category', falling back to 'kind' only when 'category' is absent.
        if 'category' in record:
            raw_category = record['category']
        else:
            raw_category = record.get('kind')

        if not isinstance(raw_category, str):
            continue
        category = raw_category.strip()
        if not category:
            continue

        if 'priority' in record:
            priority = record['priority']
            if type(priority) is int and not isinstance(priority, bool):
                number = priority
            elif isinstance(priority, str):
                stripped = priority.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    number = int(stripped)
                else:
                    continue
            else:
                continue
        else:
            number = 1

        previous = groups.get(category)
        if previous is None or number > previous:
            groups[category] = number

    result = [{'category': category, 'maximum': groups[category]}
              for category in sorted(groups)]
    return result
