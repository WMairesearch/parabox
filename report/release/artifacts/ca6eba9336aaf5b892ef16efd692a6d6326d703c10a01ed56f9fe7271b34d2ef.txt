import re


def solve(value):
    # The input is a JSON list of records.
    if not isinstance(value, list):
        return []

    # Deduplicate before all other rules: for each nonempty string 'id',
    # keep only the last occurrence. Records without such an id stay.
    last_id_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                last_id_position[identifier] = position

    groups = {}

    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        # Keep only the last occurrence of each nonempty string id.
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier != '':
            if last_id_position.get(identifier) != position:
                continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            sku = record.get('sku')
        else:
            sku = record.get('item')

        if not isinstance(sku, str):
            continue
        sku = sku.strip()
        if not sku:
            continue

        # Read 'delta', defaulting to 1 only when absent.
        if 'delta' in record:
            delta = record['delta']
            if type(delta) is int:
                # bool is not accepted because type(True) is bool, not int.
                number = delta
            elif isinstance(delta, str):
                delta_string = delta.strip()
                if re.fullmatch(r'[+-]?[0-9]+', delta_string):
                    number = int(delta_string, 10)
                else:
                    continue
            else:
                continue
        else:
            number = 1

        groups[sku] = groups.get(sku, 0) + number

    # Sort results lexicographically by normalized sku.
    result = [
        {'sku': sku, 'total': groups[sku]}
        for sku in sorted(groups)
    ]
    return result
