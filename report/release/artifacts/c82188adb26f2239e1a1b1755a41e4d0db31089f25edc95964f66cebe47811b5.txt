import re


def solve(value):
    # Deduplicate first: keep only the last occurrence of each nonempty string id.
    # Records without such an id remain independent.
    last_id_position = {}
    dropped_positions = set()
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        record_id = record.get('id')
        if isinstance(record_id, str) and record_id != '':
            previous = last_id_position.get(record_id)
            if previous is not None:
                dropped_positions.add(previous)
            last_id_position[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        if position in dropped_positions:
            continue
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Read sku, falling back to item only when sku is absent.
        if 'sku' in record:
            sku = record['sku']
        else:
            sku = record.get('item')
        if not isinstance(sku, str):
            continue
        sku = sku.strip()
        if not sku:
            continue

        # Read delta, defaulting to 1 only when absent.
        if 'delta' not in record:
            delta = 1
        else:
            delta = record['delta']

        if isinstance(delta, str):
            stripped = delta.strip()
            if re.fullmatch(r'[+-]?[0-9]+', stripped):
                delta = int(stripped)
            else:
                continue
        elif type(delta) is int:
            # bool is a subclass of int, so this deliberately rejects booleans.
            pass
        else:
            continue

        groups[sku] = groups.get(sku, 0) + delta

    return [{'sku': sku, 'total': groups[sku]} for sku in sorted(groups)]
