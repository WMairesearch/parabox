"""Inventory aggregation per PROBLEM.md."""
import re

_DELTA_RE = re.compile(r'[+-]?[0-9]+')


def _dedupe(value):
    """Retain only the last occurrence of each nonempty string id."""
    last = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last[rid] = index
    kept = []
    for index, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                if last[rid] != index:
                    continue
        kept.append(record)
    return kept


def _delta(record):
    """Return the integer delta, or None when the record must be skipped."""
    if 'delta' not in record:
        return 1
    number = record['delta']
    if isinstance(number, bool):
        return None
    if isinstance(number, int):
        return number
    if isinstance(number, str):
        text = number.strip()
        if _DELTA_RE.fullmatch(text) is None:
            return None
        return int(text, 10)
    return None


def solve(value):
    if not isinstance(value, list):
        return []
    groups = {}
    for record in _dedupe(value):
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        name = record['sku'] if 'sku' in record else record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if name == '':
            continue
        number = _delta(record)
        if number is None:
            continue
        groups[name] = groups.get(name, 0) + number
    return [{'sku': key, 'total': groups[key]} for key in sorted(groups)]
