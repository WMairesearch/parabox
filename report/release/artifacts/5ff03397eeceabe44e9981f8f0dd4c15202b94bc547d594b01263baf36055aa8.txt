import re

def solve(value):
    seen_ids = set()
    retained_indices = set()
    
    # First retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Deduplicate before all other rules.
    for i in range(len(value) - 1, -1, -1):
        record = value[i]
        id_val = None
        if isinstance(record, dict) and 'id' in record:
            val = record['id']
            if isinstance(val, str) and val != "":
                id_val = val
        
        if id_val is not None:
            if id_val in seen_ids:
                continue
            seen_ids.add(id_val)
        retained_indices.add(i)
        
    # Maintain original order, though not strictly required for grouping, it's safer
    retained = [value[i] for i in range(len(value)) if i in retained_indices]
    
    groups = {}
    
    for record in retained:
        # Skip non-object records
        if not isinstance(record, dict):
            continue
            
        # Skip records whose 'active' is false (boolean false)
        if record.get('active') is False:
            continue
            
        # Read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            key = record['path']
        elif 'route' in record:
            key = record['route']
        else:
            continue
            
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if not isinstance(key, str):
            continue
        key = key.strip()
        if key == "":
            continue
            
        # Read 'bytes', defaulting to 1 only when absent.
        if 'bytes' not in record:
            bytes_val = 1
        else:
            raw_bytes = record['bytes']
            if type(raw_bytes) is int:
                bytes_val = raw_bytes
            elif type(raw_bytes) is str:
                raw_bytes_stripped = raw_bytes.strip()
                # Accept strings matching [+-]?[0-9]+ after stripping whitespace
                if re.fullmatch(r'[+-]?[0-9]+', raw_bytes_stripped):
                    bytes_val = int(raw_bytes_stripped)
                else:
                    continue
            else:
                continue
                
        # Group retained records by normalized 'path'.
        total, count = groups.get(key, (0, 0))
        groups[key] = (total + bytes_val, count + 1)
        
    result = []
    # Sort results lexicographically by 'path'.
    for key in sorted(groups.keys()):
        result.append({
            'path': key,
            'total': groups[key][0],
            'count': groups[key][1]
        })
        
    return result
