import re


def solve(value):
    # Deduplicate before all other rules: keep only the last occurrence of each
    # nonempty string 'id'.  Records without such an id remain independent.
    last_id_index = {}
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        record_id = record.get('id')
        if isinstance(record_id, str) and record_id:
            last_id_index[record_id] = index

    retained = []
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id:
                if last_id_index.get(record_id) != index:
                    continue
        retained.append(record)

    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Read 'category', falling back to 'kind' only when 'category' is absent.
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')

        if not isinstance(name, str) or not name.strip():
            continue
        name = name.strip()

        # Read 'priority', defaulting to 1 only when absent.
        if 'priority' not in record:
            number = 1
        else:
            number = record['priority']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                number = number
            elif isinstance(number, str):
                stripped = number.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    number = int(stripped)
                else:
                    continue
            else:
                continue

        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [
        {'category': name, 'maximum': groups[name]}
        for name in sorted(groups)
    ]
