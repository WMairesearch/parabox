import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if isinstance(value, (str, bytes)) or not hasattr(value, '__iter__'):
        return []
    records = list(value)

    # Deduplicate first: keep only the last occurrence of each nonempty
    # string id. Records without such an id remain independent.
    last_occurrence = {}
    for index, record in enumerate(records):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_occurrence[rid] = index

    groups = {}
    for index, record in enumerate(records):
        if not isinstance(record, dict):
            continue
        rid = record.get('id')
        if isinstance(rid, str) and rid != '' and last_occurrence[rid] != index:
            continue
        if record.get('active') is False:
            continue
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if name == '':
            continue
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, str):
                stripped = number.strip()
                if _INT_RE.fullmatch(stripped) is None:
                    continue
                number = int(stripped, 10)
            elif isinstance(number, bool) or not isinstance(number, int):
                continue
        else:
            number = 1
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': total, 'count': count}
        for name, (total, count) in sorted(groups.items())
    ]
