import re


def solve(value):
    if not isinstance(value, list):
        return []

    last_indices = {}
    for idx, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                last_indices[rec_id] = idx

    groups = {}
    for idx, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        rec_id = record.get('id')
        if isinstance(rec_id, str) and rec_id:
            if last_indices[rec_id] != idx:
                continue

        if record.get('active') is False:
            continue

        if 'category' in record:
            raw_key = record['category']
        elif 'kind' in record:
            raw_key = record['kind']
        else:
            continue

        if not isinstance(raw_key, str):
            continue
        category = raw_key.strip()
        if not category:
            continue

        if 'priority' not in record:
            priority = 1
        else:
            raw_priority = record['priority']
            if isinstance(raw_priority, bool):
                continue
            elif isinstance(raw_priority, int):
                priority = raw_priority
            elif isinstance(raw_priority, str):
                stripped = raw_priority.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    priority = int(stripped, 10)
                else:
                    continue
            else:
                continue

        if category not in groups:
            groups[category] = priority
        else:
            groups[category] = max(groups[category], priority)

    return [{'category': cat, 'maximum': groups[cat]} for cat in sorted(groups.keys())]
