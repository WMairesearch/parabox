import re

def solve(value):
    seen_ids = set()
    keep_indices = []
    for i in range(len(value)-1, -1, -1):
        record = value[i]
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val != '':
                if id_val in seen_ids:
                    continue
                seen_ids.add(id_val)
        keep_indices.append(i)
    
    keep_indices.reverse()
    
    groups = {}
    for i in keep_indices:
        record = value[i]
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        
        if 'sku' in record:
            name = record['sku']
        elif 'item' in record:
            name = record['item']
        else:
            continue
            
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
            
        if 'delta' not in record:
            delta = 1
        else:
            delta = record['delta']
            
        if type(delta) is int:
            pass
        elif isinstance(delta, str):
            delta_str = delta.strip()
            if re.fullmatch(r'[+-]?[0-9]+', delta_str):
                delta = int(delta_str)
            else:
                continue
        else:
            continue
            
        groups[name] = groups.get(name, 0) + delta
        
    result = []
    for name in sorted(groups.keys()):
        result.append({'sku': name, 'total': groups[name]})
    return result
