import re


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: for each nonempty string 'id', keep only the last
    # occurrence (in the original input order).  Records without such an id are
    # left alone.
    last_id_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_id_position[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        record_id = record.get('id')
        if isinstance(record_id, str) and record_id != '':
            if last_id_position.get(record_id) != position:
                continue

        # Skip records whose 'active' is JSON false.
        if record.get('active') is False:
            continue

        # Read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            key = record['path']
        elif 'route' in record:
            key = record['route']
        else:
            continue

        # Keys must be strings that are nonempty after stripping.
        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue

        # Read 'bytes', defaulting to 1 only when absent.
        if 'bytes' in record:
            raw_bytes = record['bytes']
            if type(raw_bytes) is int:
                number = raw_bytes
            elif isinstance(raw_bytes, str):
                stripped = raw_bytes.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    number = int(stripped)
                else:
                    continue
            else:
                continue
        else:
            number = 1

        total, count = groups.get(key, (0, 0))
        groups[key] = (total + number, count + 1)

    result = [
        {'path': key, 'total': total, 'count': count}
        for key, (total, count) in sorted(groups.items())
    ]
    return result
