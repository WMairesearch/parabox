import re


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []

    # First retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Deduplicate before all other rules.
    last_indices = {}
    for idx, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                last_indices[rec_id] = idx

    retained_records = []
    for idx, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                if last_indices[rec_id] != idx:
                    continue
        retained_records.append(record)

    # Process retained records
    groups = {}
    for record in retained_records:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        if 'sku' in record:
            key = record['sku']
        elif 'item' in record:
            key = record['item']
        else:
            continue

        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue

        if 'delta' not in record:
            delta = 1
        else:
            raw_delta = record['delta']
            if isinstance(raw_delta, int) and not isinstance(raw_delta, bool):
                delta = raw_delta
            elif isinstance(raw_delta, str):
                stripped_delta = raw_delta.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped_delta):
                    delta = int(stripped_delta, 10)
                else:
                    continue
            else:
                continue

        groups[key] = groups.get(key, 0) + delta

    result = []
    for sku in sorted(groups.keys()):
        result.append({'sku': sku, 'total': groups[sku]})
    return result
