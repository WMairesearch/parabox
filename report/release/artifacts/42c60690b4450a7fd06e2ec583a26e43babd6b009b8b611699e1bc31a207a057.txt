import re

def solve(value):
    last_indices = {}
    for i, record in enumerate(value):
        if isinstance(record, dict) and 'id' in record:
            id_val = record['id']
            if isinstance(id_val, str) and id_val != "":
                last_indices[id_val] = i

    groups = {}
    for i, record in enumerate(value):
        if isinstance(record, dict) and 'id' in record:
            id_val = record['id']
            if isinstance(id_val, str) and id_val != "":
                if last_indices[id_val] != i:
                    continue
        
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        if 'path' in record:
            path_val = record['path']
        elif 'route' in record:
            path_val = record['route']
        else:
            continue
            
        if not isinstance(path_val, str):
            continue
        path_val = path_val.strip()
        if path_val == "":
            continue
            
        if 'bytes' not in record:
            bytes_val = 1
        else:
            raw_bytes = record['bytes']
            if type(raw_bytes) is int:
                bytes_val = raw_bytes
            elif type(raw_bytes) is str:
                stripped_bytes = raw_bytes.strip()
                if re.match(r'^[+-]?[0-9]+$', stripped_bytes):
                    bytes_val = int(stripped_bytes)
                else:
                    continue
            else:
                continue
                
        if path_val not in groups:
            groups[path_val] = {'total': 0, 'count': 0}
        
        groups[path_val]['total'] += bytes_val
        groups[path_val]['count'] += 1

    result = []
    for path in sorted(groups.keys()):
        result.append({
            'path': path,
            'total': groups[path]['total'],
            'count': groups[path]['count']
        })
        
    return result
