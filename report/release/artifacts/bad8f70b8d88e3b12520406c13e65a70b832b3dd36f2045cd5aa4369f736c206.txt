import re

def solve(value):
    if not isinstance(value, list):
        return []
        
    last_occurrence = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            if 'id' in record:
                id_val = record['id']
                if isinstance(id_val, str) and id_val:
                    last_occurrence[id_val] = i

    groups = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            if 'id' in record:
                id_val = record['id']
                if isinstance(id_val, str) and id_val:
                    if last_occurrence[id_val] != i:
                        continue
        
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        if 'category' in record:
            cat = record['category']
        elif 'kind' in record:
            cat = record['kind']
        else:
            continue
            
        if not isinstance(cat, str):
            continue
        cat = cat.strip()
        if not cat:
            continue
            
        if 'priority' not in record:
            priority = 1
        else:
            p = record['priority']
            if isinstance(p, int) and not isinstance(p, bool):
                priority = p
            elif isinstance(p, str):
                p_stripped = p.strip()
                if re.fullmatch(r'[+-]?[0-9]+', p_stripped):
                    priority = int(p_stripped)
                else:
                    continue
            else:
                continue
                
        if cat not in groups:
            groups[cat] = priority
        else:
            if priority > groups[cat]:
                groups[cat] = priority

    result = []
    for cat in sorted(groups.keys()):
        result.append({'category': cat, 'maximum': groups[cat]})
    return result
