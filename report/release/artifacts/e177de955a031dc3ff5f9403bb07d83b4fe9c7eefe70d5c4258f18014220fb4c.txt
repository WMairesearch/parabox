import re


def solve(value):
    if not isinstance(value, list):
        return []

    # First, retain only the last occurrence of each nonempty string 'id'.
    # Records without such an id remain independent.
    last_id_index = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_id_index[record_id] = index

    groups = {}

    for index, record in enumerate(value):
        # Deduplicate by id before applying all other rules.
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if last_id_index[record_id] != index:
                    continue

        # Skip non-object records and records whose 'active' is false.
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            path = record['path']
        else:
            path = record.get('route')

        if not isinstance(path, str):
            continue
        path = path.strip()
        if not path:
            continue

        # Read 'bytes', defaulting to 1 only when absent.
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, str):
                stripped = number.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    number = int(stripped)
                else:
                    continue
            if type(number) is not int:
                continue
        else:
            number = 1

        total, count = groups.get(path, (0, 0))
        groups[path] = (total + number, count + 1)

    return [
        {'path': path, 'total': groups[path][0], 'count': groups[path][1]}
        for path in sorted(groups)
    ]
