import re

_INTEGER = re.compile(r'[+-]?[0-9]+\Z')


def _valid_id(record):
    """Return the record's id when it is a nonempty string, else None."""
    if not isinstance(record, dict):
        return None
    ident = record.get('id')
    if isinstance(ident, str) and ident:
        return ident
    return None


def _category(record):
    """Normalized category (falling back to 'kind' only when absent)."""
    if 'category' in record:
        name = record['category']
    elif 'kind' in record:
        name = record['kind']
    else:
        return None
    if not isinstance(name, str):
        return None
    name = name.strip()
    if not name:
        return None
    return name


def _priority(record):
    """Priority value, defaulting to 1 only when the key is absent."""
    if 'priority' not in record:
        return 1
    value = record['priority']
    if isinstance(value, bool):
        return None
    if isinstance(value, int):
        return value
    if isinstance(value, str):
        text = value.strip()
        if _INTEGER.match(text):
            return int(text, 10)
    return None


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: only the last occurrence of each nonempty string id
    # is retained; records without such an id stay independent.
    last_position = {}
    for position, record in enumerate(value):
        ident = _valid_id(record)
        if ident is not None:
            last_position[ident] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        ident = _valid_id(record)
        if ident is not None and last_position[ident] != position:
            continue
        if record.get('active') is False:
            continue
        name = _category(record)
        if name is None:
            continue
        number = _priority(record)
        if number is None:
            continue
        if name not in groups or number > groups[name]:
            groups[name] = number

    return [{'category': name, 'maximum': groups[name]}
            for name in sorted(groups)]
