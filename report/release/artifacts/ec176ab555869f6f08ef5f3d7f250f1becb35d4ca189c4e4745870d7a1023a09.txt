import re


def solve(value):
    """Group retained JSON records by category, returning maximum priority."""
    if not isinstance(value, list):
        return []

    # Deduplicate before all other rules: keep only the last occurrence of
    # each nonempty string 'id'; records without such an id remain independent.
    last_occurrence = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_occurrence[record_id] = index

    deduplicated = []
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if last_occurrence[record_id] != index:
                    continue
        deduplicated.append(record)

    groups = {}
    for record in deduplicated:
        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'category', falling back to 'kind' only when 'category' is absent.
        if 'category' in record:
            raw_key = record['category']
        elif 'kind' in record:
            raw_key = record['kind']
        else:
            continue

        # Skip non-string or empty-after-stripping keys; strip whitespace.
        if not isinstance(raw_key, str):
            continue
        key = raw_key.strip()
        if key == '':
            continue

        # Read 'priority', defaulting to 1 only when absent.
        if 'priority' not in record:
            number = 1
        else:
            number = record['priority']
            if type(number) is int:
                # Integers (not booleans) are accepted as-is.
                pass
            elif isinstance(number, str):
                stripped = number.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    number = int(stripped)
                else:
                    continue
            else:
                continue

        if key in groups:
            if number > groups[key]:
                groups[key] = number
        else:
            groups[key] = number

    return [
        {'category': key, 'maximum': groups[key]}
        for key in sorted(groups)
    ]
