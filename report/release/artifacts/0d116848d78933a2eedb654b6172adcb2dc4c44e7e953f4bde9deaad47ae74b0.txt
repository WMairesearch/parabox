"""Inventory grouping per PROBLEM.md."""
import re

_INTEGER = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []

    # Rule 1: deduplicate first - keep only the LAST occurrence of each
    # nonempty string 'id'. Records without such an id stay independent.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        # Rule 2: skip non-object records and records whose active is false.
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != position:
            continue
        if record.get('active') is False:
            continue

        # Rule 3: 'sku', falling back to 'item' only when 'sku' is absent.
        name = record['sku'] if 'sku' in record else record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # Rule 4: 'delta', defaulting to 1 only when absent.
        if 'delta' in record:
            number = record['delta']
            if isinstance(number, str):
                text = number.strip()
                if not _INTEGER.fullmatch(text):
                    continue
                number = int(text)
            elif type(number) is not int:
                continue
        else:
            number = 1

        groups[name] = groups.get(name, 0) + number

    # Rule 5: sort lexicographically by sku.
    return [{'sku': key, 'total': total} for key, total in sorted(groups.items())]
