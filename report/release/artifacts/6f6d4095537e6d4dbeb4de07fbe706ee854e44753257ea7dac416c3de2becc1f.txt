import re

_BYTES_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Pass 1: deduplicate - retain only the last occurrence of each
    # nonempty string 'id'. Records without such an id stay independent.
    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_index[rid] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        rid = record.get('id')
        if isinstance(rid, str) and rid != '' and last_index[rid] != position:
            continue  # superseded by a later record with the same id
        if record.get('active') is False:
            continue
        name = record['path'] if 'path' in record else record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        number = record['bytes'] if 'bytes' in record else 1
        if isinstance(number, bool):
            continue
        if isinstance(number, int):
            pass
        elif isinstance(number, str):
            text = number.strip()
            if not _BYTES_RE.fullmatch(text):
                continue
            number = int(text, 10)
        else:
            continue
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [{'path': path, 'total': total, 'count': count}
            for path, (total, count) in sorted(groups.items())]
