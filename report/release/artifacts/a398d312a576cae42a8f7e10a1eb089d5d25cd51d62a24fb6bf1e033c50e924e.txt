import re

_INTEGER = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1: keep only the last occurrence of each nonempty string 'id'.
    last_seen = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            ident = record.get('id')
            if isinstance(ident, str) and ident != '':
                last_seen[ident] = index

    groups = {}
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        ident = record.get('id')
        if isinstance(ident, str) and ident != '' and last_seen.get(ident) != index:
            continue

        # Rule 2: drop records explicitly marked inactive.
        if record.get('active', True) is False:
            continue

        # Rule 3: 'path' wins; 'route' is used only when 'path' is absent.
        key = record['path'] if 'path' in record else record.get('route')
        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue

        # Rule 4: 'bytes' defaults to 1 only when the key is absent.
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                text = number.strip()
                if _INTEGER.fullmatch(text) is None:
                    continue
                number = int(text)
            else:
                continue
        else:
            number = 1

        total, count = groups.get(key, (0, 0))
        groups[key] = (total + number, count + 1)

    return [
        {'path': key, 'total': groups[key][0], 'count': groups[key][1]}
        for key in sorted(groups)
    ]
