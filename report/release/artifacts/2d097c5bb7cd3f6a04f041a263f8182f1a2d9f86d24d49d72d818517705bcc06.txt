import re


def solve(value):
    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        name = record.get('category')
        if not isinstance(name, str) or not name.strip():
            continue
        number = record.get('priority', 0)
        if type(number) is not int:
            continue
        groups[name] = max(groups[name], number) if name in groups else number
    result = []
    for name in groups:
        result.append({'category': name, 'maximum': groups[name]})
    return result
