import re

def solve(value):
    last_occurrence = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id:
                last_occurrence[record_id] = i
                
    groups = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id:
                if last_occurrence[record_id] != i:
                    continue
                    
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        if 'sku' in record:
            name = record['sku']
        elif 'item' in record:
            name = record['item']
        else:
            continue
            
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
            
        if 'delta' not in record:
            delta = 1
        else:
            delta = record['delta']
            
        if type(delta) is int:
            pass
        elif isinstance(delta, str):
            s = delta.strip()
            if re.fullmatch(r'[+-]?[0-9]+', s):
                delta = int(s)
            else:
                continue
        else:
            continue
            
        groups[name] = groups.get(name, 0) + delta
        
    result = []
    for name in sorted(groups.keys()):
        result.append({'sku': name, 'total': groups[name]})
        
    return result
