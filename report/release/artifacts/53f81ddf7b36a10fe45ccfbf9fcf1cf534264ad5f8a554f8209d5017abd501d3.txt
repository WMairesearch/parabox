"""Group access-log records by normalized path."""

import re

_INT_RE = re.compile(r'[+-]?[0-9]+\Z')


def _record_id(record):
    """Return the record's deduplication id, or None when it has none."""
    if not isinstance(record, dict):
        return None
    identifier = record.get('id')
    if isinstance(identifier, str) and identifier != '':
        return identifier
    return None


def _as_int(raw):
    """Convert a raw 'bytes' value to a base-10 int, or None if invalid."""
    if isinstance(raw, bool):
        return None
    if isinstance(raw, int):
        return raw
    if isinstance(raw, str):
        text = raw.strip()
        if _INT_RE.match(text):
            return int(text, 10)
    return None


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: keep only the last occurrence of each nonempty string id.
    last_position = {}
    for position, record in enumerate(value):
        identifier = _record_id(record)
        if identifier is not None:
            last_position[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = _record_id(record)
        if identifier is not None and last_position[identifier] != position:
            continue
        if record.get('active') is False:
            continue
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'bytes' in record:
            number = _as_int(record['bytes'])
            if number is None:
                continue
        else:
            number = 1
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': groups[name][0], 'count': groups[name][1]}
        for name in sorted(groups)
    ]
