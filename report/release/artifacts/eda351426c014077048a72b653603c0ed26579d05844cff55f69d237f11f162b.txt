import re

BYTES_RE = re.compile(r'^[+-]?[0-9]+$')


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []

    # 1. Deduplication:
    # First retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Deduplicate before all other rules.
    last_indices = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                last_indices[rec_id] = i

    groups = {}

    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                if last_indices[rec_id] != i:
                    continue
        else:
            # Skip non-object records
            continue

        # Skip records whose 'active' is false (boolean literal False)
        if record.get('active') is False:
            continue

        # Read 'path', falling back to 'route' only when 'path' is absent.
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if 'path' in record:
            raw_path = record['path']
        elif 'route' in record:
            raw_path = record['route']
        else:
            continue

        if not isinstance(raw_path, str):
            continue
        path = raw_path.strip()
        if not path:
            continue

        # Read 'bytes', defaulting to 1 only when absent.
        # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace;
        # convert strings to base-10 integers. Skip all other values, including null.
        if 'bytes' not in record:
            byte_val = 1
        else:
            raw_bytes = record['bytes']
            if isinstance(raw_bytes, int) and not isinstance(raw_bytes, bool):
                byte_val = raw_bytes
            elif isinstance(raw_bytes, str):
                s = raw_bytes.strip()
                if BYTES_RE.fullmatch(s):
                    byte_val = int(s, 10)
                else:
                    continue
            else:
                continue

        # Group retained records by normalized 'path'
        if path not in groups:
            groups[path] = [0, 0]
        groups[path][0] += byte_val
        groups[path][1] += 1

    result = []
    for path in sorted(groups.keys()):
        result.append({
            'path': path,
            'total': groups[path][0],
            'count': groups[path][1]
        })
    return result
