"""Events grouping per PROBLEM.md.

solve(value) -> list of {"category": str, "maximum": int}, sorted by category.

Rules (applied in this order):
1. Deduplicate FIRST: retain only the last occurrence of each nonempty *string*
   'id'; records lacking such an id stay independent.  A record dropped later
   does not revive its earlier duplicates.
2. Skip non-object records and records whose 'active' is False.
3. Category: use 'category' when the key is present, else fall back to 'kind'
   (presence-based, not None-based).  Must be a string; strip it; skip when the
   result is empty.
4. Priority: default 1 only when the key is absent.  Accept ints (not bools) and
   strings matching [+-]?[0-9]+ after stripping; anything else (incl. null) skips
   the record.
5. Group by the normalized category, keeping the maximum priority.
Never mutate the input.
"""

import re

_PRIORITY_RE = re.compile(r'[+-]?[0-9]+')


def _has_string_id(record):
    identifier = record.get('id')
    return isinstance(identifier, str) and identifier != ''


def _dedupe(records):
    """Retain only the last occurrence of each nonempty string 'id'."""
    last_index = {}
    for index, record in enumerate(records):
        if isinstance(record, dict) and _has_string_id(record):
            last_index[record['id']] = index
    if not last_index:
        return list(records)
    kept = []
    for index, record in enumerate(records):
        if isinstance(record, dict) and _has_string_id(record):
            if last_index[record['id']] != index:
                continue
        kept.append(record)
    return kept


def _category(record):
    if 'category' in record:
        name = record['category']
    else:
        name = record.get('kind')
    if not isinstance(name, str):
        return None
    name = name.strip()
    if not name:
        return None
    return name


def _priority(record):
    if 'priority' not in record:
        return 1
    number = record['priority']
    if isinstance(number, bool):
        return None
    if isinstance(number, int):
        return number
    if isinstance(number, str):
        text = number.strip()
        if _PRIORITY_RE.fullmatch(text) is None:
            return None
        return int(text, 10)
    return None


def solve(value):
    if not isinstance(value, list):
        return []
    groups = {}
    for record in _dedupe(value):
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        name = _category(record)
        if name is None:
            continue
        number = _priority(record)
        if number is None:
            continue
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number
    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
