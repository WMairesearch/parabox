import re

_PRIORITY_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1 (before all others): retain only the last occurrence of each
    # nonempty string 'id'. Records without such an id remain independent.
    last_index = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_index[record_id] = index

    groups = {}
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        record_id = record.get('id')
        if (isinstance(record_id, str) and record_id != ''
                and last_index[record_id] != index):
            continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # 'category' with fallback to 'kind' only when 'category' is absent.
        if 'category' in record:
            name = record['category']
        elif 'kind' in record:
            name = record['kind']
        else:
            continue
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # 'priority', defaulting to 1 only when absent.
        if 'priority' in record:
            number = record['priority']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                text = number.strip()
                if _PRIORITY_PATTERN.fullmatch(text) is None:
                    continue
                number = int(text, 10)
            else:
                continue
        else:
            number = 1

        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [{'category': name, 'maximum': groups[name]}
            for name in sorted(groups)]
