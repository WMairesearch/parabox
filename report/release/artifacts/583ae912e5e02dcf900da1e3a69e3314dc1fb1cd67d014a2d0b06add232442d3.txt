import re

_PRIORITY_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1 (before all others): retain only the last occurrence of each
    # nonempty string 'id'; records without such an id remain independent.
    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id:
                last_index[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Apply deduplication: drop superseded occurrences.
        record_id = record.get('id')
        if isinstance(record_id, str) and record_id:
            if last_index[record_id] != position:
                continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'category', falling back to 'kind' only when 'category' is absent.
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')

        # Skip non-string or empty-after-stripping keys; strip whitespace.
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # Read 'priority', defaulting to 1 only when absent.
        if 'priority' in record:
            number = record['priority']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                stripped = number.strip()
                if not _PRIORITY_PATTERN.fullmatch(stripped):
                    continue
                number = int(stripped, 10)
            else:
                # Skip all other values, including null.
                continue
        else:
            number = 1

        # Group by normalized category, keeping the maximum priority.
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    # Sort results lexicographically by 'category'.
    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
