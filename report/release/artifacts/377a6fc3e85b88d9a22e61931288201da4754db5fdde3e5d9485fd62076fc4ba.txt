import re

_INTEGER = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if isinstance(value, (str, bytes, bytearray)) or not isinstance(value, (list, tuple)):
        return []

    records = list(value)

    # Rule 1: deduplicate FIRST -- retain only the last occurrence of each
    # record carrying a nonempty string 'id'.  Other records stay independent.
    last_position = {}
    for position, record in enumerate(records):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                last_position[identifier] = position

    groups = {}
    for position, record in enumerate(records):
        # Rule 2: skip non-objects and inactive records.
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier != '':
            if last_position.get(identifier) != position:
                continue
        if record.get('active') is False:
            continue

        # Rule 3: key from 'path', falling back to 'route' only when absent.
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # Rule 4: 'bytes' defaults to 1 only when absent.
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, str):
                text = number.strip()
                if _INTEGER.fullmatch(text) is None:
                    continue
                try:
                    number = int(text)
                except ValueError:
                    continue
            elif type(number) is not int:
                continue
        else:
            number = 1

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': groups[name][0], 'count': groups[name][1]}
        for name in sorted(groups)
    ]
