import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1: deduplicate by 'id' first (last occurrence wins).
    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            ident = record.get('id')
            if isinstance(ident, str) and ident != '':
                last_index[ident] = position

    totals = {}
    for position, record in enumerate(value):
        # Rule 2: skip non-objects.
        if not isinstance(record, dict):
            continue

        # Rule 1 continued: keep only the last occurrence of each id.
        ident = record.get('id')
        if isinstance(ident, str) and ident != '':
            if last_index.get(ident) != position:
                continue

        # Rule 2: skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Rule 3: read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            raw_key = record['sku']
        else:
            raw_key = record.get('item')
        if not isinstance(raw_key, str):
            continue
        key = raw_key.strip()
        if key == '':
            continue

        # Rule 4: read 'delta', defaulting to 1 only when absent.
        if 'delta' in record:
            raw_delta = record['delta']
            if isinstance(raw_delta, bool):
                continue
            if isinstance(raw_delta, int):
                delta = raw_delta
            elif isinstance(raw_delta, str):
                text = raw_delta.strip()
                if not _INT_RE.fullmatch(text):
                    continue
                delta = int(text, 10)
            else:
                continue
        else:
            delta = 1

        totals[key] = totals.get(key, 0) + delta

    return [{'sku': key, 'total': totals[key]} for key in sorted(totals)]
