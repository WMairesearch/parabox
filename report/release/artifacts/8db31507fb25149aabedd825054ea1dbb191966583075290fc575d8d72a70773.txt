import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def _last_id_positions(records):
    """Map each nonempty string id to the index of its LAST occurrence."""
    last = {}
    for index, record in enumerate(records):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last[rid] = index
    return last


def _group_key(record):
    """'path', falling back to 'route' only when 'path' is absent."""
    if 'path' in record:
        raw = record['path']
    elif 'route' in record:
        raw = record['route']
    else:
        return None
    if not isinstance(raw, str):
        return None
    text = raw.strip()
    if not text:
        return None
    return text


def _amount(record):
    """'bytes', defaulting to 1 only when absent; None means skip record."""
    if 'bytes' not in record:
        return 1
    raw = record['bytes']
    if type(raw) is int:
        return raw
    if isinstance(raw, str):
        text = raw.strip()
        if _INT_RE.fullmatch(text) is not None:
            return int(text, 10)
    return None


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []
    records = list(value)
    last = _last_id_positions(records)
    totals = {}
    for index, record in enumerate(records):
        if not isinstance(record, dict):
            continue
        rid = record.get('id')
        if isinstance(rid, str) and rid != '':
            if last.get(rid) != index:
                continue
        if record.get('active') is False:
            continue
        key = _group_key(record)
        if key is None:
            continue
        amount = _amount(record)
        if amount is None:
            continue
        entry = totals.get(key)
        if entry is None:
            totals[key] = [amount, 1]
        else:
            entry[0] += amount
            entry[1] += 1
    result = []
    for key in sorted(totals):
        total, count = totals[key]
        result.append({'path': key, 'total': total, 'count': count})
    return result
