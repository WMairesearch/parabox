import re

_BYTES_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Deduplicate first: retain only the last occurrence of each nonempty
    # string 'id'; records without such an id remain independent.
    last_index_for_id = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_index_for_id[record_id] = index

    groups = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if (isinstance(record_id, str) and record_id != ''
                    and last_index_for_id[record_id] != index):
                continue
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, str):
                digits = number.strip()
                if not _BYTES_PATTERN.fullmatch(digits):
                    continue
                number = int(digits, 10)
            elif type(number) is not int:
                continue
        else:
            number = 1
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': total, 'count': count}
        for name, (total, count) in sorted(groups.items())
    ]
