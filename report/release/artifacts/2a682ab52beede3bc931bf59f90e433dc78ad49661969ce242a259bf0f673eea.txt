import re


def solve(value):
    if not isinstance(value, list):
        return []

    # First pass: deduplicate by nonempty string id, keeping only the last
    # occurrence of each such id. Records without such an id are independent.
    last_index = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and len(record_id) > 0:
                last_index[record_id] = index

    groups = {}
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        record_id = record.get('id')
        if isinstance(record_id, str) and len(record_id) > 0:
            if last_index.get(record_id) != index:
                continue

        if record.get('active') is False:
            continue

        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')

        if not isinstance(name, str) or not name.strip():
            continue
        name = name.strip()

        if 'priority' in record:
            number = record['priority']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                number = number
            elif isinstance(number, str):
                stripped = number.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    number = int(stripped)
                else:
                    continue
            else:
                continue
        else:
            number = 1

        current = groups.get(name)
        if current is None or number > current:
            groups[name] = number

    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
