import re

def solve(value):
    last_occurrence = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val:
                last_occurrence[id_val] = i

    groups = {}
    
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val:
                if last_occurrence[id_val] != i:
                    continue
        else:
            # wait, if not dict, we process it and then skip it later?
            pass
            
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        if 'path' in record:
            path_val = record['path']
        elif 'route' in record:
            path_val = record['route']
        else:
            continue
            
        if not isinstance(path_val, str):
            continue
            
        path_val = path_val.strip()
        if not path_val:
            continue
            
        if 'bytes' not in record:
            bytes_val = 1
        else:
            bytes_val = record['bytes']
            
        if type(bytes_val) is int:
            b = bytes_val
        elif isinstance(bytes_val, str):
            s = bytes_val.strip()
            if re.fullmatch(r'[+-]?[0-9]+', s):
                b = int(s)
            else:
                continue
        else:
            continue
            
        total, count = groups.get(path_val, (0, 0))
        groups[path_val] = (total + b, count + 1)
        
    result = []
    for path_val in sorted(groups.keys()):
        total, count = groups[path_val]
        result.append({
            'path': path_val,
            'total': total,
            'count': count
        })
        
    return result
