import re


def solve(value):
    # First pass: find the last position of each nonempty string id.
    # Records with no such id (missing, non-string, or empty string) are
    # independent and are never deduplicated.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                latest[identifier] = position

    groups = {}

    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        # Deduplicate before all other rules.
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier != '':
            if latest[identifier] != position:
                continue

        # Skip inactive records (JSON false, exactly).
        if record.get('active') is False:
            continue

        # Resolve path/route.  Fall back to route only when path is absent.
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')

        if not isinstance(name, str) or not name.strip():
            continue
        name = name.strip()

        # Resolve bytes, defaulting to 1 only when the key is absent.
        number = record.get('bytes', 1)
        if isinstance(number, str):
            stripped = number.strip()
            if re.fullmatch(r'[+-]?[0-9]+', stripped):
                number = int(stripped)
        if type(number) is not int:
            continue

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    result = [
        {'path': name, 'total': total, 'count': count}
        for name, (total, count) in groups.items()
    ]
    result.sort(key=lambda item: item['path'])
    return result
