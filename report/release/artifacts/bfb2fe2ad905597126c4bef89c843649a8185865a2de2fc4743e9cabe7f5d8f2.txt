import re


def solve(value):
    if not isinstance(value, list):
        return []

    # First retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Deduplicate before all other rules.
    last_indices = {}
    for i, item in enumerate(value):
        if isinstance(item, dict):
            item_id = item.get('id')
            if isinstance(item_id, str) and item_id != '':
                last_indices[item_id] = i

    retained = []
    for i, item in enumerate(value):
        if isinstance(item, dict):
            item_id = item.get('id')
            if isinstance(item_id, str) and item_id != '':
                if last_indices[item_id] != i:
                    continue
        retained.append(item)

    groups = {}
    for item in retained:
        # Skip non-object records and records whose 'active' is false.
        if not isinstance(item, dict):
            continue
        if item.get('active') is False:
            continue

        # Read 'path', falling back to 'route' only when 'path' is absent.
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if 'path' in item:
            key = item['path']
        elif 'route' in item:
            key = item['route']
        else:
            continue

        if not isinstance(key, str):
            continue
        normalized_path = key.strip()
        if not normalized_path:
            continue

        # Read 'bytes', defaulting to 1 only when absent.
        # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace;
        # convert strings to base-10 integers. Skip all other values, including null.
        if 'bytes' not in item:
            val_bytes = 1
        else:
            raw_bytes = item['bytes']
            if type(raw_bytes) is int:
                val_bytes = raw_bytes
            elif isinstance(raw_bytes, str):
                stripped_bytes = raw_bytes.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped_bytes):
                    val_bytes = int(stripped_bytes, 10)
                else:
                    continue
            else:
                continue

        # Group retained records by normalized 'path'
        total, count = groups.get(normalized_path, (0, 0))
        groups[normalized_path] = (total + val_bytes, count + 1)

    # Sort results lexicographically by 'path'. Return [] when no records remain.
    result = []
    for path in sorted(groups.keys()):
        result.append({
            'path': path,
            'total': groups[path][0],
            'count': groups[path][1],
        })

    return result
