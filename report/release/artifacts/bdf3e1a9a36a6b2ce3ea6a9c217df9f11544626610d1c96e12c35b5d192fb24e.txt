import re

PRIORITY_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Step 1: Deduplicate by nonempty string 'id'
    # Retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Deduplicate before all other rules.
    seen_ids = set()
    deduped_rev = []
    for record in reversed(value):
        if isinstance(record, dict) and isinstance(record.get('id'), str) and record['id'] != '':
            rec_id = record['id']
            if rec_id in seen_ids:
                continue
            seen_ids.add(rec_id)
            deduped_rev.append(record)
        else:
            deduped_rev.append(record)
    deduped = list(reversed(deduped_rev))

    # Step 2: Process records
    groups = {}
    for record in deduped:
        # Skip non-object records and records whose 'active' is false.
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Read 'category', falling back to 'kind' only when 'category' is absent.
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if 'category' in record:
            raw_cat = record['category']
        elif 'kind' in record:
            raw_cat = record['kind']
        else:
            continue

        if not isinstance(raw_cat, str):
            continue
        cat = raw_cat.strip()
        if not cat:
            continue

        # Read 'priority', defaulting to 1 only when absent.
        # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace;
        # convert strings to base-10 integers. Skip all other values, including null.
        if 'priority' not in record:
            priority = 1
        else:
            raw_priority = record['priority']
            if isinstance(raw_priority, bool):
                continue
            elif isinstance(raw_priority, int):
                priority = raw_priority
            elif isinstance(raw_priority, str):
                p_str = raw_priority.strip()
                if PRIORITY_PATTERN.fullmatch(p_str):
                    priority = int(p_str, 10)
                else:
                    continue
            else:
                continue

        # Group retained records by normalized 'category'.
        # Each output object contains 'category' and the maximum priority as 'maximum'.
        if cat in groups:
            if priority > groups[cat]:
                groups[cat] = priority
        else:
            groups[cat] = priority

    # Sort results lexicographically by 'category'. Return [] when no records remain.
    if not groups:
        return []

    result = []
    for cat in sorted(groups.keys()):
        result.append({
            'category': cat,
            'maximum': groups[cat]
        })
    return result
