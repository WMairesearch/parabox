import re

_DELTA_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    totals = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier:
            if latest[identifier] != position:
                continue
        if record.get('active') is False:
            continue
        if 'sku' in record:
            name = record['sku']
        elif 'item' in record:
            name = record['item']
        else:
            name = None
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'delta' in record:
            number = record['delta']
        else:
            number = 1
        if isinstance(number, str):
            text = number.strip()
            if _DELTA_RE.fullmatch(text):
                number = int(text, 10)
            else:
                continue
        elif type(number) is not int:
            continue
        totals[name] = totals.get(name, 0) + number

    return [{'sku': name, 'total': totals[name]} for name in sorted(totals)]
