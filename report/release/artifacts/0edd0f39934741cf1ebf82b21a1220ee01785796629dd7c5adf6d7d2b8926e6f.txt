import re

_INT_RE = re.compile(r'[+-]?[0-9]+$')


def solve(value):
    # Rule 1 (dedup first): keep only the last occurrence of each
    # nonempty string 'id'; records without such an id stay independent.
    last_by_id = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_by_id[rid] = position

    groups = {}
    for position, record in enumerate(value):
        rid = record.get('id') if isinstance(record, dict) else None
        if isinstance(rid, str) and rid != '' and last_by_id[rid] != position:
            continue  # superseded by a later record with the same id

        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                stripped = number.strip()
                if not _INT_RE.match(stripped):
                    continue
                number = int(stripped, 10)
            else:
                continue
        else:
            number = 1

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    result = []
    for name in sorted(groups):
        total, count = groups[name]
        result.append({'path': name, 'total': total, 'count': count})
    return result
