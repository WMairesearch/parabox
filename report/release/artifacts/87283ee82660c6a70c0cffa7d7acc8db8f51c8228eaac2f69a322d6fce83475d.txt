import re

_BYTES_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate before all other rules: keep only the last occurrence of
    # each nonempty string 'id'. Records without such an id remain independent.
    last_id_index = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_id_index[record_id] = index

    selected = []
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if last_id_index[record_id] == index:
                    selected.append(index)
                continue
        selected.append(index)

    groups = {}
    for index in selected:
        record = value[index]
        if not isinstance(record, dict):
            continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            raw_path = record['path']
        else:
            raw_path = record.get('route')

        if not isinstance(raw_path, str):
            continue
        path = raw_path.strip()
        if not path:
            continue

        # Read 'bytes', defaulting to 1 only when absent.
        if 'bytes' in record:
            raw_bytes = record['bytes']
            if type(raw_bytes) is int:
                number = raw_bytes
            elif isinstance(raw_bytes, str):
                stripped = raw_bytes.strip()
                if _BYTES_RE.fullmatch(stripped):
                    number = int(stripped, 10)
                else:
                    continue
            else:
                continue
        else:
            number = 1

        total, count = groups.get(path, (0, 0))
        groups[path] = (total + number, count + 1)

    return [
        {'path': path, 'total': total, 'count': count}
        for path, (total, count) in sorted(groups.items())
    ]
