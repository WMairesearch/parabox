import re

_INT = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []

    # Rule 1 (applies before every other rule): keep only the LAST occurrence of
    # each nonempty string 'id'.  Records without such an id stay independent.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        # Rule 2: skip non-object records.
        if not isinstance(record, dict):
            continue
        # Rule 1 continued: drop superseded duplicates.
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier != '':
            if latest[identifier] != position:
                continue
        # Rule 3: skip records whose 'active' is JSON false.
        if record.get('active') is False:
            continue
        # Rule 4: 'sku', falling back to 'item' only when 'sku' is absent.
        key = record['sku'] if 'sku' in record else record.get('item')
        if not isinstance(key, str):
            continue
        key = key.strip()
        if key == '':
            continue
        # Rule 5: 'delta', defaulting to 1 only when absent.
        number = record['delta'] if 'delta' in record else 1
        if isinstance(number, bool):
            continue
        if isinstance(number, str):
            text = number.strip()
            if _INT.fullmatch(text) is None:
                continue
            number = int(text)
        elif not isinstance(number, int):
            continue
        groups[key] = groups.get(key, 0) + number

    # Rule 6: lexicographic sort by 'sku'; [] when nothing remains.
    return [{'sku': sku, 'total': groups[sku]} for sku in sorted(groups)]
