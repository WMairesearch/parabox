import re

_PRIORITY_RE = re.compile(r'[+-]?[0-9]+')


def _has_id(record):
    return isinstance(record, dict) and isinstance(record.get('id'), str) and record.get('id') != ''


def solve(value):
    # Deduplicate first: keep only the last occurrence of each nonempty string id.
    last = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            ident = record.get('id')
            if isinstance(ident, str) and ident != '':
                last[ident] = position

    groups = {}
    kept = []
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        ident = record.get('id')
        if isinstance(ident, str) and ident != '' and last[ident] != position:
            continue
        if 'active' in record and record['active'] is False:
            continue
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'priority' in record:
            number = record['priority']
            if isinstance(number, str):
                text = number.strip()
                if not _PRIORITY_RE.fullmatch(text):
                    continue
                number = int(text, 10)
            elif type(number) is not int:
                continue
        else:
            number = 1
        kept.append((name, number))

    for name, number in kept:
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
