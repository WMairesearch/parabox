import re

_DELTA_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1 (before all others): retain only the last occurrence of each
    # nonempty string 'id'; records without such an id remain independent.
    last_index = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_index[rid] = i

    groups = {}
    for i, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        rid = record.get('id')
        if isinstance(rid, str) and rid != '' and last_index.get(rid) != i:
            continue  # superseded by a later record with the same id
        if record.get('active') is False:
            continue
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str) or not name.strip():
            continue
        name = name.strip()
        if 'delta' in record:
            number = record['delta']
            if isinstance(number, str):
                text = number.strip()
                if _DELTA_RE.fullmatch(text):
                    number = int(text)
                else:
                    continue
            elif type(number) is int:
                pass
            else:
                continue
        else:
            number = 1
        groups[name] = groups.get(name, 0) + number

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
