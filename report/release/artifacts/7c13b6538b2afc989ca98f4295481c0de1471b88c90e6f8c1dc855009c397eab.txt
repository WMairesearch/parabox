import re


def solve(value):
    # Deduplicate before all other rules: keep only the last occurrence of
    # each nonempty string "id".  Records without such an id stay independent.
    last_positions = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_positions[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        record_id = record.get('id')
        if isinstance(record_id, str) and record_id != '':
            if last_positions[record_id] != position:
                continue  # an earlier occurrence of the same nonempty id

        # Skip records whose 'active' is false (JSON false).
        if record.get('active') is False:
            continue

        # Read 'category', falling back to 'kind' only when 'category' is absent.
        if 'category' in record:
            key = record['category']
        else:
            key = record.get('kind')

        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue

        # Read 'priority', defaulting to 1 only when absent.
        if 'priority' in record:
            number = record['priority']
            if isinstance(number, str):
                number = number.strip()
                if re.fullmatch(r'[+-]?[0-9]+', number):
                    number = int(number)
                else:
                    continue
            if type(number) is not int:
                continue
        else:
            number = 1

        if key in groups:
            if number > groups[key]:
                groups[key] = number
        else:
            groups[key] = number

    return [
        {'category': category, 'maximum': groups[category]}
        for category in sorted(groups)
    ]
