import re

_INTEGER = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if isinstance(value, (str, bytes)) or not isinstance(value, (list, tuple)):
        return []

    records = list(value)

    last_position = {}
    for position, record in enumerate(records):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                last_position[identifier] = position

    groups = {}
    for position, record in enumerate(records):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier != '':
            if last_position.get(identifier) != position:
                continue
        if record.get('active') is False:
            continue

        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if 'delta' in record:
            number = record['delta']
            if isinstance(number, str):
                text = number.strip()
                if _INTEGER.fullmatch(text) is None:
                    continue
                number = int(text)
            elif type(number) is not int:
                continue
        else:
            number = 1

        groups[name] = groups.get(name, 0) + number

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
