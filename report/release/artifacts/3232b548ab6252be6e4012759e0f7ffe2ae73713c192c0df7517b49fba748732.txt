import re

_DELTA_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []
    # Deduplicate by id first: keep only the last occurrence of each
    # nonempty string id; records without such an id stay independent.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position
    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != position:
            continue
        if record.get('active') is False:
            continue
        name = record['sku'] if 'sku' in record else record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'delta' in record:
            number = record['delta']
            if isinstance(number, str):
                stripped = number.strip()
                if not _DELTA_PATTERN.fullmatch(stripped):
                    continue
                number = int(stripped)
            elif type(number) is not int:
                continue
        else:
            number = 1
        groups[name] = groups.get(name, 0) + number
    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
