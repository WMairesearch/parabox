import re

def solve(value):
    last_occurrence = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                last_occurrence[rec_id] = i
                
    groups = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                if last_occurrence[rec_id] != i:
                    continue
        
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
            
        if 'category' in record:
            key = record['category']
        elif 'kind' in record:
            key = record['kind']
        else:
            continue
            
        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue
            
        if 'priority' in record:
            p_val = record['priority']
        else:
            p_val = 1
            
        priority = None
        if isinstance(p_val, int) and not isinstance(p_val, bool):
            priority = p_val
        elif isinstance(p_val, str):
            p_str = p_val.strip()
            if re.match(r'^[+-]?[0-9]+$', p_str):
                priority = int(p_str)
                
        if priority is None:
            continue
            
        if key not in groups:
            groups[key] = priority
        else:
            groups[key] = max(groups[key], priority)
            
    result = []
    for key in sorted(groups.keys()):
        result.append({'category': key, 'maximum': groups[key]})
        
    return result
