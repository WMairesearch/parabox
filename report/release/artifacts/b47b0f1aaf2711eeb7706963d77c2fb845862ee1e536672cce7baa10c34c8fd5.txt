import re


_BYTES_RE = re.compile(r'[+-]?[0-9]+')


def _normalized_path(record):
    """Return the normalized path string, or None if it should be skipped."""
    if 'path' in record:
        name = record.get('path')
    else:
        name = record.get('route')
    if not isinstance(name, str):
        return None
    name = name.strip()
    if not name:
        return None
    return name


def _bytes_value(record):
    """Return the integer bytes value, or None if it should be skipped."""
    if 'bytes' in record:
        number = record.get('bytes')
    else:
        return 1
    if isinstance(number, str):
        number = number.strip()
        if _BYTES_RE.fullmatch(number):
            return int(number)
        return None
    if type(number) is int:
        return number
    return None


def solve(value):
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        identifier = record.get('id')
        if isinstance(identifier, str) and identifier != '':
            if latest[identifier] != position:
                continue

        if record.get('active') is False:
            continue

        name = _normalized_path(record)
        if name is None:
            continue

        number = _bytes_value(record)
        if number is None:
            continue

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': total, 'count': count}
        for name, (total, count) in sorted(groups.items())
    ]
