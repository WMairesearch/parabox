import re

PRIORITY_RE = re.compile(r'^[+-]?[0-9]+$')


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []

    # First retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Deduplicate before all other rules.
    last_indices = {}
    for idx, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                last_indices[rec_id] = idx

    groups = {}
    for idx, record in enumerate(value):
        # Skip non-object records
        if not isinstance(record, dict):
            continue

        # Deduplicate before all other rules.
        rec_id = record.get('id')
        if isinstance(rec_id, str) and rec_id != '':
            if last_indices[rec_id] != idx:
                continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'category', falling back to 'kind' only when 'category' is absent.
        if 'category' in record:
            raw_category = record['category']
        elif 'kind' in record:
            raw_category = record['kind']
        else:
            continue

        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if not isinstance(raw_category, str):
            continue
        category = raw_category.strip()
        if not category:
            continue

        # Read 'priority', defaulting to 1 only when absent.
        if 'priority' not in record:
            priority = 1
        else:
            raw_priority = record['priority']
            if isinstance(raw_priority, bool):
                continue
            elif isinstance(raw_priority, int):
                priority = raw_priority
            elif isinstance(raw_priority, str):
                s = raw_priority.strip()
                if not PRIORITY_RE.match(s):
                    continue
                priority = int(s, 10)
            else:
                continue

        if category not in groups:
            groups[category] = priority
        else:
            if priority > groups[category]:
                groups[category] = priority

    result = []
    for category in sorted(groups.keys()):
        result.append({'category': category, 'maximum': groups[category]})

    return result
