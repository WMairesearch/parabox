import re

_DELTA_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1 (first): retain only the last occurrence of each nonempty
    # string 'id'; records without such an id remain independent.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        # Skip non-object records.
        if not isinstance(record, dict):
            continue
        # Deduplicate (applies before all other rules).
        identifier = record.get('id')
        if (isinstance(identifier, str) and identifier != ''
                and latest[identifier] != position):
            continue
        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue
        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            name = record['sku']
        elif 'item' in record:
            name = record['item']
        else:
            continue
        # Skip non-string or empty-after-stripping keys.
        if not isinstance(name, str):
            continue
        name = name.strip()
        if name == '':
            continue
        # Read 'delta', defaulting to 1 only when absent.
        number = record['delta'] if 'delta' in record else 1
        if isinstance(number, bool):
            continue
        if isinstance(number, int):
            pass
        elif isinstance(number, str):
            text = number.strip()
            if _DELTA_PATTERN.fullmatch(text) is None:
                continue
            number = int(text, 10)
        else:
            # Skip all other values, including null.
            continue
        groups[name] = groups.get(name, 0) + number

    # Sort results lexicographically by 'sku'.
    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
