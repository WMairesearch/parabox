import re

_INTEGER = re.compile(r'[+-]?[0-9]+')


def _retained_records(value):
    """Apply the last-occurrence id deduplication, keeping input order."""
    if not isinstance(value, list):
        return []
    records = list(value)
    last = {}
    for position, record in enumerate(records):
        if isinstance(record, dict):
            key = record.get('id')
            if isinstance(key, str) and key != '':
                last[key] = position
    kept = []
    for position, record in enumerate(records):
        if isinstance(record, dict):
            key = record.get('id')
            if isinstance(key, str) and key != '':
                if last[key] != position:
                    continue
        kept.append(record)
    return kept


def solve(value):
    totals = {}
    counts = {}
    for record in _retained_records(value):
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        if 'path' in record:
            name = record.get('path')
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'bytes' in record:
            number = record.get('bytes')
            if isinstance(number, str):
                text = number.strip()
                if not _INTEGER.fullmatch(text):
                    continue
                number = int(text, 10)
            elif type(number) is not int:
                continue
        else:
            number = 1
        totals[name] = totals.get(name, 0) + number
        counts[name] = counts.get(name, 0) + 1
    return [
        {'path': name, 'total': totals[name], 'count': counts[name]}
        for name in sorted(totals)
    ]
