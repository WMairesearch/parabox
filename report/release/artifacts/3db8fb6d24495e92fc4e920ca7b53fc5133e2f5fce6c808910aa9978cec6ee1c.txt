import re


_INT_RE = re.compile(r'[+-]?[0-9]+')


def _dedup(value):
    """Keep only the last occurrence of each nonempty string id."""
    last = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            ident = record.get('id')
            if isinstance(ident, str) and ident:
                last[ident] = position
    return last


def solve(value):
    if not isinstance(value, list):
        return []
    last = _dedup(value)
    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        ident = record.get('id')
        if isinstance(ident, str) and ident and last.get(ident) != position:
            continue
        if record.get('active') is False:
            continue
        name = record['path'] if 'path' in record else record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, bool):
                continue
            if isinstance(number, str):
                text = number.strip()
                if _INT_RE.fullmatch(text) is None:
                    continue
                number = int(text, 10)
            elif not isinstance(number, int):
                continue
        else:
            number = 1
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)
    return [
        {'path': name, 'total': groups[name][0], 'count': groups[name][1]}
        for name in sorted(groups)
    ]
