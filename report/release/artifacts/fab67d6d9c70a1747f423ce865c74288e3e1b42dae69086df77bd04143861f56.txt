import re

PRIORITY_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # 1. Deduplicate: retain only the last occurrence of each nonempty
    #    string 'id'. Records without such an id remain independent.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        # 2. Skip non-object records.
        if not isinstance(record, dict):
            continue
        # Drop earlier occurrences of a duplicated id.
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != position:
            continue
        # 3. Skip records whose 'active' is false (exactly false).
        if record.get('active') is False:
            continue
        # 4. 'category', falling back to 'kind' only when 'category' is absent.
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # 5. 'priority', defaulting to 1 only when absent.
        if 'priority' in record:
            number = record['priority']
        else:
            number = 1
        if isinstance(number, bool):
            continue
        if isinstance(number, int):
            pass
        elif isinstance(number, str):
            text = number.strip()
            if not PRIORITY_PATTERN.fullmatch(text):
                continue
            number = int(text)
        else:
            continue
        # 6. Group by normalized category, keeping the maximum priority.
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    # 7. Sort results lexicographically by 'category'.
    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
