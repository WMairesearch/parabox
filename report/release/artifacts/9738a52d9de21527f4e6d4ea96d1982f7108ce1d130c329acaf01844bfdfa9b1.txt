import re

_PRIORITY_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1: keep only the last occurrence of each nonempty string 'id'.
    # Records without such an id remain independent (all kept).
    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                last_index[identifier] = position

    retained = []
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                if last_index.get(identifier) != position:
                    continue
        retained.append(record)

    groups = {}
    for record in retained:
        # Skip non-object records.
        if not isinstance(record, dict):
            continue
        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue
        # Category, falling back to 'kind' only when 'category' is absent.
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Priority, defaulting to 1 only when absent.
        if 'priority' in record:
            raw = record['priority']
            if isinstance(raw, str):
                candidate = raw.strip()
                if not _PRIORITY_RE.fullmatch(candidate):
                    continue
                number = int(candidate, 10)
            elif type(raw) is int:
                number = raw
            else:
                continue
        else:
            number = 1
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [
        {'category': name, 'maximum': groups[name]}
        for name in sorted(groups)
    ]
