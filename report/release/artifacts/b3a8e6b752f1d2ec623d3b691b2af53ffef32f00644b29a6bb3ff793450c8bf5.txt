import re


_DELTA_RE = re.compile(r'[+-]?[0-9]+\Z')


def solve(value):
    # Deduplicate before all other rules: keep only the last occurrence of each
    # record that has a nonempty string 'id'.  Records without such an id are
    # independent and are always kept at this stage.
    last_id_index = {}
    dedupe_keep = []
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_id_index[record_id] = position

    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if last_id_index.get(record_id) != position:
                    continue
        dedupe_keep.append(record)

    groups = {}
    for record in dedupe_keep:
        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records whose 'active' is JSON false.
        if record.get('active') is False:
            continue

        # Read the key, falling back from 'sku' to 'item' only when 'sku'
        # is absent.  Keys must be strings that are nonempty after stripping.
        if 'sku' in record:
            key = record.get('sku')
        else:
            key = record.get('item')
        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue

        # Read delta, defaulting to 1 only when the key is absent.
        if 'delta' in record:
            number = record.get('delta')
            if type(number) is int:
                delta = number
            elif isinstance(number, str):
                number = number.strip()
                if _DELTA_RE.match(number) is None:
                    continue
                delta = int(number, 10)
            else:
                continue
        else:
            delta = 1

        groups[key] = groups.get(key, 0) + delta

    return [
        {'sku': key, 'total': groups[key]}
        for key in sorted(groups)
    ]
