import re

_NUMBER = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []
    # Deduplicate first: keep only the last occurrence of each nonempty
    # string id; records without such an id remain independent.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position
    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != position:
            continue
        if record.get('active') is False:
            continue
        name = record.get('path', record.get('route'))
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        number = record.get('bytes', 1)
        if isinstance(number, str):
            stripped = number.strip()
            if not _NUMBER.fullmatch(stripped):
                continue
            number = int(stripped, 10)
        elif type(number) is not int:
            continue
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)
    return [
        {'path': name, 'total': total, 'count': count}
        for name, (total, count) in sorted(groups.items())
    ]
