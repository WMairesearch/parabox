import re

def solve(value):
    last_occurrence = {}
    
    # Do not modify the input, just iterate.
    # Find last occurrences of nonempty string 'id'
    for idx, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val != "":
                last_occurrence[id_val] = idx
                
    groups = {}
    
    for idx, record in enumerate(value):
        # Skip non-object records
        if not isinstance(record, dict):
            continue
            
        # Retain only the last occurrence of each nonempty string 'id'
        id_val = record.get('id')
        if isinstance(id_val, str) and id_val != "":
            if last_occurrence[id_val] != idx:
                continue
                
        # Skip records whose 'active' is false
        if record.get('active') is False:
            continue
            
        # Read 'path', falling back to 'route' only when 'path' is absent
        if 'path' in record:
            path_val = record['path']
        elif 'route' in record:
            path_val = record['route']
        else:
            continue
            
        # Skip non-string or empty-after-stripping keys
        if not isinstance(path_val, str):
            continue
            
        path_norm = path_val.strip()
        if not path_norm:
            continue
            
        # Read 'bytes', defaulting to 1 only when absent
        if 'bytes' not in record:
            bytes_val = 1
        else:
            bytes_raw = record['bytes']
            if type(bytes_raw) is int:
                bytes_val = bytes_raw
            elif isinstance(bytes_raw, str):
                bytes_str = bytes_raw.strip()
                if re.match(r'^[+-]?[0-9]+$', bytes_str):
                    bytes_val = int(bytes_str)
                else:
                    continue
            else:
                continue
                
        # Group retained records by normalized 'path'
        if path_norm not in groups:
            groups[path_norm] = {'total': 0, 'count': 0}
            
        groups[path_norm]['total'] += bytes_val
        groups[path_norm]['count'] += 1
        
    result = []
    # Sort results lexicographically by 'path'
    for path_norm in sorted(groups.keys()):
        result.append({
            'path': path_norm,
            'total': groups[path_norm]['total'],
            'count': groups[path_norm]['count']
        })
        
    return result
