import re

def solve(value):
    seen_ids = set()
    retained = []
    
    # Deduplicate before all other rules
    for record in reversed(value):
        if isinstance(record, dict) and 'id' in record:
            id_val = record['id']
            if isinstance(id_val, str) and id_val != '':
                if id_val in seen_ids:
                    continue
                seen_ids.add(id_val)
        retained.append(record)
    
    retained.reverse()
    
    groups = {}
    
    for record in retained:
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        if 'category' in record:
            cat_val = record['category']
        elif 'kind' in record:
            cat_val = record['kind']
        else:
            continue
            
        if not isinstance(cat_val, str):
            continue
            
        cat_val = cat_val.strip()
        if not cat_val:
            continue
            
        if 'priority' not in record:
            pri_val = 1
        else:
            pri_val = record['priority']
            
        if type(pri_val) is int:
            priority = pri_val
        elif isinstance(pri_val, str):
            s = pri_val.strip()
            if re.fullmatch(r'[+-]?[0-9]+', s):
                priority = int(s)
            else:
                continue
        else:
            continue
            
        if cat_val not in groups or priority > groups[cat_val]:
            groups[cat_val] = priority
            
    result = []
    for cat in sorted(groups.keys()):
        result.append({'category': cat, 'maximum': groups[cat]})
        
    return result
