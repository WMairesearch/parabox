import re

_INTEGER = re.compile(r'[+-]?[0-9]+')


def solve(value):
    """Group retained event records by normalized category.

    Returns a list of {"category": <str>, "maximum": <int>} sorted
    lexicographically by category.  The input is never modified.
    """
    # --- deduplicate: keep only the last occurrence of each nonempty string id
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier:
            if latest[identifier] != position:
                continue
        if record.get('active') is False:
            continue
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'priority' in record:
            number = record['priority']
        else:
            number = 1
        if isinstance(number, str):
            candidate = number.strip()
            if _INTEGER.fullmatch(candidate):
                number = int(candidate, 10)
            else:
                continue
        if type(number) is not int:
            continue
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [{'category': name, 'maximum': groups[name]}
            for name in sorted(groups)]
