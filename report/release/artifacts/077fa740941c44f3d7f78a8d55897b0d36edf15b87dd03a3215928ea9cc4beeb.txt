import re


def solve(value):
    # Deduplicate before all other rules: keep only the last occurrence of
    # each nonempty string 'id'; every record without such an id is kept.
    seen_ids = set()
    deduped = []
    for record in reversed(value):
        if isinstance(record, dict):
            id_value = record.get('id')
            if isinstance(id_value, str) and id_value != '':
                if id_value in seen_ids:
                    continue
                seen_ids.add(id_value)
        deduped.append(record)
    deduped.reverse()

    groups = {}

    for record in deduped:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        if 'path' in record:
            key = record['path']
        else:
            key = record.get('route')

        if not isinstance(key, str):
            continue
        key = key.strip()
        if key == '':
            continue

        if 'bytes' not in record:
            number = 1
        else:
            raw = record['bytes']
            if type(raw) is int:
                number = raw
            elif isinstance(raw, str):
                raw = raw.strip()
                if re.fullmatch(r'[+-]?[0-9]+', raw):
                    number = int(raw)
                else:
                    continue
            else:
                continue

        total, count = groups.get(key, (0, 0))
        groups[key] = (total + number, count + 1)

    result = [
        {'path': key, 'total': groups[key][0], 'count': groups[key][1]}
        for key in sorted(groups)
    ]
    return result
