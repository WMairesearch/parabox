import re

def solve(value):
    last_occurrence = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                last_occurrence[rec_id] = i

    groups = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                if last_occurrence[rec_id] != i:
                    continue

        if not isinstance(record, dict):
            continue

        if record.get('active') is False:
            continue

        if 'category' in record:
            cat = record['category']
        elif 'kind' in record:
            cat = record['kind']
        else:
            continue

        if not isinstance(cat, str):
            continue

        cat = cat.strip()
        if not cat:
            continue

        if 'priority' not in record:
            priority = 1
        else:
            p_raw = record['priority']
            if type(p_raw) is bool:
                continue
            elif type(p_raw) is int:
                priority = p_raw
            elif isinstance(p_raw, str):
                p_str = p_raw.strip()
                if re.fullmatch(r'[+-]?[0-9]+', p_str):
                    priority = int(p_str)
                else:
                    continue
            else:
                continue

        if cat not in groups or priority > groups[cat]:
            groups[cat] = priority

    result = [{'category': k, 'maximum': v} for k, v in sorted(groups.items())]
    return result
