import re

def solve(value):
    # Step 1: Deduplication
    last_indices = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val != '':
                last_indices[id_val] = i
                
    groups = {}
    
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val != '':
                if last_indices[id_val] != i:
                    continue
        
        # Step 2: Skip non-object records and 'active' == false
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        # Step 3: 'path' / 'route'
        if 'path' in record:
            p = record['path']
        elif 'route' in record:
            p = record['route']
        else:
            continue
            
        if not isinstance(p, str):
            continue
            
        p = p.strip()
        if not p:
            continue
            
        # Step 4: 'bytes'
        if 'bytes' not in record:
            b_val = 1
        else:
            b_val = record['bytes']
            
        if type(b_val) is int:
            b = b_val
        elif isinstance(b_val, str):
            s = b_val.strip()
            if re.fullmatch(r'[+-]?[0-9]+', s):
                b = int(s)
            else:
                continue
        else:
            continue
            
        # Step 5: Group
        if p not in groups:
            groups[p] = [0, 0]
        groups[p][0] += b
        groups[p][1] += 1
        
    # Step 6: Result
    result = []
    for p in sorted(groups.keys()):
        result.append({'path': p, 'total': groups[p][0], 'count': groups[p][1]})
        
    return result
