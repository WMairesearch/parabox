import re

DELTA_RE = re.compile(r'^[+-]?[0-9]+$')


def solve(value):
    if not isinstance(value, list):
        return []

    # First retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Deduplicate before all other rules.
    last_id_indices = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                last_id_indices[rec_id] = i

    retained = []
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                if last_id_indices[rec_id] != i:
                    continue
        retained.append(record)

    # Skip non-object records and records whose 'active' is false.
    # Read 'sku', falling back to 'item' only when 'sku' is absent.
    # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
    # Read 'delta', defaulting to 1 only when absent. Accept integers (not booleans)
    # and strings matching [+-]?[0-9]+ after stripping whitespace; convert strings
    # to base-10 integers. Skip all other values, including null.
    # Group retained records by normalized 'sku'. Each output object contains 'sku'
    # and the sum of deltas as 'total'. Sort results lexicographically by 'sku'.
    # Return [] when no records remain. Do not modify the input.

    groups = {}

    for record in retained:
        if not isinstance(record, dict):
            continue

        if record.get('active') is False:
            continue

        if 'sku' in record:
            raw_key = record['sku']
        elif 'item' in record:
            raw_key = record['item']
        else:
            continue

        if not isinstance(raw_key, str):
            continue
        sku = raw_key.strip()
        if not sku:
            continue

        if 'delta' not in record:
            delta = 1
        else:
            raw_delta = record['delta']
            if isinstance(raw_delta, int) and not isinstance(raw_delta, bool):
                delta = raw_delta
            elif isinstance(raw_delta, str):
                s_delta = raw_delta.strip()
                if DELTA_RE.match(s_delta):
                    delta = int(s_delta, 10)
                else:
                    continue
            else:
                continue

        groups[sku] = groups.get(sku, 0) + delta

    result = []
    for sku in sorted(groups.keys()):
        result.append({'sku': sku, 'total': groups[sku]})

    return result
