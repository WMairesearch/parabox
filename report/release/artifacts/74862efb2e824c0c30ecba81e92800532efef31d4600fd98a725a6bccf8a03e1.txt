import re
import sys

try:
    sys.set_int_max_str_digits(0)
except AttributeError:
    pass

_INT_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1 (before all others): retain only the last occurrence of each
    # nonempty string 'id'; records without such an id remain independent.
    last_index = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_index[rid] = index

    retained = []
    for index, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '' and last_index[rid] != index:
                continue
        retained.append(record)

    groups = {}
    for record in retained:
        # Skip non-object records.
        if not isinstance(record, dict):
            continue
        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue
        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        # Skip non-string or empty-after-stripping keys; strip whitespace.
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Read 'delta', defaulting to 1 only when absent.
        if 'delta' in record:
            number = record['delta']
        else:
            number = 1
        # Accept integers (not booleans) and [+-]?[0-9]+ strings (stripped).
        if isinstance(number, bool):
            continue
        if isinstance(number, int):
            delta = number
        elif isinstance(number, str):
            text = number.strip()
            if not _INT_PATTERN.fullmatch(text):
                continue
            delta = int(text, 10)
        else:
            # Skip all other values, including null.
            continue
        groups[name] = groups.get(name, 0) + delta

    # Sort results lexicographically by 'sku'; [] when nothing remains.
    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
