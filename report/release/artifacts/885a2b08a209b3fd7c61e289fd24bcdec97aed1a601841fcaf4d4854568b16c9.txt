import re


_DELTA_RE = re.compile(r'[+-]?[0-9]+')


def _deduplicate(records):
    """Keep only the last occurrence of each nonempty string id."""
    seen = set()
    deduped = []
    for record in reversed(records):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if record_id in seen:
                    continue
                seen.add(record_id)
        deduped.append(record)
    deduped.reverse()
    return deduped


def solve(value):
    records = _deduplicate(value)
    groups = {}

    for record in records:
        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records whose 'active' is JSON false.
        if record.get('active') is False:
            continue

        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            sku = record['sku']
        else:
            sku = record.get('item')

        # Skip non-string or empty-after-stripping keys; strip whitespace.
        if not isinstance(sku, str):
            continue
        sku = sku.strip()
        if not sku:
            continue

        # Read 'delta', defaulting to 1 only when absent.
        if 'delta' not in record:
            delta = 1
        else:
            delta = record['delta']
            if type(delta) is int:
                # Integers are accepted, but booleans are not integers here.
                pass
            elif isinstance(delta, str):
                delta = delta.strip()
                if _DELTA_RE.fullmatch(delta):
                    delta = int(delta, 10)
                else:
                    continue
            else:
                continue

        groups[sku] = groups.get(sku, 0) + delta

    return [{'sku': sku, 'total': total} for sku, total in sorted(groups.items())]
