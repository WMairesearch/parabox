import re
import sys

if hasattr(sys, 'set_int_max_str_digits'):
    try:
        sys.set_int_max_str_digits(0)
    except Exception:
        pass


def solve(value):
    if not isinstance(value, list):
        return []

    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != position:
            continue

        if record.get('active') is False:
            continue

        if 'sku' in record:
            name = record['sku']
        elif 'item' in record:
            name = record['item']
        else:
            continue

        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if 'delta' not in record:
            number = 1
        else:
            raw_delta = record['delta']
            if isinstance(raw_delta, bool):
                continue
            elif isinstance(raw_delta, int):
                number = raw_delta
            elif isinstance(raw_delta, str):
                s = raw_delta.strip()
                if re.fullmatch(r'[+-]?[0-9]+', s):
                    try:
                        number = int(s, 10)
                    except (ValueError, OverflowError):
                        continue
                else:
                    continue
            else:
                continue

        groups[name] = groups.get(name, 0) + number

    result = []
    for name in sorted(groups.keys()):
        result.append({'sku': name, 'total': groups[name]})
    return result
