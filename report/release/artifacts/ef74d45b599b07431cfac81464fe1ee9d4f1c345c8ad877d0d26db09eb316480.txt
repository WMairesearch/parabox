import re

# Integers optionally signed, ASCII digits only, whole-string match.
_INT_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    """Group inventory records by normalized sku and sum their deltas."""
    if not isinstance(value, list):
        return []

    # --- Deduplicate first: keep only the last record for each nonempty
    # --- string 'id'. Records without such an id stay independent.
    last_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                last_position[identifier] = position

    totals = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        identifier = record.get('id')
        if isinstance(identifier, str) and identifier:
            if last_position[identifier] != position:
                continue

        if record.get('active') is False:
            continue

        # 'sku' with fallback to 'item' only when 'sku' is absent.
        if 'sku' in record:
            key = record['sku']
        else:
            key = record.get('item')
        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue

        # 'delta' defaults to 1 only when the key is absent.
        if 'delta' in record:
            delta = record['delta']
        else:
            delta = 1
        if isinstance(delta, str):
            text = delta.strip()
            if not _INT_PATTERN.fullmatch(text):
                continue
            delta = int(text, 10)
        elif isinstance(delta, int) and not isinstance(delta, bool):
            pass
        else:
            continue

        totals[key] = totals.get(key, 0) + delta

    return [{'sku': name, 'total': totals[name]} for name in sorted(totals)]
