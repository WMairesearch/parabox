import re


def solve(value):
    # Deduplicate before all other rules: for each nonempty-string id,
    # keep only the last occurrence. Records without such an id are kept
    # independently.
    last_id_index = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            id_value = record.get('id')
            if isinstance(id_value, str) and id_value != '':
                last_id_index[id_value] = index

    deduplicated = []
    for index, record in enumerate(value):
        if isinstance(record, dict):
            id_value = record.get('id')
            if isinstance(id_value, str) and id_value != '':
                if last_id_index[id_value] != index:
                    continue
        deduplicated.append(record)

    groups = {}
    for record in deduplicated:
        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'path', falling back to 'route' only when 'path' is absent.
        key = record.get('path', record.get('route'))
        if not isinstance(key, str) or not key.strip():
            continue
        name = key.strip()

        # Read 'bytes', defaulting to 1 only when absent.
        if 'bytes' not in record:
            number = 1
        else:
            raw_bytes = record['bytes']
            if isinstance(raw_bytes, str):
                stripped = raw_bytes.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    number = int(stripped)
                else:
                    continue
            elif type(raw_bytes) is int:
                number = raw_bytes
            else:
                continue

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    result = []
    for name in sorted(groups):
        total, count = groups[name]
        result.append({'path': name, 'total': total, 'count': count})
    return result
