import re

_DELTA_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1 (before all other rules): for each nonempty string 'id',
    # retain only the last occurrence. Records without such an id
    # remain independent (all are kept).
    last_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                last_position[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                if last_position.get(identifier) != position:
                    continue
        else:
            # Skip non-object records.
            continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # Read 'delta', defaulting to 1 only when absent.
        if 'delta' in record:
            number = record['delta']
        else:
            number = 1
        if isinstance(number, bool):
            continue
        if isinstance(number, int):
            pass
        elif isinstance(number, str):
            text = number.strip()
            if not _DELTA_PATTERN.fullmatch(text):
                continue
            number = int(text, 10)
        else:
            continue

        groups[name] = groups.get(name, 0) + number

    # Sort lexicographically by 'sku'; [] when nothing remains.
    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
