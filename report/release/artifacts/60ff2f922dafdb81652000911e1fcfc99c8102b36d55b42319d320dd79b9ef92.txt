import re

def solve(value):
    last_occurrences = {}
    
    # 1. Deduplicate
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val != '':
                last_occurrences[id_val] = i
                
    retained = []
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val != '':
                if last_occurrences[id_val] != i:
                    continue # Skip non-last occurrences
        retained.append(record)
        
    groups = {}
    for record in retained:
        # 2. Skip non-object records and records whose 'active' is false
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
            
        # 3. Read 'sku', falling back to 'item' only when 'sku' is absent
        if 'sku' in record:
            name_val = record['sku']
        elif 'item' in record:
            name_val = record['item']
        else:
            continue
            
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace
        if not isinstance(name_val, str):
            continue
        name_val = name_val.strip()
        if not name_val:
            continue
            
        # 4. Read 'delta'
        if 'delta' not in record:
            delta = 1
        else:
            delta = record['delta']
            
        if isinstance(delta, bool):
            continue
        elif isinstance(delta, int):
            delta_val = delta
        elif isinstance(delta, str):
            stripped_delta = delta.strip()
            if re.fullmatch(r'[+-]?[0-9]+', stripped_delta):
                delta_val = int(stripped_delta, 10)
            else:
                continue
        else:
            continue
            
        # 5. Group by normalized 'sku'
        groups[name_val] = groups.get(name_val, 0) + delta_val
        
    # Sort results lexicographically by 'sku'
    result = []
    for name in sorted(groups.keys()):
        result.append({'sku': name, 'total': groups[name]})
        
    return result
