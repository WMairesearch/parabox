import re


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate before all other rules: keep only the last occurrence of
    # each nonempty string 'id'; records without such an id stay independent.
    seen_ids = set()
    deduped_reversed = []
    for record in reversed(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if record_id in seen_ids:
                    continue
                seen_ids.add(record_id)
        deduped_reversed.append(record)
    records = list(reversed(deduped_reversed))

    groups = {}

    for record in records:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            sku = record['sku']
        elif 'item' in record:
            sku = record['item']
        else:
            continue

        if not isinstance(sku, str):
            continue
        sku = sku.strip()
        if not sku:
            continue

        # Read 'delta', defaulting to 1 only when absent.
        if 'delta' not in record:
            delta = 1
        else:
            delta = record['delta']
            if type(delta) is int:
                pass
            elif isinstance(delta, str):
                text = delta.strip()
                if re.fullmatch(r'[+-]?[0-9]+', text):
                    delta = int(text)
                else:
                    continue
            else:
                continue

        groups[sku] = groups.get(sku, 0) + delta

    return [{'sku': sku, 'total': total} for sku, total in sorted(groups.items())]
