import re


def solve(value):
    # Deduplicate by nonempty string 'id' before all other rules.
    # Walk the input backwards so the first time we meet an id is its last
    # occurrence; keep that record and drop any earlier duplicates.
    retained = []
    seen_ids = set()
    for record in reversed(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if record_id in seen_ids:
                    continue
                seen_ids.add(record_id)
        retained.append(record)
    retained.reverse()

    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Use 'category' when present; otherwise fall back to 'kind'.
        if 'category' in record:
            key = record['category']
        else:
            key = record.get('kind')

        if not isinstance(key, str) or not key.strip():
            continue
        key = key.strip()

        # Priority defaults to 1 only when the key is absent.
        if 'priority' in record:
            number = record['priority']
            if isinstance(number, str):
                number = number.strip()
                if re.fullmatch(r'[+-]?[0-9]+', number):
                    number = int(number)
                else:
                    continue
            if type(number) is not int:
                continue
        else:
            number = 1

        if key in groups:
            if number > groups[key]:
                groups[key] = number
        else:
            groups[key] = number

    return [
        {'category': category, 'maximum': groups[category]}
        for category in sorted(groups)
    ]
