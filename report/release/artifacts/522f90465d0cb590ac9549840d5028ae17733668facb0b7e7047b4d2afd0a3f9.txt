import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def _last_positions(value):
    latest = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier:
            latest[identifier] = position
    return latest


def _delta(record):
    if 'delta' not in record:
        return 1
    number = record['delta']
    if isinstance(number, str):
        text = number.strip()
        if _INT_RE.fullmatch(text):
            return int(text, 10)
        return None
    if type(number) is int:
        return number
    return None


def solve(value):
    if not isinstance(value, list):
        return []
    latest = _last_positions(value)
    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != position:
            continue
        if record.get('active') is False:
            continue
        name = record['sku'] if 'sku' in record else record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        number = _delta(record)
        if number is None:
            continue
        groups[name] = groups.get(name, 0) + number
    return [{'sku': sku, 'total': groups[sku]} for sku in sorted(groups)]
