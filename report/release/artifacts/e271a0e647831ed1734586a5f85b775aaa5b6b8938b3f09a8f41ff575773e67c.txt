import re

def solve(value):
    if not value:
        return []

    # 1. Deduplicate
    last_occurrence = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                last_occurrence[rec_id] = i

    groups = {}
    for i, record in enumerate(value):
        # 1. Deduplicate rule application
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                if last_occurrence[rec_id] != i:
                    continue

        # 2. Skip non-object records
        if not isinstance(record, dict):
            continue
            
        # 3. Skip records whose 'active' is false
        if record.get('active') is False:
            continue
            
        # 4. Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            sku_val = record['sku']
        elif 'item' in record:
            sku_val = record['item']
        else:
            continue
            
        # 5. Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if not isinstance(sku_val, str):
            continue
        sku = sku_val.strip()
        if not sku:
            continue
            
        # 6. Read 'delta', defaulting to 1 only when absent.
        if 'delta' not in record:
            delta = 1
        else:
            delta_val = record['delta']
            if type(delta_val) is int:
                delta = delta_val
            elif isinstance(delta_val, str):
                delta_str = delta_val.strip()
                if re.fullmatch(r'[+-]?[0-9]+', delta_str):
                    delta = int(delta_str)
                else:
                    continue
            else:
                continue
                
        # 7. Group and sum
        groups[sku] = groups.get(sku, 0) + delta

    # 8. Return results
    result = []
    for sku in sorted(groups.keys()):
        result.append({'sku': sku, 'total': groups[sku]})
        
    return result
