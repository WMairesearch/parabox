import re

_BYTES_RE = re.compile(r'[+-]?[0-9]+')


def _dedupe(value):
    """Keep only the last occurrence of each nonempty string 'id'."""
    last = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            key = record.get('id')
            if isinstance(key, str) and key != '':
                last[key] = position
    retained = []
    for position, record in enumerate(value):
        if isinstance(record, dict):
            key = record.get('id')
            if isinstance(key, str) and key != '' and last[key] != position:
                continue
        retained.append(record)
    return retained


def solve(value):
    if not isinstance(value, list):
        return []

    records = _dedupe(value)

    groups = {}
    for record in records:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        if 'path' in record:
            name = record['path']
        elif 'route' in record:
            name = record['route']
        else:
            continue
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if 'bytes' in record:
            amount = record['bytes']
            if type(amount) is int:
                number = amount
            elif isinstance(amount, str):
                text = amount.strip()
                if _BYTES_RE.fullmatch(text) is None:
                    continue
                number = int(text, 10)
            else:
                continue
        else:
            number = 1

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': totals[0], 'count': totals[1]}
        for name, totals in sorted(groups.items())
    ]
