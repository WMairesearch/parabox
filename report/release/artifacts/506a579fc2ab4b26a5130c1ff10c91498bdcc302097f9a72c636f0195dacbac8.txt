import re

_DELTA_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Pass 1 (dedupe first): find last position of each nonempty string 'id'.
    last_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_position[rid] = position

    # Pass 2: process records in order.
    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        rid = record.get('id')
        if isinstance(rid, str) and rid != '' and last_position[rid] != position:
            continue
        if record.get('active') is False:
            continue
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'delta' in record:
            number = record['delta']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                text = number.strip()
                if _DELTA_RE.fullmatch(text):
                    number = int(text, 10)
                else:
                    continue
            else:
                continue
        else:
            number = 1
        groups[name] = groups.get(name, 0) + number

    return [{'sku': key, 'total': groups[key]} for key in sorted(groups)]
