import re


def solve(value):
    if not isinstance(value, list) or not value:
        return []

    # Rule: First retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Deduplicate before all other rules.
    last_seen_ids = set()
    deduped_reversed = []
    for record in reversed(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                if rec_id in last_seen_ids:
                    continue
                last_seen_ids.add(rec_id)
        deduped_reversed.append(record)

    deduped = list(reversed(deduped_reversed))

    # Rule: Skip non-object records and records whose 'active' is false.
    # Rule: Read 'path', falling back to 'route' only when 'path' is absent.
    # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
    # Rule: Read 'bytes', defaulting to 1 only when absent. Accept integers (not booleans)
    # and strings matching [+-]?[0-9]+ after stripping whitespace; convert strings to base-10 integers.
    # Skip all other values, including null.
    groups = {}
    for record in deduped:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        if 'path' in record:
            path_val = record['path']
        elif 'route' in record:
            path_val = record['route']
        else:
            continue

        if not isinstance(path_val, str):
            continue
        norm_path = path_val.strip()
        if not norm_path:
            continue

        if 'bytes' not in record:
            byte_count = 1
        else:
            raw_bytes = record['bytes']
            if type(raw_bytes) is int:
                byte_count = raw_bytes
            elif isinstance(raw_bytes, str):
                s = raw_bytes.strip()
                if re.fullmatch(r'[+-]?[0-9]+', s):
                    byte_count = int(s, 10)
                else:
                    continue
            else:
                continue

        if norm_path not in groups:
            groups[norm_path] = [0, 0]
        groups[norm_path][0] += byte_count
        groups[norm_path][1] += 1

    # Rule: Group retained records by normalized 'path'. Each output object contains
    # 'path' and the sum of bytes as 'total' and the number of retained records as 'count'.
    # Sort results lexicographically by 'path'. Return [] when no records remain.
    result = []
    for norm_path in sorted(groups.keys()):
        result.append({
            'path': norm_path,
            'total': groups[norm_path][0],
            'count': groups[norm_path][1],
        })
    return result
