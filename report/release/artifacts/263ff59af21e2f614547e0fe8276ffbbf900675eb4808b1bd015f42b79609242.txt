import re

_BYTES_RE = re.compile(r'[+-]?[0-9]+')


def _valid_id(record):
    ident = record.get('id')
    if isinstance(ident, str) and ident != '':
        return ident
    return None


def solve(value):
    if not isinstance(value, list):
        return []

    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            ident = _valid_id(record)
            if ident is not None:
                last_index[ident] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        ident = _valid_id(record)
        if ident is not None and last_index.get(ident) != position:
            continue
        if record.get('active') is False:
            continue
        if 'path' in record:
            raw_name = record['path']
        else:
            raw_name = record.get('route')
        if not isinstance(raw_name, str):
            continue
        name = raw_name.strip()
        if not name:
            continue
        if 'bytes' in record:
            raw_number = record['bytes']
            if isinstance(raw_number, bool):
                continue
            if type(raw_number) is int:
                number = raw_number
            elif isinstance(raw_number, str):
                text = raw_number.strip()
                if _BYTES_RE.fullmatch(text) is None:
                    continue
                number = int(text, 10)
            else:
                continue
        else:
            number = 1
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': total, 'count': count}
        for name, (total, count) in sorted(groups.items())
    ]
