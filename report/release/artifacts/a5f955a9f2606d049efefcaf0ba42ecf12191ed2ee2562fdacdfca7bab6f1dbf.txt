import re


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate before all other rules: for each nonempty string id,
    # retain only the record where that id occurs last.
    last_index = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_index[record_id] = index

    groups = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if last_index[record_id] != index:
                    continue
        elif not isinstance(record, dict):
            continue

        # Non-object records are skipped.  For objects, active must not be false.
        if record.get('active') is False:
            continue

        # Category, falling back to kind only when category is absent.
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')

        if not isinstance(name, str) or name.strip() == '':
            continue
        name = name.strip()

        # Priority defaults to 1 only when absent.
        if 'priority' in record:
            number = record['priority']
            if isinstance(number, str):
                stripped = number.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    number = int(stripped)
                else:
                    continue
            # Accept integers only, never booleans.
            if type(number) is not int:
                continue
        else:
            number = 1

        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [
        {'category': name, 'maximum': groups[name]}
        for name in sorted(groups)
    ]
