import re

def solve(value):
    seen_ids = set()
    retained = []
    for record in reversed(value):
        if isinstance(record, dict) and 'id' in record:
            record_id = record['id']
            if isinstance(record_id, str) and record_id != '':
                if record_id in seen_ids:
                    continue
                seen_ids.add(record_id)
        retained.append(record)
    retained.reverse()

    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        
        if 'path' in record:
            name_val = record['path']
        elif 'route' in record:
            name_val = record['route']
        else:
            continue
            
        if not isinstance(name_val, str):
            continue
        
        name = name_val.strip()
        if not name:
            continue
            
        if 'bytes' not in record:
            num = 1
        else:
            bytes_val = record['bytes']
            if isinstance(bytes_val, bool):
                continue
            elif isinstance(bytes_val, int):
                num = bytes_val
            elif isinstance(bytes_val, str):
                stripped = bytes_val.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    num = int(stripped)
                else:
                    continue
            else:
                continue
        
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + num, count + 1)
        
    result = []
    for name in sorted(groups.keys()):
        result.append({'path': name, 'total': groups[name][0], 'count': groups[name][1]})
    return result
