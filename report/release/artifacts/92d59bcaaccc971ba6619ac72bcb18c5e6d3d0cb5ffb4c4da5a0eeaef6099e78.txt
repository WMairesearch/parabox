import re

_BYTES_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1 (before all others): keep only the last occurrence of each
    # nonempty string 'id'; records without such an id stay independent.
    last_index = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_index[rid] = i

    groups = {}
    for i, record in enumerate(value):
        # Skip non-object records.
        if not isinstance(record, dict):
            continue
        # Deduplicate: drop every record that is not the last for its id.
        rid = record.get('id')
        if isinstance(rid, str) and rid != '' and last_index[rid] != i:
            continue
        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue
        # Read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        # Skip non-string or empty-after-stripping keys; strip whitespace.
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Read 'bytes', defaulting to 1 only when absent.
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                n = number
            elif isinstance(number, str):
                s = number.strip()
                if _BYTES_RE.fullmatch(s) is None:
                    continue
                n = int(s, 10)
            else:
                # All other values (null, float, list, dict, ...) are skipped.
                continue
        else:
            n = 1
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + n, count + 1)

    # Sort lexicographically by 'path'.
    return [{'path': name, 'total': t, 'count': c}
            for name, (t, c) in sorted(groups.items())]
