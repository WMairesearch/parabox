import re


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: for records with a nonempty string 'id', keep only
    # the last occurrence.  All other records remain independent.
    last_index_by_id = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        record_id = record.get('id')
        if isinstance(record_id, str) and record_id != '':
            last_index_by_id[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        record_id = record.get('id')
        if isinstance(record_id, str) and record_id != '':
            if last_index_by_id.get(record_id) != position:
                continue

        # Skip records whose 'active' member is JSON false.
        if record.get('active') is False:
            continue

        # Read 'category', falling back to 'kind' only when 'category' is absent.
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')

        if not isinstance(name, str) or not name.strip():
            continue
        name = name.strip()

        # Read 'priority', defaulting to 1 only when the key is absent.
        if 'priority' in record:
            number = record['priority']
        else:
            number = 1

        if isinstance(number, str):
            stripped = number.strip()
            if re.fullmatch(r'[+-]?[0-9]+', stripped):
                number = int(stripped)
            else:
                continue
        if type(number) is not int:
            continue

        previous = groups.get(name)
        if previous is None or number > previous:
            groups[name] = number

    return [{'category': name, 'maximum': groups[name]}
            for name in sorted(groups)]
