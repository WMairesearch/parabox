import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def _ident(record):
    """Return the record's id if it is a nonempty string, else None."""
    if not isinstance(record, dict):
        return None
    ident = record.get('id')
    if isinstance(ident, str) and ident != '':
        return ident
    return None


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: keep only the last occurrence of each nonempty
    # string id; records without such an id remain independent.
    last_index = {}
    for index, record in enumerate(value):
        ident = _ident(record)
        if ident is not None:
            last_index[ident] = index

    groups = {}
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        ident = _ident(record)
        if ident is not None and last_index.get(ident) != index:
            continue
        if record.get('active') is False:
            continue
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, str):
                stripped = number.strip()
                if not _INT_RE.fullmatch(stripped):
                    continue
                number = int(stripped, 10)
            elif type(number) is int:
                pass
            else:
                continue
        else:
            number = 1
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    result = []
    for name in sorted(groups):
        total, count = groups[name]
        result.append({'path': name, 'total': total, 'count': count})
    return result
