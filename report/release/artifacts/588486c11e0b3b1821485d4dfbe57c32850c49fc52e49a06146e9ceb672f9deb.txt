"""Group event records by normalized category, reporting the maximum priority.

Specification (from PROBLEM.md):
  * Deduplicate first: only the last occurrence of each nonempty string 'id' is
    retained; records lacking such an id stay independent.
  * Skip non-object records and records whose 'active' is exactly false.
  * 'category' wins over 'kind', which is used only when 'category' is absent.
  * Keys must be strings and nonempty after stripping; surrounding whitespace is
    stripped.
  * 'priority' defaults to 1 only when absent; integers (not booleans) and
    strings matching [+-]?[0-9]+ after stripping are accepted, strings being
    converted to base-10 integers.  Everything else (including null) is skipped.
  * Output is one {'category', 'maximum'} object per group, sorted
    lexicographically by 'category'.  The input is never mutated.
"""

import re

_INTEGER = re.compile(r'[+-]?[0-9]+')


def _identifier(record):
    """Return the record's deduplication id, or None if it has none."""
    value = record.get('id')
    if isinstance(value, str) and value:
        return value
    return None


def _priority(record):
    """Return the record's priority, or None when the value must be skipped."""
    if 'priority' not in record:
        return 1
    value = record['priority']
    if isinstance(value, bool):
        return None
    if isinstance(value, int):
        return int(value)
    if isinstance(value, str):
        text = value.strip()
        if _INTEGER.fullmatch(text):
            try:
                return int(text, 10)
            except ValueError:  # pragma: no cover - defensive
                return None
    return None


def _category(record):
    """Return the record's normalized category key, or None when unusable."""
    if 'category' in record:
        value = record['category']
    elif 'kind' in record:
        value = record['kind']
    else:
        return None
    if not isinstance(value, str):
        return None
    value = value.strip()
    if not value:
        return None
    return value


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1: deduplicate on the last occurrence of each nonempty string id.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = _identifier(record)
            if identifier is not None:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = _identifier(record)
        if identifier is not None and latest[identifier] != position:
            continue
        if record.get('active') is False:
            continue
        name = _category(record)
        if name is None:
            continue
        number = _priority(record)
        if number is None:
            continue
        previous = groups.get(name)
        if previous is None or number > previous:
            groups[name] = number

    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
