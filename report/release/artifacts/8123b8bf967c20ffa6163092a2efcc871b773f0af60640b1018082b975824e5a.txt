import re

_DELTA_PATTERN = re.compile(r'[+-]?[0-9]+')


def _is_usable_id(value):
    """An id is usable only when it is a nonempty string."""
    return isinstance(value, str) and len(value) > 0


def solve(value):
    records = value if isinstance(value, list) else []

    # Rule 1: keep only the last occurrence of each nonempty string 'id'.
    # This happens before every other rule.
    last_position = {}
    for position, record in enumerate(records):
        if isinstance(record, dict):
            identifier = record.get('id')
            if _is_usable_id(identifier):
                last_position[identifier] = position

    groups = {}
    for position, record in enumerate(records):
        # Rule 2: only object records survive.
        if not isinstance(record, dict):
            continue

        # Deduplication (already computed above).
        identifier = record.get('id')
        if _is_usable_id(identifier) and last_position.get(identifier) != position:
            continue

        # Rule 3: drop records explicitly marked inactive.
        if record.get('active') is False:
            continue

        # Rule 4: 'sku' wins; 'item' is only a fallback when 'sku' is absent.
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # Rule 5: 'delta' defaults to 1 only when the key is absent.
        if 'delta' in record:
            delta = record['delta']
        else:
            delta = 1
        if isinstance(delta, bool):
            continue
        if isinstance(delta, int):
            amount = delta
        elif isinstance(delta, str):
            text = delta.strip()
            if _DELTA_PATTERN.fullmatch(text) is None:
                continue
            amount = int(text, 10)
        else:
            continue

        groups[name] = groups.get(name, 0) + amount

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
