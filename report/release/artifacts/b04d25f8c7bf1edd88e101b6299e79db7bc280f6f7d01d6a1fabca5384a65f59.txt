import re


def solve(value):
    # Deduplicate before all other rules: keep only the last occurrence of
    # each nonempty-string "id".  Records without such an id stay independent.
    last_positions = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_positions[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        record_id = None
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if last_positions.get(record_id) != position:
                    continue

        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records whose active value is JSON false.
        if record.get('active') is False:
            continue

        # Read sku, falling back to item only when sku is absent.
        if 'sku' in record:
            key = record.get('sku')
        else:
            key = record.get('item')

        if not isinstance(key, str) or not key.strip():
            continue
        sku = key.strip()

        # Read delta, defaulting to 1 only when the key is absent.
        if 'delta' not in record:
            delta = 1
        else:
            raw = record.get('delta')
            if type(raw) is int:
                delta = raw
            elif isinstance(raw, str):
                raw = raw.strip()
                if re.fullmatch(r'[+-]?[0-9]+', raw):
                    delta = int(raw, 10)
                else:
                    continue
            else:
                continue

        groups[sku] = groups.get(sku, 0) + delta

    return [{'sku': sku, 'total': groups[sku]} for sku in sorted(groups)]
