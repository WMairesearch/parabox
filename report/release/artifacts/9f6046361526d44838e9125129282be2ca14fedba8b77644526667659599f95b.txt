import re

def solve(value):
    if not isinstance(value, list):
        return []

    # 1. Deduplicate
    last_occurrences = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != "":
                last_occurrences[rid] = i

    keep_indices = set(last_occurrences.values())

    filtered_value = []
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != "":
                if i not in keep_indices:
                    continue
        filtered_value.append(record)

    # 2. Process
    groups = {}
    for record in filtered_value:
        # Skip non-object records
        if not isinstance(record, dict):
            continue
        
        # Skip records whose 'active' is false
        if record.get('active') is False:
            continue
        
        # Read 'category'
        if 'category' in record:
            c = record['category']
        elif 'kind' in record:
            c = record['kind']
        else:
            continue
        
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace
        if not isinstance(c, str):
            continue
        c_stripped = c.strip()
        if not c_stripped:
            continue
        
        # Read 'priority'
        if 'priority' not in record:
            priority = 1
        else:
            p = record['priority']
            if isinstance(p, int) and not isinstance(p, bool):
                priority = p
            elif isinstance(p, str):
                p_stripped = p.strip()
                if re.fullmatch(r'[+-]?[0-9]+', p_stripped):
                    priority = int(p_stripped)
                else:
                    continue
            else:
                continue
                
        # Group
        if c_stripped not in groups:
            groups[c_stripped] = priority
        else:
            if priority > groups[c_stripped]:
                groups[c_stripped] = priority

    # Output
    result = [{'category': k, 'maximum': v} for k, v in sorted(groups.items())]
    return result
