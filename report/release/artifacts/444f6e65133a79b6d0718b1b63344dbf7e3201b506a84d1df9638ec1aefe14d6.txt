import re

_DELTA_RE = re.compile(r'^[+-]?[0-9]+$')


def _has_key(record, key):
    return key in record


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1: retain only the last occurrence of each nonempty string 'id'.
    # Records without such an id remain independent.
    last_with_id = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_with_id[rid] = position

    groups = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '' and last_with_id.get(rid) != position:
                continue
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        if _has_key(record, 'sku'):
            name = record.get('sku')
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if _has_key(record, 'delta'):
            number = record.get('delta')
            if type(number) is int:
                pass
            elif isinstance(number, str):
                text = number.strip()
                if not _DELTA_RE.match(text):
                    continue
                number = int(text, 10)
            else:
                continue
        else:
            number = 1

        groups[name] = groups.get(name, 0) + number

    result = []
    for name in sorted(groups):
        result.append({'sku': name, 'total': groups[name]})
    return result
