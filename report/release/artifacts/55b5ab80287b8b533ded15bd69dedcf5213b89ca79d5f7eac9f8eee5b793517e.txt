import re

_DELTA_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1 (first): retain only the last occurrence of each nonempty
    # string 'id'; records without such an id remain independent.
    last_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id:
                last_position[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        # Deduplication happens before all other rules.
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id:
                if last_position[record_id] != position:
                    continue

        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            key = record['sku']
        elif 'item' in record:
            key = record['item']
        else:
            key = None
        # Skip non-string or empty-after-stripping keys; strip whitespace.
        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue

        # Read 'delta', defaulting to 1 only when absent.
        if 'delta' in record:
            raw = record['delta']
            if isinstance(raw, bool):
                continue
            if isinstance(raw, int):
                delta = raw
            elif isinstance(raw, str):
                text = raw.strip()
                if not _DELTA_PATTERN.fullmatch(text):
                    continue
                delta = int(text, 10)
            else:
                continue
        else:
            delta = 1

        groups[key] = groups.get(key, 0) + delta

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
