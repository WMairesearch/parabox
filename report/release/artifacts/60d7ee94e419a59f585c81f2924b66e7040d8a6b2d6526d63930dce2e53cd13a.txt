import re


def solve(value):
    if not isinstance(value, list):
        return []

    # Step 1: Deduplicate by nonempty string 'id', retaining only the last occurrence.
    # Records without such an id remain independent. Deduplicate before all other rules.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        # Skip non-object records
        if not isinstance(record, dict):
            continue

        # Retain only the last occurrence of each nonempty string 'id'
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != position:
            continue

        # Skip records whose 'active' is false
        if record.get('active') is False:
            continue

        # Read 'category', falling back to 'kind' only when 'category' is absent.
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if 'category' in record:
            raw_cat = record['category']
        elif 'kind' in record:
            raw_cat = record['kind']
        else:
            continue

        if not isinstance(raw_cat, str):
            continue
        category = raw_cat.strip()
        if not category:
            continue

        # Read 'priority', defaulting to 1 only when absent.
        # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace;
        # convert strings to base-10 integers. Skip all other values, including null.
        if 'priority' not in record:
            priority = 1
        else:
            raw_priority = record['priority']
            if type(raw_priority) is int:
                priority = raw_priority
            elif isinstance(raw_priority, str):
                stripped = raw_priority.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    priority = int(stripped, 10)
                else:
                    continue
            else:
                continue

        # Group retained records by normalized 'category'
        if category not in groups or priority > groups[category]:
            groups[category] = priority

    # Sort results lexicographically by 'category'
    return [{'category': category, 'maximum': groups[category]} for category in sorted(groups)]
