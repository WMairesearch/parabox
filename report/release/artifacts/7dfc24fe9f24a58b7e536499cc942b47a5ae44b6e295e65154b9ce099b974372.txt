import re
import sys

if hasattr(sys, 'set_int_max_str_digits'):
    try:
        sys.set_int_max_str_digits(0)
    except Exception:
        pass


def solve(value):
    if not isinstance(value, list):
        return []

    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != position:
            continue

        if record.get('active') is False:
            continue

        if 'path' in record:
            name = record['path']
        elif 'route' in record:
            name = record['route']
        else:
            continue

        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if 'bytes' not in record:
            number = 1
        else:
            raw_bytes = record['bytes']
            if isinstance(raw_bytes, bool):
                continue
            elif isinstance(raw_bytes, int):
                number = raw_bytes
            elif isinstance(raw_bytes, str):
                s = raw_bytes.strip()
                if re.fullmatch(r'[+-]?[0-9]+', s):
                    try:
                        number = int(s, 10)
                    except (ValueError, OverflowError):
                        continue
                else:
                    continue
            else:
                continue

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    result = []
    for name in sorted(groups.keys()):
        result.append({
            'path': name,
            'total': groups[name][0],
            'count': groups[name][1]
        })
    return result
