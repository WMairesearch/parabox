import re

def solve(value):
    # Deduplicate before all other rules
    last_idx = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                last_idx[rec_id] = i

    groups = {}
    
    for i, record in enumerate(value):
        # 1. Deduplication check
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                if last_idx[rec_id] != i:
                    continue
        
        # 2. Skip non-object records
        if not isinstance(record, dict):
            continue
            
        # 3. Skip records whose 'active' is false
        if record.get('active') is False:
            continue
            
        # 4. Read 'path', falling back to 'route'
        if 'path' in record:
            name = record['path']
        elif 'route' in record:
            name = record['route']
        else:
            continue
            
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
            
        # 5. Read 'bytes', defaulting to 1 only when absent
        if 'bytes' not in record:
            number = 1
        else:
            bytes_val = record['bytes']
            if type(bytes_val) is int:
                number = bytes_val
            elif isinstance(bytes_val, str):
                s = bytes_val.strip()
                if re.fullmatch(r'[+-]?[0-9]+', s):
                    number = int(s)
                else:
                    continue
            else:
                continue
                
        # 6. Group by normalized 'path'
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)
        
    result = []
    for name in sorted(groups.keys()):
        result.append({'path': name, 'total': groups[name][0], 'count': groups[name][1]})
    return result
