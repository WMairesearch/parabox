import re

_INT = re.compile(r'[+-]?[0-9]+')


def solve(value):
    """Group event records by normalized category, keeping the maximum priority."""
    if not isinstance(value, list):
        return []

    # Rule 1: deduplicate BEFORE every other rule. Keep only the last
    # occurrence of each nonempty string 'id'; everything else is independent.
    last_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            key = record.get('id')
            if isinstance(key, str) and key != '':
                last_position[key] = position

    groups = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            key = record.get('id')
            if isinstance(key, str) and key != '' and last_position[key] != position:
                continue
        else:
            # Rule 2: non-object records are skipped.
            continue

        # Rule 2: records whose 'active' is false are skipped.
        if record.get('active') is False:
            continue

        # Rule 3: 'category', falling back to 'kind' only when absent.
        if 'category' in record:
            name = record['category']
        elif 'kind' in record:
            name = record['kind']
        else:
            continue

        # Rule 4: key must be a non-empty string after stripping.
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # Rule 5: priority defaults to 1 only when absent; ints (not bools)
        # and integer strings are accepted, everything else skips the record.
        if 'priority' in record:
            raw = record['priority']
            if type(raw) is int:
                number = raw
            elif isinstance(raw, str):
                text = raw.strip()
                if _INT.fullmatch(text) is None:
                    continue
                number = int(text)
            else:
                continue
        else:
            number = 1

        best = groups.get(name)
        if best is None or number > best:
            groups[name] = number

    # Rule 6: sorted lexicographically by category.
    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
