import re

_DELTA_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Pass 1: dedupe by nonempty string 'id', retaining the last occurrence.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    # Pass 2: filter, normalize, and group.
    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != position:
            continue
        if record.get('active') is False:
            continue
        name = record.get('sku', record.get('item'))
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        number = record['delta'] if 'delta' in record else 1
        if isinstance(number, str):
            stripped = number.strip()
            if not _DELTA_PATTERN.fullmatch(stripped):
                continue
            number = int(stripped, 10)
        elif type(number) is int:
            pass
        else:
            continue
        groups[name] = groups.get(name, 0) + number

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
