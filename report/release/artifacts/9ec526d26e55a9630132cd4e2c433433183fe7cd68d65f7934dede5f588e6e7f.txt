import re

_PRIORITY = re.compile(r'[+-]?[0-9]+')


def _records(value):
    if isinstance(value, (str, bytes)) or value is None:
        return []
    if isinstance(value, list):
        return value
    try:
        return list(value)
    except TypeError:
        return []


def solve(value):
    records = _records(value)

    # Rule 1: deduplicate by id (last occurrence wins), before all other rules.
    last = {}
    for position, record in enumerate(records):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                last[identifier] = position

    groups = {}
    for position, record in enumerate(records):
        # Rule 2: skip non-object records.
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier != '' and last[identifier] != position:
            continue
        # Rule 2b: skip records whose 'active' is false.
        if record.get('active') is False:
            continue
        # Rule 3: category, falling back to kind only when category is absent.
        if 'category' in record:
            name = record.get('category')
        else:
            name = record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if name == '':
            continue
        # Rule 4: priority, defaulting to 1 only when absent.
        if 'priority' in record:
            number = record.get('priority')
        else:
            number = 1
        if isinstance(number, str):
            text = number.strip()
            if _PRIORITY.fullmatch(text) is not None:
                number = int(text, 10)
        if type(number) is not int:
            continue
        # Rule 5: group by normalized category, keep maximum priority.
        current = groups.get(name)
        if current is None or number > current:
            groups[name] = number

    # Rule 6: sort lexicographically by category.
    return [{'category': key, 'maximum': groups[key]} for key in sorted(groups)]
