import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def _bytes_of(record):
    """Return the byte value for a record, or None when it must be skipped."""
    if 'bytes' not in record:
        return 1
    value = record['bytes']
    if isinstance(value, bool):
        return None
    if isinstance(value, int):
        return value
    if isinstance(value, str):
        text = value.strip()
        if _INT_RE.fullmatch(text):
            return int(text, 10)
    return None


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: keep only the last occurrence of each nonempty
    # string id; records without such an id remain independent.
    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            key = record.get('id')
            if isinstance(key, str) and key:
                last_index[key] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        key = record.get('id')
        if isinstance(key, str) and key and last_index.get(key) != position:
            continue
        if record.get('active') is False:
            continue

        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        amount = _bytes_of(record)
        if amount is None:
            continue

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + amount, count + 1)

    return [
        {'path': name, 'total': groups[name][0], 'count': groups[name][1]}
        for name in sorted(groups)
    ]
