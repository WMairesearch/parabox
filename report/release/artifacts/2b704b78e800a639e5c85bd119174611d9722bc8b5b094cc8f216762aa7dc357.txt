import re

def solve(value):
    last_occurrence = {}
    for i, record in enumerate(value):
        if isinstance(record, dict) and 'id' in record:
            rec_id = record['id']
            if isinstance(rec_id, str) and rec_id != '':
                last_occurrence[rec_id] = i

    groups = {}
    for i, record in enumerate(value):
        # Rule 2: Deduplication
        if isinstance(record, dict) and 'id' in record:
            rec_id = record['id']
            if isinstance(rec_id, str) and rec_id != '':
                if last_occurrence[rec_id] != i:
                    continue
        
        # Rule 3: Skip non-object records and records whose 'active' is false
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Rule 4: Read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            key = record['path']
        elif 'route' in record:
            key = record['route']
        else:
            continue
            
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue

        # Rule 5: Read 'bytes', defaulting to 1 only when absent. 
        if 'bytes' not in record:
            b_val = 1
        else:
            b_val = record['bytes']
            if isinstance(b_val, str):
                b_val = b_val.strip()
                if re.fullmatch(r'[+-]?[0-9]+', b_val):
                    b_val = int(b_val)
                else:
                    continue
            elif type(b_val) is int:
                pass
            else:
                continue

        # Rule 6: Group retained records by normalized 'path'
        if key not in groups:
            groups[key] = {'total': 0, 'count': 0}
        groups[key]['total'] += b_val
        groups[key]['count'] += 1

    result = []
    for key in sorted(groups.keys()):
        result.append({
            'path': key,
            'total': groups[key]['total'],
            'count': groups[key]['count']
        })
    return result
