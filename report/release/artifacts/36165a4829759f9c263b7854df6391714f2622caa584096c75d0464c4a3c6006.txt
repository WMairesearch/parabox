import re


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []

    # Step 1: Retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Deduplicate before all other rules.
    last_seen = {}
    for idx, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                last_seen[rec_id] = idx

    retained_records = []
    for idx, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                if last_seen[rec_id] != idx:
                    continue
        retained_records.append(record)

    # Step 2: Filter and group retained records
    groups = {}
    for record in retained_records:
        # Skip non-object records and records whose 'active' is false
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Read 'path', falling back to 'route' only when 'path' is absent
        if 'path' in record:
            raw_path = record['path']
        elif 'route' in record:
            raw_path = record['route']
        else:
            continue

        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace
        if not isinstance(raw_path, str):
            continue
        path = raw_path.strip()
        if not path:
            continue

        # Read 'bytes', defaulting to 1 only when absent.
        # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace;
        # convert strings to base-10 integers. Skip all other values, including null.
        if 'bytes' not in record:
            bytes_val = 1
        else:
            raw_bytes = record['bytes']
            if isinstance(raw_bytes, bool):
                continue
            elif isinstance(raw_bytes, int):
                bytes_val = raw_bytes
            elif isinstance(raw_bytes, str):
                stripped_bytes = raw_bytes.strip()
                if not re.fullmatch(r'[+-]?[0-9]+', stripped_bytes):
                    continue
                bytes_val = int(stripped_bytes, 10)
            else:
                continue

        if path not in groups:
            groups[path] = {'total': 0, 'count': 0}
        groups[path]['total'] += bytes_val
        groups[path]['count'] += 1

    result = []
    for path in sorted(groups.keys()):
        result.append({
            'path': path,
            'total': groups[path]['total'],
            'count': groups[path]['count']
        })
    return result
