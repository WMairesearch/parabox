import re


def solve(value):
    # First retain only the last occurrence of each nonempty string id.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                latest[identifier] = position

    groups = {}

    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        # Deduplicate before all other rules.
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier != '':
            if latest[identifier] != position:
                continue

        if record.get('active') is False:
            continue

        # Read sku, falling back to item only when sku is absent.
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')

        if not isinstance(name, str) or not name.strip():
            continue
        name = name.strip()

        # Read delta, defaulting to 1 only when absent.
        if 'delta' in record:
            number = record['delta']
        else:
            number = 1

        if isinstance(number, str):
            stripped = number.strip()
            if re.fullmatch(r'[+-]?[0-9]+', stripped):
                number = int(stripped, 10)
            else:
                continue

        if type(number) is not int:
            continue

        groups[name] = groups.get(name, 0) + number

    return [
        {'sku': name, 'total': groups[name]}
        for name in sorted(groups)
    ]
