import re

def solve(value):
    seen_ids = set()
    retained_records = []
    
    for record in reversed(value):
        if isinstance(record, dict) and 'id' in record and isinstance(record['id'], str) and record['id'] != "":
            record_id = record['id']
            if record_id in seen_ids:
                continue
            seen_ids.add(record_id)
        retained_records.append(record)
    
    retained_records.reverse()
    
    groups = {}
    for record in retained_records:
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        if 'sku' in record:
            key = record['sku']
        elif 'item' in record:
            key = record['item']
        else:
            continue
            
        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue
            
        if 'delta' not in record:
            delta = 1
        else:
            delta = record['delta']
            
        if type(delta) is int:
            parsed_delta = delta
        elif isinstance(delta, str):
            delta_str = delta.strip()
            if re.fullmatch(r'[+-]?[0-9]+', delta_str):
                parsed_delta = int(delta_str)
            else:
                continue
        else:
            continue
            
        groups[key] = groups.get(key, 0) + parsed_delta
        
    result = []
    for key in sorted(groups.keys()):
        result.append({'sku': key, 'total': groups[key]})
        
    return result
