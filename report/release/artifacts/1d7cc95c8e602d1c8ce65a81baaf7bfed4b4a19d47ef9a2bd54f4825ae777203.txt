import re
import json

def solve(value):
    seen_ids = set()
    retained = []
    # reverse iterate
    for i in range(len(value)-1, -1, -1):
        record = value[i]
        is_independent = True
        if isinstance(record, dict) and 'id' in record:
            id_val = record['id']
            if isinstance(id_val, str) and id_val != '':
                is_independent = False
                if id_val not in seen_ids:
                    seen_ids.add(id_val)
                    retained.append(record)
        if is_independent:
            retained.append(record)
            
    retained.reverse()
    
    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
            
        if 'path' in record:
            path_val = record['path']
        elif 'route' in record:
            path_val = record['route']
        else:
            continue
            
        if not isinstance(path_val, str):
            continue
        path_val = path_val.strip()
        if not path_val:
            continue
            
        if 'bytes' not in record:
            byte_count = 1
        else:
            b = record['bytes']
            if type(b) is int:
                byte_count = b
            elif isinstance(b, str):
                b_str = b.strip()
                if re.match(r'^[+-]?[0-9]+$', b_str):
                    byte_count = int(b_str)
                else:
                    continue
            else:
                continue
                
        if path_val not in groups:
            groups[path_val] = {'count': 0, 'total': 0}
        groups[path_val]['count'] += 1
        groups[path_val]['total'] += byte_count
        
    result = []
    for path_val in sorted(groups.keys()):
        result.append({
            'path': path_val,
            'total': groups[path_val]['total'],
            'count': groups[path_val]['count']
        })
    return result
