import re

def solve(value):
    retained = []
    seen_ids = set()
    for record in reversed(value):
        if isinstance(record, dict) and 'id' in record:
            rec_id = record['id']
            if isinstance(rec_id, str) and rec_id != '':
                if rec_id in seen_ids:
                    continue
                seen_ids.add(rec_id)
        retained.append(record)
    
    retained.reverse()
    
    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        if 'sku' in record:
            sku_val = record['sku']
        elif 'item' in record:
            sku_val = record['item']
        else:
            continue
            
        if not isinstance(sku_val, str):
            continue
        
        sku_val = sku_val.strip()
        if not sku_val:
            continue
            
        if 'delta' not in record:
            delta_val = 1
        else:
            raw_delta = record['delta']
            if type(raw_delta) is int:
                delta_val = raw_delta
            elif isinstance(raw_delta, str):
                raw_delta_stripped = raw_delta.strip()
                if re.fullmatch(r'[+-]?[0-9]+', raw_delta_stripped):
                    delta_val = int(raw_delta_stripped)
                else:
                    continue
            else:
                continue
                
        groups[sku_val] = groups.get(sku_val, 0) + delta_val
        
    result = []
    for sku in sorted(groups.keys()):
        result.append({'sku': sku, 'total': groups[sku]})
        
    return result
