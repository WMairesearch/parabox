"""Group event records by normalized category and report the maximum priority."""

import re

_INTEGER = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []

    # Rule 1: keep only the LAST occurrence of each nonempty string id.
    # Records without such an id stay independent.  Done before every other rule.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        # Rule 2: skip non-object records.
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if (isinstance(identifier, str) and identifier != ''
                and latest[identifier] != position):
            continue
        # Rule 3: skip records whose 'active' is exactly boolean false.
        if record.get('active') is False:
            continue
        # Rule 4: 'category', falling back to 'kind' only when 'category' is absent.
        name = record['category'] if 'category' in record else record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if name == '':
            continue
        # Rule 5: 'priority', defaulting to 1 only when the key is absent.
        if 'priority' in record:
            number = record['priority']
            if isinstance(number, str):
                text = number.strip()
                if not _INTEGER.fullmatch(text):
                    continue
                number = int(text, 10)
            if type(number) is not int:      # rejects bools, floats, None, ...
                continue
        else:
            number = 1
        if name not in groups or number > groups[name]:
            groups[name] = number

    # Rule 6: sorted lexicographically by category.
    return [{'category': key, 'maximum': groups[key]} for key in sorted(groups)]
