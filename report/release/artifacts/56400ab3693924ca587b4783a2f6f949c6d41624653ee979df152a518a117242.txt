import re


def solve(value):
    if not isinstance(value, list):
        return []

    # First retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Deduplicate before all other rules.
    last_id_indices = {}
    for idx, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id:
                last_id_indices[rec_id] = idx

    groups = {}
    for idx, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        rec_id = record.get('id')
        if isinstance(rec_id, str) and rec_id and last_id_indices[rec_id] != idx:
            continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'category', falling back to 'kind' only when 'category' is absent.
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if 'category' in record:
            raw_cat = record['category']
        elif 'kind' in record:
            raw_cat = record['kind']
        else:
            continue

        if not isinstance(raw_cat, str):
            continue
        cat = raw_cat.strip()
        if not cat:
            continue

        # Read 'priority', defaulting to 1 only when absent.
        # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace;
        # convert strings to base-10 integers. Skip all other values, including null.
        if 'priority' not in record:
            priority = 1
        else:
            raw_p = record['priority']
            if type(raw_p) is int:
                priority = raw_p
            elif isinstance(raw_p, str):
                p_str = raw_p.strip()
                if re.fullmatch(r'[+-]?[0-9]+', p_str):
                    priority = int(p_str, 10)
                else:
                    continue
            else:
                continue

        # Group retained records by normalized 'category'.
        # Each output object contains 'category' and the maximum priority as 'maximum'.
        if cat in groups:
            if priority > groups[cat]:
                groups[cat] = priority
        else:
            groups[cat] = priority

    # Sort results lexicographically by 'category'.
    return [{'category': cat, 'maximum': groups[cat]} for cat in sorted(groups)]
