import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def _normalize_path(record):
    if 'path' in record:
        raw = record['path']
    elif 'route' in record:
        raw = record['route']
    else:
        return None
    if not isinstance(raw, str):
        return None
    stripped = raw.strip()
    if not stripped:
        return None
    return stripped


def _read_bytes(record):
    """Return (ok, value). Absent -> 1; only ints (not bools) and integer strings."""
    if 'bytes' not in record:
        return True, 1
    raw = record['bytes']
    if isinstance(raw, bool):
        return False, None
    if isinstance(raw, int):
        return True, raw
    if isinstance(raw, str):
        text = raw.strip()
        if _INT_RE.fullmatch(text):
            try:
                return True, int(text, 10)
            except ValueError:
                return False, None
    return False, None


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: keep only the last occurrence of each nonempty string id.
    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                last_index[identifier] = position

    groups = {}
    order = []
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier != '':
            if last_index.get(identifier) != position:
                continue
        if record.get('active') is False:
            continue
        path = _normalize_path(record)
        if path is None:
            continue
        ok, number = _read_bytes(record)
        if not ok:
            continue
        if path in groups:
            total, count = groups[path]
            groups[path] = (total + number, count + 1)
        else:
            groups[path] = (number, 1)
            order.append(path)

    return [
        {'path': path, 'total': groups[path][0], 'count': groups[path][1]}
        for path in sorted(groups)
    ]
