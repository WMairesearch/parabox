import re

_BYTES_RE = re.compile(r'[+-]?[0-9]+\Z')


def _bytes_value(record):
    """Return (ok, amount) for the record's 'bytes' field."""
    if 'bytes' not in record:
        return True, 1
    value = record['bytes']
    if isinstance(value, bool):
        return False, 0
    if isinstance(value, int):
        return True, int(value)
    if isinstance(value, str):
        text = value.strip()
        if _BYTES_RE.match(text):
            return True, int(text, 10)
    return False, 0


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: keep only the last occurrence of each nonempty string id.
    seen = set()
    retained = []
    for record in reversed(value):
        identity = None
        if isinstance(record, dict):
            raw_id = record.get('id')
            if isinstance(raw_id, str) and raw_id != '':
                identity = raw_id
        if identity is not None:
            if identity in seen:
                continue
            seen.add(identity)
        retained.append(record)
    retained.reverse()

    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        ok, amount = _bytes_value(record)
        if not ok:
            continue

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + amount, count + 1)

    return [
        {'path': name, 'total': groups[name][0], 'count': groups[name][1]}
        for name in sorted(groups)
    ]
