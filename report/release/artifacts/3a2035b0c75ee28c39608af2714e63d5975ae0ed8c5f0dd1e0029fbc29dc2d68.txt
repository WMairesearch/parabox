import re

_BYTES_TEXT = re.compile(r'[+-]?[0-9]+')


def _retained_after_dedup(records):
    # Retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Dedup precedes all
    # other rules, so every record participates regardless of validity.
    last = {}
    for position, record in enumerate(records):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                last[identifier] = position
    retained = []
    for position, record in enumerate(records):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                if last[identifier] != position:
                    continue
        retained.append(record)
    return retained


def solve(value):
    records = value if isinstance(value, list) else []
    groups = {}
    for record in _retained_after_dedup(records):
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        if 'path' in record:
            name = record['path']
        elif 'route' in record:
            name = record['route']
        else:
            continue
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'bytes' not in record:
            number = 1
        else:
            raw = record['bytes']
            if isinstance(raw, bool):
                continue
            if isinstance(raw, int):
                number = raw
            elif isinstance(raw, str):
                text = raw.strip()
                if not _BYTES_TEXT.fullmatch(text):
                    continue
                number = int(text, 10)
            else:
                continue
        entry = groups.get(name)
        if entry is None:
            groups[name] = [number, 1]
        else:
            entry[0] += number
            entry[1] += 1
    result = [
        {'path': name, 'total': totals[0], 'count': totals[1]}
        for name, totals in groups.items()
    ]
    result.sort(key=lambda item: item['path'])
    return result
