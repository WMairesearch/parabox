import re

def solve(value):
    # Deduplicate before all other rules.
    last_occurrence = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                last_occurrence[rec_id] = i
                
    retained = []
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                if last_occurrence[rec_id] == i:
                    retained.append(record)
            else:
                retained.append(record)
        else:
            retained.append(record)
            
    # Process retained records
    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
            
        if 'path' in record:
            key_val = record['path']
        elif 'route' in record:
            key_val = record['route']
        else:
            continue
            
        if not isinstance(key_val, str):
            continue
        key_val = key_val.strip()
        if key_val == '':
            continue
            
        if 'bytes' not in record:
            bytes_val = 1
        else:
            bytes_val = record['bytes']
            
        if type(bytes_val) is int:
            num_bytes = bytes_val
        elif isinstance(bytes_val, str):
            bytes_val_stripped = bytes_val.strip()
            if re.fullmatch(r'[+-]?[0-9]+', bytes_val_stripped):
                num_bytes = int(bytes_val_stripped)
            else:
                continue
        else:
            continue
            
        if key_val not in groups:
            groups[key_val] = {'total': 0, 'count': 0}
        groups[key_val]['total'] += num_bytes
        groups[key_val]['count'] += 1
        
    result = []
    for key in sorted(groups.keys()):
        result.append({
            'path': key,
            'total': groups[key]['total'],
            'count': groups[key]['count']
        })
        
    return result
