import re

_DELTA_PATTERN = re.compile(r'^[+-]?[0-9]+$')
_MISSING = object()


def solve(value):
    # Rule 1 (before all others): retain only the last occurrence of each
    # nonempty string 'id'; records without such an id remain independent.
    last_index_for_id = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id:
                last_index_for_id[record_id] = index

    groups = {}
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        record_id = record.get('id')
        if isinstance(record_id, str) and record_id:
            if index != last_index_for_id[record_id]:
                continue  # superseded by a later record with the same id

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            name = record['sku']
        elif 'item' in record:
            name = record['item']
        else:
            continue
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # Read 'delta', defaulting to 1 only when absent.
        if 'delta' in record:
            number = record['delta']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                text = number.strip()
                if not _DELTA_PATTERN.match(text):
                    continue
                number = int(text, 10)
            else:
                continue
        else:
            number = 1

        groups[name] = groups.get(name, 0) + number

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
