import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Deduplicate first: retain only the last occurrence of each record with a
    # nonempty string 'id'; records without such an id remain independent.
    last_index = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_index[rid] = i

    groups = {}
    for i, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        rid = record.get('id')
        if isinstance(rid, str) and rid != '' and last_index[rid] != i:
            continue
        if record.get('active') is False:
            continue
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                text = number.strip()
                if _INT_RE.fullmatch(text):
                    number = int(text)
                else:
                    continue
            else:
                continue
        else:
            number = 1
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    result = [{'path': name, 'total': totals[0], 'count': totals[1]}
              for name, totals in groups.items()]
    result.sort(key=lambda item: item['path'])
    return result
