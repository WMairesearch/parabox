import re


def solve(value):
    # Rule 1: deduplicate first - retain only the last occurrence of each
    # nonempty string 'id'; records without such an id remain independent.
    last_index = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_index[rid] = i

    groups = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '' and last_index[rid] != i:
                continue  # superseded by a later record with the same id
        else:
            continue  # skip non-object records

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'category', falling back to 'kind' only when 'category' is absent.
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # Read 'priority', defaulting to 1 only when absent.
        if 'priority' in record:
            number = record['priority']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                stripped = number.strip()
                if not re.fullmatch(r'[+-]?[0-9]+', stripped):
                    continue
                number = int(stripped)
            else:
                continue
        else:
            number = 1

        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
