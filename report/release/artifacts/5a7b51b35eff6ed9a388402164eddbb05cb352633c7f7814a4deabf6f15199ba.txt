import re

_NUMBER = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: keep only the last record for each nonempty string id.
    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                last_index[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier != '':
            if last_index[identifier] != position:
                continue

        if record.get('active') is False:
            continue

        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')

        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if 'priority' in record:
            number = record['priority']
            if isinstance(number, str):
                text = number.strip()
                if _NUMBER.fullmatch(text) is None:
                    continue
                number = int(text, 10)
            elif type(number) is not int:
                continue
        else:
            number = 1

        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [
        {'category': name, 'maximum': groups[name]}
        for name in sorted(groups)
    ]
