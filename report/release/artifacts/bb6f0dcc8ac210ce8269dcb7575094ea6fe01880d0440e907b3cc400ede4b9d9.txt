"""Events grouping.

Rules (per PROBLEM.md):
  1. Deduplicate first: keep only the last occurrence of each nonempty string
     'id'; records without such an id stay independent.
  2. Skip non-object records and records whose 'active' is false.
  3. Category comes from 'category', falling back to 'kind' only when
     'category' is absent.  Skip non-string / blank-after-strip values;
     otherwise strip surrounding whitespace.
  4. Priority defaults to 1 only when absent.  Accept ints (not bools) and
     strings matching [+-]?[0-9]+ after stripping; skip everything else
     (including null, floats, bools, lists, objects).
  5. Group by normalized category, emit {'category', 'maximum'} with the
     maximum priority, sorted lexicographically by category.
The input is never modified.
"""

import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def _dedupe(value):
    """Return records keeping only the last occurrence of each string id."""
    last_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            key = record.get('id')
            if isinstance(key, str) and key != '':
                last_position[key] = position
    if not last_position:
        return list(value)
    kept = []
    for position, record in enumerate(value):
        if isinstance(record, dict):
            key = record.get('id')
            if isinstance(key, str) and key != '':
                if last_position[key] != position:
                    continue
        kept.append(record)
    return kept


def _priority(record):
    """Return the integer priority, or None when the record must be skipped."""
    if 'priority' not in record:
        return 1
    raw = record['priority']
    if isinstance(raw, bool):
        return None
    if isinstance(raw, int):
        return int(raw)
    if isinstance(raw, str):
        text = raw.strip()
        if _INT_RE.fullmatch(text):
            return int(text, 10)
    return None


def _category(record):
    """Return the stripped category string, or None when unusable."""
    if 'category' in record:
        name = record['category']
    else:
        name = record.get('kind')
    if not isinstance(name, str):
        return None
    name = name.strip()
    if not name:
        return None
    return name


def solve(value):
    if not isinstance(value, list):
        return []
    groups = {}
    for record in _dedupe(value):
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        name = _category(record)
        if name is None:
            continue
        number = _priority(record)
        if number is None:
            continue
        if name not in groups or number > groups[name]:
            groups[name] = number
    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
