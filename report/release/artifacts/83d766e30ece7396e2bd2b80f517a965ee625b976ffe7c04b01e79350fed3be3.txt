import re

def solve(value):
    # Step 1: deduplicate
    last_occurrences = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val:
                last_occurrences[id_val] = i

    groups = {}
    for i, record in enumerate(value):
        # Apply deduplication rule
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val:
                if i != last_occurrences[id_val]:
                    continue
        
        # Rule 2: Skip non-object records and records whose 'active' is false
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
            
        # Rule 3: Read 'category', fallback to 'kind' only when 'category' is absent
        if 'category' in record:
            name = record['category']
        elif 'kind' in record:
            name = record['kind']
        else:
            continue
            
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
            
        # Rule 4: Read 'priority', default to 1 only when absent
        if 'priority' not in record:
            priority = 1
        else:
            p_val = record['priority']
            if type(p_val) is int:
                priority = p_val
            elif type(p_val) is str:
                p_val_stripped = p_val.strip()
                if re.fullmatch(r'[+-]?[0-9]+', p_val_stripped):
                    priority = int(p_val_stripped)
                else:
                    continue
            else:
                continue
                
        # Group and max priority
        if name not in groups or priority > groups[name]:
            groups[name] = priority
            
    result = [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
    return result
