import re

def solve(value):
    ids_seen = set()
    kept_indices = set()
    
    for i in range(len(value) - 1, -1, -1):
        record = value[i]
        if isinstance(record, dict) and 'id' in record:
            record_id = record['id']
            if isinstance(record_id, str) and record_id:
                if record_id not in ids_seen:
                    ids_seen.add(record_id)
                    kept_indices.add(i)
                continue
        kept_indices.add(i)
        
    groups = {}
    
    for i in range(len(value)):
        if i not in kept_indices:
            continue
            
        record = value[i]
        
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        if 'path' in record:
            path = record['path']
        elif 'route' in record:
            path = record['route']
        else:
            continue
            
        if not isinstance(path, str):
            continue
            
        path = path.strip()
        if not path:
            continue
            
        if 'bytes' not in record:
            bytes_val = 1
        else:
            bytes_val = record['bytes']
            
        if type(bytes_val) is int:
            num = bytes_val
        elif isinstance(bytes_val, str):
            s = bytes_val.strip()
            if re.fullmatch(r'[+-]?[0-9]+', s):
                num = int(s)
            else:
                continue
        else:
            continue
            
        total, count = groups.get(path, (0, 0))
        groups[path] = (total + num, count + 1)
        
    result = []
    for path in sorted(groups.keys()):
        result.append({'path': path, 'total': groups[path][0], 'count': groups[path][1]})
        
    return result
