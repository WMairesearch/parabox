import re

def solve(value):
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position
                
    groups = {}
    for position, record in enumerate(value):
        # First retain only the last occurrence of each nonempty string 'id'
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier and latest[identifier] != position:
                continue
                
        # Skip non-object records
        if not isinstance(record, dict):
            continue
            
        # Skip records whose 'active' is false
        if record.get('active') is False:
            continue
            
        # Read 'path', falling back to 'route' only when 'path' is absent
        if 'path' in record:
            name = record['path']
        elif 'route' in record:
            name = record['route']
        else:
            name = None
            
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace
        if type(name) is not str:
            continue
        name = name.strip()
        if not name:
            continue
            
        # Read 'bytes', defaulting to 1 only when absent
        if 'bytes' in record:
            number = record['bytes']
        else:
            number = 1
            
        # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace
        if type(number) is int:
            pass
        elif type(number) is str:
            stripped = number.strip()
            if re.fullmatch(r'[+-]?[0-9]+', stripped):
                number = int(stripped)
            else:
                continue
        else:
            continue
            
        if name not in groups:
            groups[name] = {'total': 0, 'count': 0}
        groups[name]['total'] += number
        groups[name]['count'] += 1
        
    result = []
    for name in sorted(groups.keys()):
        result.append({'path': name, 'total': groups[name]['total'], 'count': groups[name]['count']})
        
    return result
