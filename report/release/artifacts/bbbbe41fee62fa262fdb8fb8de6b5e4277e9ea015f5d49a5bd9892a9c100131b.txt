import re

_DELTA_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1 (applied first): retain only the last occurrence of each
    # nonempty string 'id'; records without such an id remain independent.
    last_occurrence = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_occurrence[record_id] = index

    groups = {}
    for index, record in enumerate(value):
        # Skip non-object records.
        if not isinstance(record, dict):
            continue
        # Drop records superseded by a later record with the same id.
        record_id = record.get('id')
        if (isinstance(record_id, str) and record_id != ''
                and last_occurrence[record_id] != index):
            continue
        # Skip records whose 'active' is false.
        if record.get('active', True) is False:
            continue
        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Read 'delta', defaulting to 1 only when absent.
        number = record.get('delta', 1)
        if isinstance(number, bool):
            continue
        if isinstance(number, int):
            pass
        elif isinstance(number, str):
            stripped = number.strip()
            if _DELTA_PATTERN.fullmatch(stripped) is None:
                continue
            number = int(stripped, 10)
        else:
            continue
        groups[name] = groups.get(name, 0) + number

    return [{'sku': sku, 'total': groups[sku]} for sku in sorted(groups)]
