import re

def solve(value):
    last_idx_of_id = {}
    for i, record in enumerate(value):
        if isinstance(record, dict) and 'id' in record:
            val_id = record['id']
            if isinstance(val_id, str) and val_id != "":
                last_idx_of_id[val_id] = i

    retained = []
    for i, record in enumerate(value):
        if isinstance(record, dict) and 'id' in record:
            val_id = record['id']
            if isinstance(val_id, str) and val_id != "":
                if last_idx_of_id[val_id] == i:
                    retained.append(record)
                continue
        retained.append(record)

    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        if 'category' in record:
            name = record['category']
        elif 'kind' in record:
            name = record['kind']
        else:
            continue

        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if 'priority' in record:
            priority_val = record['priority']
        else:
            priority_val = 1

        if type(priority_val) is bool:
            continue
        elif type(priority_val) is int:
            p = priority_val
        elif type(priority_val) is float:
            # Just to be sure we don't accept float if it matches int, 
            # though isinstance(..., int) doesn't catch floats. 
            # type(val) is int is safer.
            continue
        elif isinstance(priority_val, str):
            s = priority_val.strip()
            if re.fullmatch(r'[+-]?[0-9]+', s):
                p = int(s)
            else:
                continue
        else:
            continue

        if name not in groups:
            groups[name] = p
        else:
            if p > groups[name]:
                groups[name] = p

    result = []
    for category in sorted(groups.keys()):
        result.append({"category": category, "maximum": groups[category]})

    return result
