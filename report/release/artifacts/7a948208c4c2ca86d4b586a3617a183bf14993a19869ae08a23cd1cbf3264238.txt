import re


def solve(value):
    if not isinstance(value, list):
        return []

    # Retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent.
    # Deduplicate before all other rules.
    last_indices = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                last_indices[rec_id] = i

    deduped = []
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                if last_indices[rec_id] != i:
                    continue
        deduped.append(record)

    groups = {}
    for record in deduped:
        # Skip non-object records and records whose 'active' is false.
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Read 'category', falling back to 'kind' only when 'category' is absent.
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if 'category' in record:
            raw_cat = record['category']
        elif 'kind' in record:
            raw_cat = record['kind']
        else:
            continue

        if not isinstance(raw_cat, str):
            continue
        cat = raw_cat.strip()
        if not cat:
            continue

        # Read 'priority', defaulting to 1 only when absent.
        # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace;
        # convert strings to base-10 integers. Skip all other values, including null.
        if 'priority' not in record:
            p = 1
        else:
            raw_p = record['priority']
            if isinstance(raw_p, bool):
                continue
            elif isinstance(raw_p, int):
                p = raw_p
            elif isinstance(raw_p, str):
                p_stripped = raw_p.strip()
                if re.fullmatch(r'[+-]?[0-9]+', p_stripped):
                    p = int(p_stripped, 10)
                else:
                    continue
            else:
                continue

        if cat not in groups:
            groups[cat] = p
        else:
            if p > groups[cat]:
                groups[cat] = p

    return [
        {'category': cat, 'maximum': groups[cat]}
        for cat in sorted(groups.keys())
    ]
