"""Inventory grouping per PROBLEM.md."""

import re

_INT_TEXT = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplication comes first: only the last occurrence of each nonempty
    # string id survives. Records without such an id stay independent.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    totals = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        identifier = record.get('id')
        if isinstance(identifier, str) and identifier:
            if latest[identifier] != position:
                continue

        if record.get('active') is False:
            continue

        # 'sku' wins; 'item' is used only when 'sku' is entirely absent.
        name = record.get('sku') if 'sku' in record else record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # 'delta' defaults to 1 only when absent; everything unrecognised is skipped.
        raw = record['delta'] if 'delta' in record else 1
        if isinstance(raw, bool):
            continue
        if isinstance(raw, int):
            amount = raw
        elif isinstance(raw, str):
            text = raw.strip()
            if not _INT_TEXT.fullmatch(text):
                continue
            amount = int(text, 10)
        else:
            continue

        totals[name] = totals.get(name, 0) + amount

    return [{'sku': name, 'total': totals[name]} for name in sorted(totals)]
