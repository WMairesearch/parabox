import re

_DELTA_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1 (before all others): retain only the last occurrence of each
    # nonempty string 'id'; records without such an id stay independent.
    last = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last[rid] = i

    groups = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '' and last[rid] != i:
                continue  # superseded by a later record with the same id
        else:
            continue  # skip non-object records
        if record.get('active') is False:
            continue
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'delta' in record:
            number = record['delta']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str):
                s = number.strip()
                if not _DELTA_RE.fullmatch(s):
                    continue
                number = int(s, 10)
            else:
                continue
        else:
            number = 1
        groups[name] = groups.get(name, 0) + number

    return [{'sku': key, 'total': groups[key]} for key in sorted(groups)]
