import re

def solve(value):
    # Deduplicate before all other rules
    last_seen = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val:
                last_seen[id_val] = i

    retained_indices = set()
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val:
                if last_seen[id_val] == i:
                    retained_indices.add(i)
            else:
                retained_indices.add(i)
        else:
            retained_indices.add(i)

    groups = {}
    for i in sorted(retained_indices):
        record = value[i]
        
        # Skip non-object records
        if not isinstance(record, dict):
            continue
            
        # Skip records whose 'active' is false
        if record.get('active') is False:
            continue
            
        # Read 'path', falling back to 'route' only when 'path' is absent
        if 'path' in record:
            key = record['path']
        elif 'route' in record:
            key = record['route']
        else:
            continue
            
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace
        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue
            
        # Read 'bytes', defaulting to 1 only when absent
        if 'bytes' not in record:
            num = 1
        else:
            val = record['bytes']
            if isinstance(val, bool):
                continue
            elif isinstance(val, int):
                num = val
            elif isinstance(val, str):
                s = val.strip()
                if re.fullmatch(r'[+-]?[0-9]+', s):
                    num = int(s)
                else:
                    continue
            else:
                continue
                
        # Group retained records
        total, count = groups.get(key, (0, 0))
        groups[key] = (total + num, count + 1)
        
    result = []
    for key in sorted(groups.keys()):
        result.append({'path': key, 'total': groups[key][0], 'count': groups[key][1]})
    return result
