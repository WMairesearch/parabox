import re

def solve(value):
    last_occurrences = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val != '':
                last_occurrences[id_val] = i
                
    filtered_value = []
    for i, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val != '':
                if last_occurrences[id_val] != i:
                    continue
        filtered_value.append(record)
        
    groups = {}
    for record in filtered_value:
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        if 'category' in record:
            key_val = record['category']
        elif 'kind' in record:
            key_val = record['kind']
        else:
            continue
            
        if not isinstance(key_val, str):
            continue
            
        key_val = key_val.strip()
        if not key_val:
            continue
            
        if 'priority' not in record:
            priority = 1
        else:
            p_val = record['priority']
            if type(p_val) is int:
                priority = p_val
            elif isinstance(p_val, str):
                p_val_stripped = p_val.strip()
                if re.match(r'^[+-]?[0-9]+$', p_val_stripped):
                    priority = int(p_val_stripped)
                else:
                    continue 
            else:
                continue 
                
        if key_val in groups:
            groups[key_val] = max(groups[key_val], priority)
        else:
            groups[key_val] = priority
            
    result = []
    for name in sorted(groups.keys()):
        result.append({'category': name, 'maximum': groups[name]})
    return result
