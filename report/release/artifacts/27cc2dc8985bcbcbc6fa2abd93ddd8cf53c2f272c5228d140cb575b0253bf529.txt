import re

_DELTA_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1 (first): retain only the last occurrence of each nonempty string 'id'.
    last_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_position[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        # Deduplication: drop superseded occurrences of an id.
        record_id = record.get('id')
        if isinstance(record_id, str) and record_id != '':
            if last_position[record_id] != position:
                continue
        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue
        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str) or not name.strip():
            continue
        name = name.strip()
        # Read 'delta', defaulting to 1 only when absent.
        if 'delta' in record:
            number = record['delta']
            if isinstance(number, bool):
                continue
            if isinstance(number, int):
                pass
            elif isinstance(number, str) and _DELTA_RE.fullmatch(number.strip()):
                number = int(number.strip(), 10)
            else:
                continue
        else:
            number = 1
        groups[name] = groups.get(name, 0) + number

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
