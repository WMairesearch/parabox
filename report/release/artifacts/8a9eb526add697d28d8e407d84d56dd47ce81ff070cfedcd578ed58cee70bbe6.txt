import re


def solve(value):
    if not isinstance(value, list):
        return []

    # First retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Deduplicate before all other rules.
    last_indices = {}
    for idx, rec in enumerate(value):
        if isinstance(rec, dict):
            rec_id = rec.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                last_indices[rec_id] = idx

    groups = {}
    for idx, rec in enumerate(value):
        # Non-object records do not have an id and are skipped
        if not isinstance(rec, dict):
            continue

        rec_id = rec.get('id')
        if isinstance(rec_id, str) and rec_id != '':
            if last_indices[rec_id] != idx:
                continue

        # Skip records whose 'active' is false
        if rec.get('active') is False:
            continue

        # Read 'sku', falling back to 'item' only when 'sku' is absent
        if 'sku' in rec:
            raw_key = rec['sku']
        elif 'item' in rec:
            raw_key = rec['item']
        else:
            continue

        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace
        if not isinstance(raw_key, str):
            continue
        key = raw_key.strip()
        if not key:
            continue

        # Read 'delta', defaulting to 1 only when absent
        if 'delta' not in rec:
            delta = 1
        else:
            raw_delta = rec['delta']
            if type(raw_delta) is int:
                delta = raw_delta
            elif isinstance(raw_delta, str):
                stripped = raw_delta.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    delta = int(stripped, 10)
                else:
                    continue
            else:
                continue

        groups[key] = groups.get(key, 0) + delta

    return [{'sku': k, 'total': groups[k]} for k in sorted(groups.keys())]
