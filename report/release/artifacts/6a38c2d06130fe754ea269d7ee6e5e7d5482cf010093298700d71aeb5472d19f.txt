import re

_PRIORITY_STRING = re.compile(r'[+-]?[0-9]+')


def _identity(record):
    """Return a hashable dedupe key for the record's 'id', or None."""
    identifier = record.get('id')
    if isinstance(identifier, str) and identifier != '':
        return ('id', identifier)
    return None


def _category(record):
    if 'category' in record:
        name = record['category']
    elif 'kind' in record:
        name = record['kind']
    else:
        return None
    if not isinstance(name, str):
        return None
    name = name.strip()
    if not name:
        return None
    return name


def _priority(record):
    if 'priority' not in record:
        return 1
    value = record['priority']
    if isinstance(value, bool):
        return None
    if isinstance(value, int):
        return value
    if isinstance(value, str):
        text = value.strip()
        if _PRIORITY_STRING.fullmatch(text):
            return int(text, 10)
    return None


def solve(value):
    if not isinstance(value, list):
        return []

    records = [record for record in value if isinstance(record, dict)]

    # Deduplicate before all other rules: keep the last occurrence of each
    # nonempty string 'id'; records without such an id stay independent.
    last_position = {}
    for position, record in enumerate(records):
        key = _identity(record)
        if key is not None:
            last_position[key] = position

    retained = []
    for position, record in enumerate(records):
        key = _identity(record)
        if key is not None and last_position[key] != position:
            continue
        retained.append(record)

    groups = {}
    for record in retained:
        if record.get('active') is False:
            continue
        name = _category(record)
        if name is None:
            continue
        number = _priority(record)
        if number is None:
            continue
        previous = groups.get(name)
        if previous is None or number > previous:
            groups[name] = number

    return [
        {'category': name, 'maximum': groups[name]}
        for name in sorted(groups)
    ]
