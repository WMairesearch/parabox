import re

_BYTES_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1 (before all others): retain only the last occurrence of each
    # nonempty string 'id'; records without such an id stay independent.
    last_index_for_id = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id:
                last_index_for_id[record_id] = index

    groups = {}
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        record_id = record.get('id')
        if isinstance(record_id, str) and record_id:
            if last_index_for_id[record_id] != index:
                continue  # superseded by a later record with the same id
        if record.get('active') is False:
            continue
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        number = record.get('bytes', 1)
        if isinstance(number, bool):
            continue
        if isinstance(number, int):
            amount = number
        elif isinstance(number, str):
            text = number.strip()
            if not _BYTES_PATTERN.fullmatch(text):
                continue
            amount = int(text, 10)
        else:
            continue
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + amount, count + 1)

    return [
        {'path': name, 'total': groups[name][0], 'count': groups[name][1]}
        for name in sorted(groups)
    ]
