import re

def solve(value):
    # Pass 1: find the last occurrence of each nonempty string 'id'
    last_occurrence = {}
    for index, record in enumerate(value):
        if isinstance(record, dict) and 'id' in record:
            record_id = record['id']
            if type(record_id) is str and record_id != "":
                last_occurrence[record_id] = index

    # Pass 2: filter and process
    groups = {}
    for index, record in enumerate(value):
        # Deduplication
        if isinstance(record, dict) and 'id' in record:
            record_id = record['id']
            if type(record_id) is str and record_id != "":
                if last_occurrence[record_id] != index:
                    continue
        
        # Skip non-object records
        if not isinstance(record, dict):
            continue
            
        # Skip records whose 'active' is false
        if record.get('active') is False:
            continue
            
        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            sku_val = record['sku']
        elif 'item' in record:
            sku_val = record['item']
        else:
            continue
            
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace
        if type(sku_val) is not str:
            continue
        
        sku = sku_val.strip()
        if not sku:
            continue
            
        # Read 'delta', defaulting to 1 only when absent
        if 'delta' not in record:
            delta = 1
        else:
            delta_val = record['delta']
            # Accept integers (not booleans)
            if type(delta_val) is int:
                delta = delta_val
            # and strings matching [+-]?[0-9]+ after stripping whitespace
            elif type(delta_val) is str:
                delta_str = delta_val.strip()
                if re.fullmatch(r'[+-]?[0-9]+', delta_str):
                    delta = int(delta_str)
                else:
                    continue
            else:
                # Skip all other values, including null.
                continue
                
        # Group retained records by normalized 'sku'
        groups[sku] = groups.get(sku, 0) + delta
        
    # Each output object contains 'sku' and the sum of deltas as 'total'
    # Sort results lexicographically by 'sku'
    result = []
    for sku in sorted(groups.keys()):
        result.append({'sku': sku, 'total': groups[sku]})
        
    return result
