"""Inventory grouping per PROBLEM.md specification."""

import re

_INT_RE = re.compile(r'[+-]?[0-9]+\Z')


def _valid_id(record):
    """Return the raw 'id' if it is a nonempty string, else None."""
    ident = record.get('id')
    if isinstance(ident, str) and ident != '':
        return ident
    return None


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1 (before everything else): keep only the last occurrence of each
    # nonempty string id.  Records without such an id stay independent.
    last_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            ident = _valid_id(record)
            if ident is not None:
                last_position[ident] = position

    kept = []
    for position, record in enumerate(value):
        if isinstance(record, dict):
            ident = _valid_id(record)
            if ident is not None and last_position[ident] != position:
                continue
        kept.append(record)

    groups = {}
    for record in kept:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # 'sku' with fallback to 'item' only when 'sku' is absent.
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # 'delta' defaults to 1 only when absent.
        if 'delta' in record:
            delta = record['delta']
            if isinstance(delta, bool):
                continue
            if isinstance(delta, int):
                amount = delta
            elif isinstance(delta, str):
                text = delta.strip()
                if _INT_RE.match(text) is None:
                    continue
                amount = int(text, 10)
            else:
                continue
        else:
            amount = 1

        groups[name] = groups.get(name, 0) + amount

    return [{'sku': key, 'total': groups[key]} for key in sorted(groups)]
