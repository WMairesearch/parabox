"""Group event records by normalized category.

Rules (see PROBLEM.md):
  * keep only the last occurrence of each nonempty string ``id``
    (deduplication happens before every other rule);
  * skip non-object records and records whose ``active`` is ``False``;
  * the grouping key is ``category``, falling back to ``kind`` only when
    ``category`` is absent; non-string / blank-after-stripping keys are skipped;
  * ``priority`` defaults to 1 only when absent, otherwise it must be an
    integer (not a bool) or a string matching ``[+-]?[0-9]+`` after
    stripping; every other value skips the record;
  * each group reports the maximum priority as ``maximum`` and the results
    are sorted lexicographically by ``category``.
"""

import re

_INTEGER = re.compile(r"[+-]?[0-9]+")


def _identity(record):
    """Return the deduplication key of a record, or None when it is independent."""
    key = record.get("id")
    if isinstance(key, str) and key:
        return key
    return None


def _normalized_category(record):
    """Return the stripped grouping key, or None when the record is unusable."""
    raw = record["category"] if "category" in record else record.get("kind")
    if not isinstance(raw, str):
        return None
    key = raw.strip()
    return key or None


def _normalized_priority(record):
    """Return the integer priority, or None when the record must be skipped."""
    if "priority" not in record:
        return 1
    raw = record["priority"]
    if isinstance(raw, bool):
        return None
    if isinstance(raw, int):
        return raw
    if isinstance(raw, str):
        text = raw.strip()
        if _INTEGER.fullmatch(text) is not None:
            return int(text, 10)
    return None


def solve(value):
    if isinstance(value, (list, tuple)):
        records = list(value)
    else:
        records = []

    # Last occurrence of every nonempty string id wins; everything else stays.
    survivors = {}
    for position, record in enumerate(records):
        if isinstance(record, dict):
            key = _identity(record)
            if key is not None:
                survivors[key] = position

    groups = {}
    for position, record in enumerate(records):
        if not isinstance(record, dict):
            continue
        key = _identity(record)
        if key is not None and survivors[key] != position:
            continue
        if record.get("active") is False:
            continue
        category = _normalized_category(record)
        if category is None:
            continue
        number = _normalized_priority(record)
        if number is None:
            continue
        if category not in groups or number > groups[category]:
            groups[category] = number

    return [
        {"category": category, "maximum": maximum}
        for category, maximum in sorted(groups.items())
    ]
