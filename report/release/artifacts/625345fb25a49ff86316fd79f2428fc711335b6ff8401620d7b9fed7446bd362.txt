import re

_PRIORITY_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    records = value if isinstance(value, list) else []

    # Rule 1 (before all others): retain only the last occurrence of each
    # nonempty string 'id'; records without such an id remain independent.
    last_occurrence = {}
    for position, record in enumerate(records):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_occurrence[record_id] = position

    groups = {}
    for position, record in enumerate(records):
        # Skip non-object records.
        if not isinstance(record, dict):
            continue
        # Drop superseded duplicates.
        record_id = record.get('id')
        if (isinstance(record_id, str) and record_id != ''
                and last_occurrence[record_id] != position):
            continue
        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue
        # Read 'category', falling back to 'kind' only when 'category' is absent.
        if 'category' in record:
            name = record['category']
        elif 'kind' in record:
            name = record['kind']
        else:
            continue
        # Skip non-string or empty-after-stripping keys; strip whitespace.
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Read 'priority', defaulting to 1 only when absent.
        if 'priority' in record:
            number = record['priority']
            if type(number) is int:  # accepts integers, excludes booleans
                pass
            elif isinstance(number, str):
                text = number.strip()
                if _PRIORITY_PATTERN.fullmatch(text) is None:
                    continue
                number = int(text, 10)
            else:
                continue  # skip all other values, including null
        else:
            number = 1
        # Group by normalized category, keeping the maximum priority.
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    # Sort results lexicographically by 'category'.
    return [{'category': name, 'maximum': groups[name]}
            for name in sorted(groups)]
