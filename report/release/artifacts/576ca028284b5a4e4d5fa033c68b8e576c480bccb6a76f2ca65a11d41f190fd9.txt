import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []
    records = list(value)

    # Deduplicate first: keep only the last record for each nonempty string id.
    last_index = {}
    for position, record in enumerate(records):
        if not isinstance(record, dict):
            continue
        ident = record.get('id')
        if isinstance(ident, str) and ident != '':
            last_index[ident] = position

    retained = []
    for position, record in enumerate(records):
        if isinstance(record, dict):
            ident = record.get('id')
            if isinstance(ident, str) and ident != '' and last_index.get(ident) != position:
                continue
        retained.append(record)

    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if 'delta' in record:
            raw = record['delta']
        else:
            raw = 1
        if isinstance(raw, bool):
            continue
        if isinstance(raw, int):
            number = int(raw)
        elif isinstance(raw, str):
            text = raw.strip()
            if not _INT_RE.fullmatch(text):
                continue
            number = int(text, 10)
        else:
            continue

        groups[name] = groups.get(name, 0) + number

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
