import re


_DELTA_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Input must be a JSON list.
    if not isinstance(value, list):
        return []

    # Step 1: deduplicate before all other rules.
    # Retain only the last occurrence of each nonempty string 'id'.
    # Records without such an id remain independent.
    last_positions = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_positions[record_id] = position

    records = []
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if last_positions.get(record_id) != position:
                    continue
        records.append(record)

    # Step 2: apply filtering and grouping.
    groups = {}
    for record in records:
        if not isinstance(record, dict):
            continue

        if record.get('active') is False:
            continue

        if 'sku' in record:
            sku = record['sku']
        else:
            sku = record.get('item')

        if not isinstance(sku, str):
            continue
        sku = sku.strip()
        if sku == '':
            continue

        if 'delta' not in record:
            delta = 1
        else:
            delta = record['delta']
            if type(delta) is int:
                delta = delta
            elif isinstance(delta, str):
                delta = delta.strip()
                if _DELTA_RE.fullmatch(delta):
                    delta = int(delta)
                else:
                    continue
            else:
                continue

        groups[sku] = groups.get(sku, 0) + delta

    return [{'sku': sku, 'total': groups[sku]} for sku in sorted(groups)]
