import re


_DELTA_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Retain only the last occurrence of each nonempty string id. Records
    # without such an id remain independent. This happens before every other
    # filtering rule.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        identifier = record.get('id')
        if isinstance(identifier, str) and identifier:
            if latest[identifier] != position:
                continue

        if record.get('active') is False:
            continue

        name = record['sku'] if 'sku' in record else record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if 'delta' in record:
            number = record['delta']
        else:
            number = 1
        if isinstance(number, str):
            candidate = number.strip()
            if not _DELTA_RE.fullmatch(candidate):
                continue
            number = int(candidate, 10)
        if type(number) is not int:
            continue

        groups[name] = groups.get(name, 0) + number

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
