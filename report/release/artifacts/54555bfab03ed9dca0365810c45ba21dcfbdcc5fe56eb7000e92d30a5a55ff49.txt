import re

def solve(value):
    # 1. Deduplication
    last_occurrence = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            if 'id' in record:
                rec_id = record['id']
                if isinstance(rec_id, str) and rec_id != '':
                    last_occurrence[rec_id] = i
                    
    retained = []
    for i, record in enumerate(value):
        if isinstance(record, dict) and 'id' in record:
            rec_id = record['id']
            if isinstance(rec_id, str) and rec_id != '':
                if last_occurrence[rec_id] != i:
                    continue
        retained.append(record)

    # 2. Process records
    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue
        
        if record.get('active') is False:
            continue
            
        if 'category' in record:
            name = record['category']
        elif 'kind' in record:
            name = record['kind']
        else:
            continue
            
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
            
        if 'priority' not in record:
            priority = 1
        else:
            priority = record['priority']
            
        if type(priority) is int:
            pass
        elif isinstance(priority, str):
            priority_str = priority.strip()
            if re.fullmatch(r'[+-]?[0-9]+', priority_str):
                priority = int(priority_str)
            else:
                continue
        else:
            continue
            
        if name in groups:
            if priority > groups[name]:
                groups[name] = priority
        else:
            groups[name] = priority
            
    # 3. Sort and construct output
    result = []
    for name in sorted(groups.keys()):
        result.append({'category': name, 'maximum': groups[name]})
        
    return result
