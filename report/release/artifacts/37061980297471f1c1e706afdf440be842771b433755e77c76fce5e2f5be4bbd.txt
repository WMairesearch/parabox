import re

def solve(value):
    if not isinstance(value, list):
        return []

    # 1. Deduplicate by last occurrence of nonempty string 'id'
    last_indices = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_indices[record_id] = i

    groups = {}
    
    for i, record in enumerate(value):
        # 1. Deduplicate continuation
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if last_indices[record_id] != i:
                    continue
        
        # 2. Skip non-object records and records whose 'active' is false.
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
            
        # 3. Read 'sku', falling back to 'item' only when 'sku' is absent.
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if 'sku' in record:
            raw_sku = record['sku']
        elif 'item' in record:
            raw_sku = record['item']
        else:
            continue
            
        if not isinstance(raw_sku, str):
            continue
        
        sku = raw_sku.strip()
        if not sku:
            continue
            
        # 4. Read 'delta', defaulting to 1 only when absent.
        if 'delta' not in record:
            delta = 1
        else:
            delta_raw = record['delta']
            if type(delta_raw) is int:
                delta = delta_raw
            elif isinstance(delta_raw, str):
                delta_str = delta_raw.strip()
                if re.fullmatch(r'[+-]?[0-9]+', delta_str):
                    delta = int(delta_str)
                else:
                    continue
            else:
                continue
                
        # 5. Group by sku
        groups[sku] = groups.get(sku, 0) + delta

    # 6. Sort and format results
    result = []
    for sku in sorted(groups.keys()):
        result.append({'sku': sku, 'total': groups[sku]})
        
    return result
