import re

_INT_RE = re.compile(r'^[+-]?[0-9]+$')


def solve(value):
    if not isinstance(value, list):
        return []

    # Determine the last occurrence index for each nonempty string id.
    last_index = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        record_id = record.get('id')
        if isinstance(record_id, str) and record_id != '':
            last_index[record_id] = position

    groups = {}
    order = []
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        record_id = record.get('id')
        if isinstance(record_id, str) and record_id != '':
            if last_index.get(record_id) != position:
                continue

        if record.get('active') is False:
            continue

        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if 'bytes' in record:
            number = record['bytes']
            if type(number) is int:
                amount = number
            elif isinstance(number, str):
                text = number.strip()
                if not _INT_RE.match(text):
                    continue
                amount = int(text, 10)
            else:
                continue
        else:
            amount = 1

        if name in groups:
            groups[name][0] += amount
            groups[name][1] += 1
        else:
            groups[name] = [amount, 1]
            order.append(name)

    result = []
    for name in sorted(groups):
        total, count = groups[name]
        result.append({'path': name, 'total': total, 'count': count})
    return result
