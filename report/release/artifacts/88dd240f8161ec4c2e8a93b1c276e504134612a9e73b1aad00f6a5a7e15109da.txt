import re

_INT_RE = re.compile(r'[+-]?[0-9]+\Z')


def _priority(record):
    """Return the record's priority, or None when it must be skipped."""
    if 'priority' not in record:
        return 1
    value = record['priority']
    if isinstance(value, bool):
        return None
    if isinstance(value, int):
        return value
    if isinstance(value, str):
        text = value.strip()
        if _INT_RE.match(text):
            return int(text, 10)
    return None


def _category(record):
    """Return the normalized category, or None when it must be skipped."""
    if 'category' in record:
        value = record['category']
    elif 'kind' in record:
        value = record['kind']
    else:
        return None
    if not isinstance(value, str):
        return None
    value = value.strip()
    return value or None


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: keep only the last occurrence of each nonempty
    # string 'id'; records without such an id stay independent.
    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            ident = record.get('id')
            if isinstance(ident, str) and ident:
                last_index[ident] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        ident = record.get('id')
        if isinstance(ident, str) and ident and last_index[ident] != position:
            continue
        if record.get('active') is False:
            continue
        name = _category(record)
        if name is None:
            continue
        number = _priority(record)
        if number is None:
            continue
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [
        {'category': name, 'maximum': groups[name]}
        for name in sorted(groups)
    ]
