import re

def solve(value):
    # Pass 1: find last occurrence of each non-empty string 'id'
    last_id_idx = {}
    for i, record in enumerate(value):
        if isinstance(record, dict) and 'id' in record:
            record_id = record['id']
            if isinstance(record_id, str) and record_id != '':
                last_id_idx[record_id] = i

    groups = {}
    
    # Pass 2: filter and process
    for i, record in enumerate(value):
        # 1. Deduplicate
        if isinstance(record, dict) and 'id' in record:
            record_id = record['id']
            if isinstance(record_id, str) and record_id != '':
                if last_id_idx[record_id] != i:
                    continue
        
        # 2. Skip non-object records
        if not isinstance(record, dict):
            continue
            
        # 3. Skip records whose 'active' is false
        if record.get('active') is False:
            continue
            
        # 4. Read 'sku', falling back to 'item' only when 'sku' is absent
        if 'sku' in record:
            raw_sku = record['sku']
        elif 'item' in record:
            raw_sku = record['item']
        else:
            continue
            
        # 5. Skip non-string or empty-after-stripping keys; strip surrounding whitespace
        if not isinstance(raw_sku, str):
            continue
        sku = raw_sku.strip()
        if not sku:
            continue
            
        # 6. Read 'delta', defaulting to 1 only when absent
        if 'delta' not in record:
            delta = 1
        else:
            raw_delta = record['delta']
            if type(raw_delta) is int:
                delta = raw_delta
            elif isinstance(raw_delta, str):
                stripped_delta = raw_delta.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped_delta):
                    delta = int(stripped_delta)
                else:
                    continue
            else:
                continue
                
        # 7. Group
        groups[sku] = groups.get(sku, 0) + delta
        
    result = []
    for sku in sorted(groups.keys()):
        result.append({'sku': sku, 'total': groups[sku]})
        
    return result
