"""Inventory rollup: group deltas by normalized sku.

Spec (PROBLEM.md):
  * Deduplicate first: keep only the last occurrence of each nonempty string
    'id'; records without such an id stay independent.
  * Skip non-object records and records whose 'active' is false.
  * Key = 'sku', falling back to 'item' only when 'sku' is absent; skip
    non-string / blank-after-strip keys; strip surrounding whitespace.
  * 'delta' defaults to 1 only when absent; accept ints (not bools) and
    strings matching [+-]?[0-9]+ after stripping; skip everything else.
  * Group by normalized sku, sum deltas into 'total', sort by 'sku'.
  * Never mutate the input.
"""

import re

_INTEGER = re.compile(r'[+-]?[0-9]+')
_MISSING = object()


def _last_positions(value):
    """Map each nonempty string id to the index of its last occurrence."""
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position
    return latest


def _coerce_delta(raw):
    """Return the integer delta, or None when the value is not acceptable."""
    if isinstance(raw, str):
        text = raw.strip()
        if _INTEGER.fullmatch(text):
            return int(text, 10)
        return None
    if isinstance(raw, bool):
        return None
    if isinstance(raw, int):
        return raw
    return None


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []
    latest = _last_positions(value)
    totals = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier:
            if latest.get(identifier) != position:
                continue
        if record.get('active') is False:
            continue
        name = record['sku'] if 'sku' in record else record.get('item', _MISSING)
        if name is _MISSING or not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        delta = record.get('delta', _MISSING)
        delta = 1 if delta is _MISSING else _coerce_delta(delta)
        if delta is None:
            continue
        totals[name] = totals.get(name, 0) + delta
    return [{'sku': sku, 'total': totals[sku]} for sku in sorted(totals)]
