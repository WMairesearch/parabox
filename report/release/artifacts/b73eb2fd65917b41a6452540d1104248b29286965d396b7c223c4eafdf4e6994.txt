import re

def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate by id
    last_occurrence = {}
    for idx, record in enumerate(value):
        if isinstance(record, dict) and 'id' in record:
            rid = record['id']
            if isinstance(rid, str) and rid:
                last_occurrence[rid] = idx

    retained = []
    for idx, record in enumerate(value):
        if isinstance(record, dict) and 'id' in record:
            rid = record['id']
            if isinstance(rid, str) and rid:
                if last_occurrence[rid] != idx:
                    continue
        retained.append(record)

    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        if 'sku' in record:
            raw_key = record['sku']
        elif 'item' in record:
            raw_key = record['item']
        else:
            continue
            
        if not isinstance(raw_key, str):
            continue
            
        key = raw_key.strip()
        if not key:
            continue
            
        if 'delta' not in record:
            delta_val = 1
        else:
            raw_delta = record['delta']
            if type(raw_delta) is bool:
                continue
            elif type(raw_delta) is int:
                delta_val = raw_delta
            elif type(raw_delta) is str:
                stripped = raw_delta.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    delta_val = int(stripped, 10)
                else:
                    continue
            else:
                continue
                
        groups[key] = groups.get(key, 0) + delta_val

    result = []
    for key in sorted(groups.keys()):
        result.append({'sku': key, 'total': groups[key]})
        
    return result
