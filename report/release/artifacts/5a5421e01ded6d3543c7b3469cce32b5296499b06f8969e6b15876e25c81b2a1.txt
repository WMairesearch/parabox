import re

def solve(value):
    seen_ids = set()
    retained_records = []
    for record in reversed(value):
        if isinstance(record, dict) and 'id' in record:
            rec_id = record['id']
            if isinstance(rec_id, str) and rec_id != '':
                if rec_id in seen_ids:
                    continue
                seen_ids.add(rec_id)
        retained_records.append(record)
    
    retained_records.reverse()
    
    groups = {}
    for record in retained_records:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
            
        if 'category' in record:
            name = record['category']
        elif 'kind' in record:
            name = record['kind']
        else:
            continue
            
        if not isinstance(name, str) or not name.strip():
            continue
        name = name.strip()
        
        if 'priority' not in record:
            number = 1
        else:
            number = record['priority']
            
        if type(number) is int:
            pass
        elif isinstance(number, str) and re.fullmatch(r'[+-]?[0-9]+', number.strip()):
            number = int(number.strip())
        else:
            continue
            
        if name not in groups or number > groups[name]:
            groups[name] = number
            
    result = []
    for name in sorted(groups.keys()):
        result.append({'category': name, 'maximum': groups[name]})
    return result
