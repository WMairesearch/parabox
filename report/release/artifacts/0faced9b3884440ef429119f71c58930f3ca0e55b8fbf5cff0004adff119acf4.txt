import re


_DELTA_RE = re.compile(r'[+-]?[0-9]+\Z')


def _parse_delta(record):
    """Return the valid delta for record, or None if the record must be skipped."""
    if 'delta' not in record:
        return 1
    value = record['delta']
    if type(value) is int:
        return value
    if isinstance(value, str):
        stripped = value.strip()
        if _DELTA_RE.match(stripped):
            return int(stripped)
    return None


def solve(value):
    # Deduplicate first: for each nonempty string id, only the last record
    # carrying that id is kept.  Records without such an id all remain.
    last_positions = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_positions[record_id] = position

    groups = {}

    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        # Deduplication is applied before every other rule below.
        record_id = record.get('id')
        if isinstance(record_id, str) and record_id != '':
            if last_positions.get(record_id) != position:
                continue

        # Active false skips the record.
        if record.get('active') is False:
            continue

        # sku, falling back to item only when sku is absent.
        if 'sku' in record:
            raw_sku = record['sku']
        else:
            raw_sku = record.get('item')
        if not isinstance(raw_sku, str):
            continue
        sku = raw_sku.strip()
        if sku == '':
            continue

        delta = _parse_delta(record)
        if delta is None:
            continue

        groups[sku] = groups.get(sku, 0) + delta

    return [{'sku': sku, 'total': groups[sku]} for sku in sorted(groups)]
