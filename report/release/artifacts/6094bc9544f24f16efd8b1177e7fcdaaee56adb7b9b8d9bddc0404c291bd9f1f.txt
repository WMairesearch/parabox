import re


_BYTES_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # First, keep only the last occurrence of each nonempty string id.
    # Records without such an id (including non-dicts) remain independent.
    last_positions = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_positions[record_id] = position

    deduplicated = []
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if last_positions[record_id] != position:
                    continue
        deduplicated.append(record)

    groups = {}
    for record in deduplicated:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        if 'path' in record:
            path = record.get('path')
        else:
            path = record.get('route')

        if not isinstance(path, str) or not path.strip():
            continue
        path = path.strip()

        if 'bytes' not in record:
            number = 1
        else:
            raw_bytes = record['bytes']
            if type(raw_bytes) is int:
                number = raw_bytes
            elif isinstance(raw_bytes, str):
                stripped = raw_bytes.strip()
                if _BYTES_RE.fullmatch(stripped):
                    number = int(stripped)
                else:
                    continue
            else:
                continue

        total, count = groups.get(path, (0, 0))
        groups[path] = (total + number, count + 1)

    result = [
        {'path': path, 'total': total, 'count': count}
        for path, (total, count) in sorted(groups.items())
    ]
    return result
