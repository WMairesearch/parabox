import re


BYTES_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Deduplicate first: keep only the last occurrence of each nonempty
    # string 'id'.  Records without such an id remain independent.
    last_positions = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_positions[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if last_positions.get(record_id) != position:
                    continue

        # Skip non-object records and records whose 'active' is false.
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            raw_path = record['path']
        else:
            raw_path = record.get('route')

        if not isinstance(raw_path, str):
            continue
        path = raw_path.strip()
        if not path:
            continue

        # Read 'bytes', defaulting to 1 only when absent.
        if 'bytes' in record:
            raw_bytes = record['bytes']
            if type(raw_bytes) is int:
                number = raw_bytes
            elif isinstance(raw_bytes, str):
                raw_bytes = raw_bytes.strip()
                if BYTES_PATTERN.fullmatch(raw_bytes):
                    number = int(raw_bytes, 10)
                else:
                    continue
            else:
                continue
        else:
            number = 1

        total, count = groups.get(path, (0, 0))
        groups[path] = (total + number, count + 1)

    result = [
        {'path': path, 'total': groups[path][0], 'count': groups[path][1]}
        for path in sorted(groups)
    ]
    return result
