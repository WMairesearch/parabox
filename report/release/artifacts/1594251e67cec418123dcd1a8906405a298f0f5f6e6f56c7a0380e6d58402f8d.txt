import re

def solve(value):
    last_occurrence = {}
    for idx, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val != "":
                last_occurrence[id_val] = idx
                
    groups = {}
    for idx, record in enumerate(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val != "":
                if last_occurrence[id_val] != idx:
                    continue
                    
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        if 'sku' in record:
            raw_sku = record['sku']
        elif 'item' in record:
            raw_sku = record['item']
        else:
            continue
            
        if not isinstance(raw_sku, str):
            continue
        
        sku = raw_sku.strip()
        if not sku:
            continue
            
        if 'delta' in record:
            raw_delta = record['delta']
        else:
            raw_delta = 1
            
        delta = None
        if type(raw_delta) is int:
            delta = raw_delta
        elif isinstance(raw_delta, str):
            stripped_delta = raw_delta.strip()
            if re.fullmatch(r'[+-]?[0-9]+', stripped_delta):
                delta = int(stripped_delta)
        
        if delta is None:
            continue
            
        groups[sku] = groups.get(sku, 0) + delta
        
    result = []
    for sku in sorted(groups.keys()):
        result.append({"sku": sku, "total": groups[sku]})
        
    return result
