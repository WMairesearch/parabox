import re

_PRIORITY = re.compile(r'[+-]?[0-9]+')


def solve(value):
    try:
        records = list(value)
    except TypeError:
        return []

    latest = {}
    for position, record in enumerate(records):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(records):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != position:
            continue
        if record.get('active') is False:
            continue
        name = record.get('category', record.get('kind'))
        if not isinstance(name, str) or not name.strip():
            continue
        name = name.strip()
        number = record.get('priority', 1)
        if isinstance(number, str):
            text = number.strip()
            if _PRIORITY.fullmatch(text) is None:
                continue
            number = int(text, 10)
        if type(number) is not int:
            continue
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
