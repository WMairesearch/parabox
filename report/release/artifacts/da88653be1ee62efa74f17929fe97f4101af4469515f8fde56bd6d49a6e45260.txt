import re

def solve(value):
    if not isinstance(value, list):
        return []

    keep_indices = set()
    seen_ids = set()
    
    # Deduplicate before all other rules
    for i in range(len(value) - 1, -1, -1):
        record = value[i]
        is_valid_id = False
        if isinstance(record, dict) and 'id' in record:
            record_id = record['id']
            if isinstance(record_id, str) and len(record_id) > 0:
                is_valid_id = True
                if record_id not in seen_ids:
                    seen_ids.add(record_id)
                    keep_indices.add(i)
        
        if not is_valid_id:
            keep_indices.add(i)
            
    groups = {}
    for i, record in enumerate(value):
        if i not in keep_indices:
            continue
            
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        if 'sku' in record:
            key_val = record['sku']
        elif 'item' in record:
            key_val = record['item']
        else:
            continue
            
        if not isinstance(key_val, str):
            continue
            
        sku = key_val.strip()
        if not sku:
            continue
            
        if 'delta' not in record:
            delta = 1
        else:
            delta_val = record['delta']
            if type(delta_val) is int:
                delta = delta_val
            elif isinstance(delta_val, str):
                delta_str = delta_val.strip()
                if re.fullmatch(r'[+-]?[0-9]+', delta_str):
                    delta = int(delta_str)
                else:
                    continue
            else:
                continue
                
        groups[sku] = groups.get(sku, 0) + delta
        
    result = []
    for name in sorted(groups.keys()):
        result.append({'sku': name, 'total': groups[name]})
        
    return result
