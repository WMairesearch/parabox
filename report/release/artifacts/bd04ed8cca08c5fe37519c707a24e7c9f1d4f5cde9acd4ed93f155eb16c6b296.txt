import re

def solve(value):
    if not isinstance(value, list):
        return []
        
    seen_ids = set()
    retained = []
    
    # Deduplicate by id, retaining only the last occurrence of each nonempty string 'id'
    for record in reversed(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and len(id_val) > 0:
                if id_val in seen_ids:
                    continue
                seen_ids.add(id_val)
        retained.append(record)
    
    retained.reverse()
    
    groups = {}
    
    for record in retained:
        # Skip non-object records
        if not isinstance(record, dict):
            continue
            
        # Skip records whose 'active' is false
        if record.get('active') is False:
            continue
            
        # Read 'sku', falling back to 'item' only when 'sku' is absent
        if 'sku' in record:
            raw_sku = record['sku']
        elif 'item' in record:
            raw_sku = record['item']
        else:
            continue
            
        # Skip non-string or empty-after-stripping keys
        if not isinstance(raw_sku, str):
            continue
            
        sku = raw_sku.strip()
        if not sku:
            continue
            
        # Read 'delta', defaulting to 1 only when absent
        if 'delta' not in record:
            delta = 1
        else:
            raw_delta = record['delta']
            # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace
            if isinstance(raw_delta, bool):
                continue
            elif isinstance(raw_delta, int):
                delta = raw_delta
            elif isinstance(raw_delta, str):
                d_str = raw_delta.strip()
                if re.fullmatch(r'[+-]?[0-9]+', d_str):
                    delta = int(d_str)
                else:
                    continue
            else:
                continue
                
        # Group retained records by normalized 'sku'
        groups[sku] = groups.get(sku, 0) + delta
        
    result = []
    # Sort results lexicographically by 'sku'
    for sku in sorted(groups.keys()):
        result.append({'sku': sku, 'total': groups[sku]})
        
    return result
