"""access_logs: group retained records by normalized path.

Implements PROBLEM.md exactly:
  1. id de-duplication first (last occurrence of each nonempty string id wins;
     records without such an id are independent).
  2. skip non-objects and records whose 'active' is false.
  3. path key: 'path', falling back to 'route' only when 'path' is absent;
     must be a string, stripped, non-empty.
  4. 'bytes': default 1 only when absent; ints (never bools) or strings
     matching [+-]?[0-9]+ after stripping; every other value skips the record.
  5. group by normalized path -> {'path', 'total' (sum), 'count'} sorted by path.
The input is never modified.
"""

import re

_INTEGER_TEXT = re.compile(r'[+-]?[0-9]+')


def _dedupe_id(record):
    """The de-duplication key of a record, or None when it has no usable id."""
    if not isinstance(record, dict):
        return None
    key = record.get('id')
    if isinstance(key, str) and key != '':
        return key
    return None


def _deduplicate(records):
    """Retain only the last occurrence of each nonempty string id."""
    last_position = {}
    for position, record in enumerate(records):
        key = _dedupe_id(record)
        if key is not None:
            last_position[key] = position
    kept = []
    for position, record in enumerate(records):
        key = _dedupe_id(record)
        if key is None or last_position[key] == position:
            kept.append(record)
    return kept


def _normalized_path(record):
    """Stripped non-empty path, or None when the record must be skipped."""
    raw = record['path'] if 'path' in record else record.get('route')
    if not isinstance(raw, str):
        return None
    text = raw.strip()
    if text == '':
        return None
    return text


def _byte_count(record):
    """Integer byte count, or None when the record must be skipped."""
    if 'bytes' not in record:
        return 1
    raw = record['bytes']
    if isinstance(raw, bool):
        return None
    if isinstance(raw, int):
        return raw
    if isinstance(raw, str):
        text = raw.strip()
        if _INTEGER_TEXT.fullmatch(text):
            return int(text, 10)
    return None


def solve(value):
    if not isinstance(value, list):
        return []
    groups = {}
    for record in _deduplicate(value):
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        path = _normalized_path(record)
        if path is None:
            continue
        number = _byte_count(record)
        if number is None:
            continue
        total, count = groups.get(path, (0, 0))
        groups[path] = (total + number, count + 1)
    return [{'path': path, 'total': groups[path][0], 'count': groups[path][1]}
            for path in sorted(groups)]
