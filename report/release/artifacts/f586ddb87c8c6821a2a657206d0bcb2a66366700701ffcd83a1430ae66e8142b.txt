import re


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []

    last_occurrence = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                last_occurrence[rec_id] = i

    retained = []
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                if last_occurrence[rec_id] != i:
                    continue
        retained.append(record)

    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        if 'category' in record:
            raw_category = record['category']
        elif 'kind' in record:
            raw_category = record['kind']
        else:
            continue

        if not isinstance(raw_category, str):
            continue
        category = raw_category.strip()
        if not category:
            continue

        if 'priority' not in record:
            priority = 1
        else:
            raw_priority = record['priority']
            if isinstance(raw_priority, bool):
                continue
            elif isinstance(raw_priority, int):
                priority = raw_priority
            elif isinstance(raw_priority, str):
                stripped_priority = raw_priority.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped_priority):
                    priority = int(stripped_priority, 10)
                else:
                    continue
            else:
                continue

        if category not in groups:
            groups[category] = priority
        else:
            groups[category] = max(groups[category], priority)

    result = []
    for category in sorted(groups.keys()):
        result.append({'category': category, 'maximum': groups[category]})
    return result
