import re
import sys

if hasattr(sys, 'set_int_max_str_digits'):
    sys.set_int_max_str_digits(0)

BYTES_REGEX = re.compile(r'^[+-]?[0-9]+$')


def solve(value):
    if not isinstance(value, list):
        return []

    # Step 1: Deduplicate before all other rules.
    # Retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent.
    last_index = {}
    for i, rec in enumerate(value):
        if isinstance(rec, dict):
            rec_id = rec.get('id')
            if isinstance(rec_id, str) and rec_id:
                last_index[rec_id] = i

    groups = {}
    for i, rec in enumerate(value):
        # Skip non-object records
        if not isinstance(rec, dict):
            continue

        # If it has a nonempty string id, retain only the last occurrence
        rec_id = rec.get('id')
        if isinstance(rec_id, str) and rec_id:
            if last_index[rec_id] != i:
                continue

        # Skip records whose 'active' is false
        if rec.get('active') is False:
            continue

        # Read 'path', falling back to 'route' only when 'path' is absent.
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if 'path' in rec:
            raw_path = rec['path']
        elif 'route' in rec:
            raw_path = rec['route']
        else:
            continue

        if not isinstance(raw_path, str):
            continue

        path = raw_path.strip()
        if not path:
            continue

        # Read 'bytes', defaulting to 1 only when absent.
        # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace;
        # convert strings to base-10 integers. Skip all other values, including null.
        if 'bytes' not in rec:
            bytes_val = 1
        else:
            raw_bytes = rec['bytes']
            if isinstance(raw_bytes, bool):
                continue
            elif isinstance(raw_bytes, int):
                bytes_val = raw_bytes
            elif isinstance(raw_bytes, str):
                stripped_bytes = raw_bytes.strip()
                if BYTES_REGEX.fullmatch(stripped_bytes):
                    bytes_val = int(stripped_bytes, 10)
                else:
                    continue
            else:
                continue

        # Group retained records by normalized 'path'
        if path not in groups:
            groups[path] = {'total': 0, 'count': 0}
        groups[path]['total'] += bytes_val
        groups[path]['count'] += 1

    # Format and sort results lexicographically by 'path'
    result = []
    for path in sorted(groups.keys()):
        result.append({
            'path': path,
            'total': groups[path]['total'],
            'count': groups[path]['count'],
        })
    return result
