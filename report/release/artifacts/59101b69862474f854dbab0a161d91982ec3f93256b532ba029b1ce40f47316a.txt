import re

_PRIORITY_PATTERN = re.compile(r"[+-]?[0-9]+")


def _normalize_priority(raw):
    if type(raw) is int:          # bool excluded on purpose
        return raw
    if type(raw) is str:
        text = raw.strip()
        if _PRIORITY_PATTERN.fullmatch(text):
            return int(text)
    return None


def solve(value):
    if isinstance(value, (list, tuple)):
        items = value
    elif isinstance(value, (str, bytes, bytearray, dict)) or value is None:
        return []
    else:
        try:
            items = list(value)
        except TypeError:
            return []

    # Keep only the LAST occurrence of each nonempty string id.
    last_position = {}
    for position, record in enumerate(items):
        if isinstance(record, dict):
            identifier = record.get("id")
            if isinstance(identifier, str) and identifier != "":
                last_position[identifier] = position

    groups = {}
    for position, record in enumerate(items):
        if not isinstance(record, dict):
            continue
        identifier = record.get("id")
        if isinstance(identifier, str) and identifier != "" \
                and last_position.get(identifier) != position:
            continue                          # deduped away
        if record.get("active") is False:
            continue                          # only JSON false skips
        name = record.get("category") if "category" in record else record.get("kind")
        if not isinstance(name, str):
            continue
        name = name.strip()
        if name == "":
            continue
        priority = _normalize_priority(record.get("priority", 1))
        if priority is None:
            continue
        if name not in groups or priority > groups[name]:
            groups[name] = priority
    return [{"category": n, "maximum": groups[n]} for n in sorted(groups)]
