import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1: deduplicate by nonempty string 'id', keeping the last occurrence.
    # Records without such an id are independent and always retained.
    seen = set()
    retained = []
    if isinstance(value, list):
        for record in reversed(value):
            if isinstance(record, dict):
                ident = record.get('id')
                if isinstance(ident, str) and ident != '':
                    if ident in seen:
                        continue
                    seen.add(ident)
            retained.append(record)
        retained.reverse()
    else:
        retained = list(value)

    groups = {}
    for record in retained:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        # Normalized key: prefer 'sku', fall back to 'item' only if 'sku' absent.
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Delta: default 1 only when the key is absent.
        if 'delta' in record:
            number = record['delta']
            if type(number) is int:
                pass
            elif isinstance(number, str):
                text = number.strip()
                if _INT_RE.fullmatch(text):
                    number = int(text, 10)
                else:
                    continue
            else:
                continue
        else:
            number = 1
        groups[name] = groups.get(name, 0) + number

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
