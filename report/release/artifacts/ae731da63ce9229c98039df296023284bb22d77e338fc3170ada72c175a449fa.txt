import re

BYTES_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1 (before all others): retain only the last occurrence of each
    # nonempty string 'id'; records without such an id remain independent.
    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_index[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if (isinstance(record_id, str) and record_id != ''
                    and last_index.get(record_id) != position):
                continue  # superseded by a later record with the same id
        else:
            continue  # skip non-object records

        if record.get('active') is False:
            continue

        if 'path' in record:
            key = record['path']
        elif 'route' in record:
            key = record['route']
        else:
            continue
        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue

        if 'bytes' not in record:
            number = 1
        else:
            raw = record['bytes']
            if isinstance(raw, bool):
                continue
            if isinstance(raw, int):
                number = raw
            elif isinstance(raw, str):
                text = raw.strip()
                if not BYTES_PATTERN.fullmatch(text):
                    continue
                number = int(text, 10)
            else:
                continue

        total, count = groups.get(key, (0, 0))
        groups[key] = (total + number, count + 1)

    result = []
    for name in sorted(groups):
        total, count = groups[name]
        result.append({'path': name, 'total': total, 'count': count})
    return result
