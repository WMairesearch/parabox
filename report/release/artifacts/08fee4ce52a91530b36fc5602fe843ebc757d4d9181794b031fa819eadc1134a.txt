import re

_NUMBER = re.compile(r'[+-]?[0-9]+')


def _record_id(record):
    """Return the dedup id of a record, or None when it has no usable id."""
    if not isinstance(record, dict):
        return None
    identifier = record.get('id')
    if isinstance(identifier, str) and identifier != '':
        return identifier
    return None


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: keep only the last occurrence of each nonempty
    # string id.  Records without such an id stay independent.
    last_occurrence = {}
    for position, record in enumerate(value):
        identifier = _record_id(record)
        if identifier is not None:
            last_occurrence[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        identifier = _record_id(record)
        if identifier is not None and last_occurrence[identifier] != position:
            continue

        if not isinstance(record, dict):
            continue

        if record.get('active') is False:
            continue

        # 'path' wins; 'route' is used only when 'path' is absent.
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')

        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        # Default of 1 applies only when 'bytes' is absent.
        if 'bytes' in record:
            number = record['bytes']
        else:
            number = 1
            total, count = groups.get(name, (0, 0))
            groups[name] = (total + number, count + 1)
            continue

        if isinstance(number, str):
            stripped = number.strip()
            if not _NUMBER.fullmatch(stripped):
                continue
            number = int(stripped, 10)
        elif isinstance(number, bool) or type(number) is not int:
            continue

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    result = []
    for name in sorted(groups):
        total, count = groups[name]
        result.append({'path': name, 'total': total, 'count': count})
    return result
