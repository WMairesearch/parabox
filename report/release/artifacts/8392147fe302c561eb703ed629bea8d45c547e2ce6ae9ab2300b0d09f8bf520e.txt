import re


def solve(value):
    # Pass 1: deduplicate - keep only the LAST occurrence of each nonempty string 'id'.
    last_index = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_index[rid] = i

    groups = {}
    for i, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        rid = record.get('id')
        if isinstance(rid, str) and rid != '' and last_index[rid] != i:
            continue  # superseded by a later record with the same id
        if record.get('active') is False:
            continue
        if 'sku' in record:
            name = record['sku']
        elif 'item' in record:
            name = record['item']
        else:
            continue
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'delta' in record:
            number = record['delta']
            if type(number) is int:
                pass
            elif isinstance(number, str):
                text = number.strip()
                if re.fullmatch(r'[+-]?[0-9]+', text) is None:
                    continue
                number = int(text, 10)
            else:
                continue
        else:
            number = 1
        groups[name] = groups.get(name, 0) + number

    return [{'sku': key, 'total': groups[key]} for key in sorted(groups)]
