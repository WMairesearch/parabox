import re

_PRIORITY_RE = re.compile(r'[+-]?[0-9]+')


def _valid_id(record):
    """Return the record's id if it is a nonempty string, else None."""
    if not isinstance(record, dict):
        return None
    identifier = record.get('id')
    if isinstance(identifier, str) and identifier != '':
        return identifier
    return None


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: keep only the last record for each nonempty string id.
    last_position = {}
    for position, record in enumerate(value):
        identifier = _valid_id(record)
        if identifier is not None:
            last_position[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = _valid_id(record)
        if identifier is not None and last_position[identifier] != position:
            continue
        if record.get('active') is False:
            continue

        # category, falling back to kind only when category is absent
        if 'category' in record:
            raw_name = record['category']
        elif 'kind' in record:
            raw_name = record['kind']
        else:
            continue
        if not isinstance(raw_name, str):
            continue
        name = raw_name.strip()
        if not name:
            continue

        # priority defaults to 1 only when the key is absent
        if 'priority' in record:
            raw_priority = record['priority']
            if type(raw_priority) is int:
                priority = raw_priority
            elif isinstance(raw_priority, str):
                text = raw_priority.strip()
                if not _PRIORITY_RE.fullmatch(text):
                    continue
                priority = int(text, 10)
            else:
                continue
        else:
            priority = 1

        if name in groups:
            if priority > groups[name]:
                groups[name] = priority
        else:
            groups[name] = priority

    return [
        {'category': name, 'maximum': groups[name]}
        for name in sorted(groups)
    ]
