"""Group event records according to the specification in PROBLEM.md."""

import re

_INTEGER = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # ---- Deduplicate first: keep only the last occurrence of each nonempty
    # ---- string 'id'.  Records without such an id stay independent.
    last_seen = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                last_seen[identifier] = position

    kept = []
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                if last_seen[identifier] != position:
                    continue
        kept.append(record)

    # ---- Group the surviving records by normalized category.
    maximum_by_category = {}
    for record in kept:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        if 'category' in record:
            raw_name = record['category']
        else:
            raw_name = record.get('kind')
        if not isinstance(raw_name, str):
            continue
        name = raw_name.strip()
        if not name:
            continue

        if 'priority' in record:
            raw_number = record['priority']
        else:
            raw_number = 1
        if isinstance(raw_number, str):
            text = raw_number.strip()
            if not _INTEGER.fullmatch(text):
                continue
            number = int(text, 10)
        elif type(raw_number) is int:
            number = raw_number
        else:
            continue

        previous = maximum_by_category.get(name)
        if previous is None or number > previous:
            maximum_by_category[name] = number

    return [
        {'category': name, 'maximum': maximum_by_category[name]}
        for name in sorted(maximum_by_category)
    ]
