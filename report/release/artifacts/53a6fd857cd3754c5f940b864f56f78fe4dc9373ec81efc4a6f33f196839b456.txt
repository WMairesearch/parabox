import re


_PRIORITY_RE = re.compile(r'[+-]?[0-9]+\Z')


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate before all other rules: keep only the last occurrence of
    # each nonempty string 'id'; records without such an id stay independent.
    last_id_index = {}
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        id_value = record.get('id')
        if isinstance(id_value, str) and id_value != '':
            last_id_index[id_value] = index

    deduplicated = []
    for index, record in enumerate(value):
        if isinstance(record, dict):
            id_value = record.get('id')
            if isinstance(id_value, str) and id_value != '':
                if last_id_index[id_value] != index:
                    continue
        deduplicated.append(record)

    groups = {}
    for record in deduplicated:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        if 'category' in record:
            raw_category = record['category']
        else:
            raw_category = record.get('kind')

        if not isinstance(raw_category, str):
            continue
        category = raw_category.strip()
        if not category:
            continue

        if 'priority' in record:
            priority = record['priority']
            if type(priority) is int:
                priority = int(priority)
            elif isinstance(priority, str):
                stripped_priority = priority.strip()
                if _PRIORITY_RE.fullmatch(stripped_priority):
                    priority = int(stripped_priority, 10)
                else:
                    continue
            else:
                continue
        else:
            priority = 1

        previous = groups.get(category)
        if previous is None or priority > previous:
            groups[category] = priority

    return [
        {'category': category, 'maximum': groups[category]}
        for category in sorted(groups)
    ]
