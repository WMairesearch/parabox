"""Inventory grouping per PROBLEM.md.

Rules:
  1. Deduplicate first: for records whose 'id' is a nonempty string, retain only
     the last occurrence; records without such an id stay independent.
  2. Skip non-object records and records whose 'active' is false.
  3. Key = 'sku', falling back to 'item' only when 'sku' is absent; skip
     non-string keys and keys empty after stripping; strip surrounding whitespace.
  4. 'delta' defaults to 1 only when absent; accept ints (not bools) and strings
     matching [+-]?[0-9]+ after stripping; skip everything else (including null).
  5. Group by normalized key, sum deltas, sort lexicographically by sku.
"""

import re

_INT_RE = re.compile(r"[+-]?[0-9]+")


def _dedup_last_by_id(records):
    """Keep only the last record for each nonempty string id."""
    last_index = {}
    for index, record in enumerate(records):
        if isinstance(record, dict):
            ident = record.get("id")
            if isinstance(ident, str) and ident != "":
                last_index[ident] = index
    retained = []
    for index, record in enumerate(records):
        if isinstance(record, dict):
            ident = record.get("id")
            if isinstance(ident, str) and ident != "" and last_index[ident] != index:
                continue
        retained.append(record)
    return retained


def _delta_of(record):
    """Return the integer delta, or None when the record must be skipped."""
    if "delta" not in record:
        return 1
    raw = record["delta"]
    if isinstance(raw, bool):
        return None
    if isinstance(raw, int):
        return raw
    if isinstance(raw, str):
        text = raw.strip()
        if not _INT_RE.fullmatch(text):
            return None
        return int(text, 10)
    return None


def _key_of(record):
    """Return the normalized sku key, or None when the record must be skipped."""
    if "sku" in record:
        raw = record["sku"]
    elif "item" in record:
        raw = record["item"]
    else:
        return None
    if not isinstance(raw, str):
        return None
    key = raw.strip()
    if key == "":
        return None
    return key


def solve(value):
    if not isinstance(value, list):
        return []
    totals = {}
    for record in _dedup_last_by_id(value):
        if not isinstance(record, dict):
            continue
        if record.get("active") is False:
            continue
        key = _key_of(record)
        if key is None:
            continue
        delta = _delta_of(record)
        if delta is None:
            continue
        totals[key] = totals.get(key, 0) + delta
    return [{"sku": name, "total": totals[name]} for name in sorted(totals)]
