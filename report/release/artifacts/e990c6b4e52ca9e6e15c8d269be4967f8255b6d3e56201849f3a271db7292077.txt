import re


_PRIORITY_RE = re.compile(r'[+-]?[0-9]+\Z')


def solve(value):
    # Deduplicate before all other rules: for each nonempty string 'id',
    # keep only the last record containing it.  Records without such an id
    # remain independent.
    last_id_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_id_position[record_id] = position

    retained_positions = set()
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if last_id_position.get(record_id) == position:
                    retained_positions.add(position)
            else:
                retained_positions.add(position)
        else:
            retained_positions.add(position)

    groups = {}
    for position in retained_positions:
        record = value[position]

        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'category', falling back to 'kind' only when 'category' is absent.
        if 'category' in record:
            raw_category = record['category']
        elif 'kind' in record:
            raw_category = record['kind']
        else:
            continue

        if not isinstance(raw_category, str):
            continue
        category = raw_category.strip()
        if category == '':
            continue

        # Read 'priority', defaulting to 1 only when absent.
        if 'priority' not in record:
            priority = 1
        else:
            raw_priority = record['priority']
            if type(raw_priority) is int:
                priority = raw_priority
            elif isinstance(raw_priority, str):
                stripped_priority = raw_priority.strip()
                if _PRIORITY_RE.fullmatch(stripped_priority):
                    priority = int(stripped_priority, 10)
                else:
                    continue
            else:
                continue

        current = groups.get(category)
        if current is None or priority > current:
            groups[category] = priority

    return [
        {'category': category, 'maximum': groups[category]}
        for category in sorted(groups)
    ]
