"""Inventory aggregation - see PROBLEM.md.

Rules (applied in this order):
  1. Deduplicate first: keep only the LAST occurrence of each nonempty string
     'id'; records without such an id stay independent.
  2. Skip non-dict records and records whose 'active' is JSON false.
  3. Group key: 'sku', falling back to 'item' only when 'sku' is absent.
     Key must be a string, nonempty after stripping; the stripped form is used.
  4. Value: 'delta', defaulting to 1 only when the key is absent.  Accepts
     ints (bools excluded) and strings matching [+-]?[0-9]+ after stripping.
     Anything else skips the whole record.
  5. Emit [{'sku': k, 'total': sum}, ...] sorted lexicographically by key.
The input is never mutated.
"""
import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def _last_id_positions(records):
    """Map each nonempty string id to the index of its last occurrence."""
    last = {}
    for index, record in enumerate(records):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last[rid] = index
    return last


def _group_key(record, primary, fallback):
    if primary in record:
        raw = record[primary]
    elif fallback in record:
        raw = record[fallback]
    else:
        return None
    if not isinstance(raw, str):
        return None
    stripped = raw.strip()
    return stripped or None


def _delta(record, field):
    if field not in record:
        return 1
    raw = record[field]
    if type(raw) is int:
        return raw
    if isinstance(raw, str):
        stripped = raw.strip()
        if _INT_RE.fullmatch(stripped) is not None:
            return int(stripped, 10)
    return None


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []
    records = list(value)
    last = _last_id_positions(records)
    totals = {}
    for index, record in enumerate(records):
        if not isinstance(record, dict):
            continue
        rid = record.get('id')
        if isinstance(rid, str) and rid != '' and last.get(rid) != index:
            continue
        if record.get('active') is False:
            continue
        key = _group_key(record, 'sku', 'item')
        if key is None:
            continue
        amount = _delta(record, 'delta')
        if amount is None:
            continue
        totals[key] = totals.get(key, 0) + amount
    return [{'sku': key, 'total': totals[key]} for key in sorted(totals)]
