import re


_INT_PATTERN = re.compile(r'[+-]?[0-9]+')


def _retained_indices(value):
    """Return indices of records kept by the id de-duplication rule."""
    last = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier != '':
            last[identifier] = position
    keep = []
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            keep.append(position)
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier != '':
            if last[identifier] != position:
                continue
        keep.append(position)
    return keep


def solve(value):
    groups = {}
    for position in _retained_indices(value):
        record = value[position]
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'bytes' in record:
            number = record['bytes']
            if isinstance(number, str):
                stripped = number.strip()
                if not _INT_PATTERN.fullmatch(stripped):
                    continue
                number = int(stripped, 10)
            elif type(number) is not int:
                continue
        else:
            number = 1
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)
    return [
        {'path': name, 'total': groups[name][0], 'count': groups[name][1]}
        for name in sorted(groups)
    ]
