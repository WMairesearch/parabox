import re

def solve(value):
    if not isinstance(value, list):
        return []

    seen_ids = set()
    retained_records = []
    for record in reversed(value):
        if isinstance(record, dict) and 'id' in record:
            rec_id = record['id']
            if isinstance(rec_id, str) and rec_id != '':
                if rec_id in seen_ids:
                    continue
                seen_ids.add(rec_id)
        retained_records.append(record)
    retained_records.reverse()

    groups = {}
    for record in retained_records:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        
        if 'sku' in record:
            key = record['sku']
        elif 'item' in record:
            key = record['item']
        else:
            continue
            
        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue
            
        if 'delta' not in record:
            delta = 1
        else:
            delta_raw = record['delta']
            if type(delta_raw) is int:
                delta = delta_raw
            elif isinstance(delta_raw, str):
                delta_str = delta_raw.strip()
                if re.fullmatch(r'[+-]?[0-9]+', delta_str):
                    delta = int(delta_str)
                else:
                    continue
            else:
                continue
                
        groups[key] = groups.get(key, 0) + delta

    result = []
    # Sort results lexicographically by 'sku'
    for name in sorted(groups.keys()):
        result.append({'sku': name, 'total': groups[name]})
    return result
