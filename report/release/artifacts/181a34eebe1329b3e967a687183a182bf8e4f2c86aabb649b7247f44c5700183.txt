import re


def solve(value):
    if not isinstance(value, list):
        return []

    seen_ids = set()
    retained_reversed = []
    for record in reversed(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if isinstance(rec_id, str) and rec_id != '':
                if rec_id in seen_ids:
                    continue
                seen_ids.add(rec_id)
        retained_reversed.append(record)

    retained = list(reversed(retained_reversed))

    groups = {}

    for record in retained:
        if not isinstance(record, dict):
            continue

        if record.get('active') is False:
            continue

        if 'path' in record:
            raw_path = record['path']
        elif 'route' in record:
            raw_path = record['route']
        else:
            continue

        if not isinstance(raw_path, str):
            continue

        path = raw_path.strip()
        if not path:
            continue

        if 'bytes' not in record:
            byte_count = 1
        else:
            raw_bytes = record['bytes']
            if type(raw_bytes) is int:
                byte_count = raw_bytes
            elif isinstance(raw_bytes, str):
                stripped_bytes = raw_bytes.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped_bytes):
                    byte_count = int(stripped_bytes, 10)
                else:
                    continue
            else:
                continue

        if path not in groups:
            groups[path] = {'path': path, 'total': 0, 'count': 0}
        groups[path]['total'] += byte_count
        groups[path]['count'] += 1

    return [groups[k] for k in sorted(groups.keys())]
