import re

def solve(value):
    last_id_indices = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            if 'id' in record:
                rec_id = record['id']
                if isinstance(rec_id, str) and rec_id != "":
                    last_id_indices[rec_id] = i

    retained_records = []
    for i, record in enumerate(value):
        if isinstance(record, dict) and 'id' in record:
            rec_id = record['id']
            if isinstance(rec_id, str) and rec_id != "":
                if last_id_indices[rec_id] != i:
                    continue
        retained_records.append(record)
        
    groups = {}
    for record in retained_records:
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        if 'sku' in record:
            key_val = record['sku']
        elif 'item' in record:
            key_val = record['item']
        else:
            continue
            
        if not isinstance(key_val, str):
            continue
        key_val = key_val.strip()
        if not key_val:
            continue
            
        if 'delta' in record:
            delta_val = record['delta']
        else:
            delta_val = 1
            
        if isinstance(delta_val, int) and not isinstance(delta_val, bool):
            delta = delta_val
        elif isinstance(delta_val, str):
            s = delta_val.strip()
            if re.fullmatch(r'[+-]?[0-9]+', s):
                delta = int(s)
            else:
                continue
        else:
            continue
            
        groups[key_val] = groups.get(key_val, 0) + delta

    result = []
    for sku in sorted(groups.keys()):
        result.append({'sku': sku, 'total': groups[sku]})
        
    return result
