import re

_BYTES_PATTERN = re.compile(r'[+-]?[0-9]+')


def _parse_bytes(raw):
    """Return the integer value of a present 'bytes' field, or None if invalid."""
    if isinstance(raw, bool):
        return None
    if isinstance(raw, int):
        return raw
    if isinstance(raw, str):
        text = raw.strip()
        if _BYTES_PATTERN.fullmatch(text):
            try:
                return int(text, 10)
            except ValueError:
                return None
    return None


def solve(value):
    # Rule 1 (before all others): keep only the last occurrence of each
    # nonempty string 'id'; records without such an id remain independent.
    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_index[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if last_index[record_id] != position:
                    continue

        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            key = record['path']
        else:
            key = record.get('route')
        if not isinstance(key, str):
            continue
        key = key.strip()
        if not key:
            continue

        # Read 'bytes', defaulting to 1 only when absent.
        if 'bytes' in record:
            number = _parse_bytes(record['bytes'])
            if number is None:
                continue
        else:
            number = 1

        total, count = groups.get(key, (0, 0))
        groups[key] = (total + number, count + 1)

    result = []
    for name in sorted(groups):
        total, count = groups[name]
        result.append({'path': name, 'total': total, 'count': count})
    return result
