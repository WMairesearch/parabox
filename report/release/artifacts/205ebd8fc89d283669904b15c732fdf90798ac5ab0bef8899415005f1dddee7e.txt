import re

_PRIORITY_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Deduplicate first: retain only the last occurrence of each nonempty
    # string 'id'; records without such an id remain independent.
    last_occurrence = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_occurrence[rid] = index

    groups = {}
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        rid = record.get('id')
        if isinstance(rid, str) and rid != '' and last_occurrence.get(rid) != index:
            continue
        if record.get('active') is False:
            continue
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'priority' in record:
            number = record['priority']
            if type(number) is int:
                pass
            elif isinstance(number, str):
                text = number.strip()
                if not _PRIORITY_RE.fullmatch(text):
                    continue
                number = int(text, 10)
            else:
                continue
        else:
            number = 1
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
