import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def _record_id(record):
    """Return a dedupe id for a record, or None when it has no usable id."""
    if not isinstance(record, dict):
        return None
    value = record.get('id')
    if isinstance(value, str) and value:
        return value
    return None


def _priority(record):
    """Return the parsed priority, or None when the value is unacceptable."""
    if 'priority' in record:
        value = record['priority']
    else:
        value = 1
    if type(value) is int:
        return value
    if isinstance(value, str):
        text = value.strip()
        if _INT_RE.fullmatch(text):
            return int(text, 10)
    return None


def _category(record):
    """Return the normalized category, or None when it is unusable."""
    if 'category' in record:
        value = record['category']
    else:
        value = record.get('kind')
    if not isinstance(value, str):
        return None
    value = value.strip()
    if not value:
        return None
    return value


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: only the last occurrence of each nonempty string id
    # survives; records without such an id stay independent.
    last_positions = {}
    for position, record in enumerate(value):
        key = _record_id(record)
        if key is not None:
            last_positions[key] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        key = _record_id(record)
        if key is not None and last_positions[key] != position:
            continue
        name = _category(record)
        if name is None:
            continue
        number = _priority(record)
        if number is None:
            continue
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [{'category': name, 'maximum': groups[name]}
            for name in sorted(groups)]
