import re

def solve(value):
    seen_ids = set()
    groups = {}
    for record in reversed(value):
        if isinstance(record, dict):
            id_val = record.get('id')
            if isinstance(id_val, str) and id_val:
                if id_val in seen_ids:
                    continue
                seen_ids.add(id_val)
                
        # Rule 2: Skip non-object and inactive
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        # Rule 3: sku key
        if 'sku' in record:
            sku_val = record['sku']
        elif 'item' in record:
            sku_val = record['item']
        else:
            continue
            
        if not isinstance(sku_val, str):
            continue
            
        sku_val = sku_val.strip()
        if not sku_val:
            continue
            
        # Rule 4: delta value
        if 'delta' in record:
            delta_val = record['delta']
            if isinstance(delta_val, bool):
                continue
            elif isinstance(delta_val, int):
                delta = delta_val
            elif isinstance(delta_val, str):
                s = delta_val.strip()
                if re.fullmatch(r'[+-]?[0-9]+', s):
                    delta = int(s)
                else:
                    continue
            else:
                continue
        else:
            delta = 1
            
        # Rule 5: group by normalized sku
        groups[sku_val] = groups.get(sku_val, 0) + delta
        
    result = []
    for sku in sorted(groups.keys()):
        result.append({'sku': sku, 'total': groups[sku]})
    return result
