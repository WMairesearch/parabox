import re

_DELTA_PATTERN = re.compile(r'[+-]?[0-9]+')


def _valid_id(record):
    """Return the record's id if it is a nonempty string, else None."""
    identifier = record.get('id')
    if isinstance(identifier, str) and identifier != '':
        return identifier
    return None


def solve(value):
    if not isinstance(value, list):
        return []

    # 1. Deduplicate: keep only the last occurrence of each nonempty string id.
    #    Records without such an id stay independent.
    last_index = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = _valid_id(record)
            if identifier is not None:
                last_index[identifier] = position

    kept = []
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = _valid_id(record)
            if identifier is not None and last_index[identifier] != position:
                continue
        kept.append(record)

    # 2. Filter and accumulate.
    groups = {}
    for record in kept:
        if not isinstance(record, dict):
            continue
        if record.get('active', True) is False:
            continue

        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if 'delta' in record:
            number = record['delta']
            if isinstance(number, str):
                text = number.strip()
                if not _DELTA_PATTERN.fullmatch(text):
                    continue
                number = int(text, 10)
            elif type(number) is not int:
                continue
        else:
            number = 1

        groups[name] = groups.get(name, 0) + number

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
