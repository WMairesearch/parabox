import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def _valid_id(record):
    if not isinstance(record, dict):
        return None
    if 'id' not in record:
        return None
    ident = record['id']
    if isinstance(ident, str) and ident != '':
        return ident
    return None


def _priority(record):
    if 'priority' not in record:
        return 1
    value = record['priority']
    if type(value) is int:
        return value
    if isinstance(value, str):
        text = value.strip()
        if _INT_RE.fullmatch(text):
            try:
                return int(text, 10)
            except ValueError:
                return None
    return None


def _category(record):
    if 'category' in record:
        name = record['category']
    elif 'kind' in record:
        name = record['kind']
    else:
        name = None
    if not isinstance(name, str):
        return None
    name = name.strip()
    if not name:
        return None
    return name


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: keep only the last occurrence of each nonempty string id.
    last_seen = {}
    for position, record in enumerate(value):
        ident = _valid_id(record)
        if ident is not None:
            last_seen[ident] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        ident = _valid_id(record)
        if ident is not None and last_seen[ident] != position:
            continue
        if record.get('active') is False:
            continue
        name = _category(record)
        if name is None:
            continue
        number = _priority(record)
        if number is None:
            continue
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [{'category': name, 'maximum': groups[name]}
            for name in sorted(groups)]
