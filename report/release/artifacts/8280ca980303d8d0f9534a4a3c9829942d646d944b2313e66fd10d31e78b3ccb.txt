import re

_PRIORITY_RE = re.compile(r'[+-]?[0-9]+')


def _dedupe(records):
    """Keep only the last occurrence of each nonempty string 'id'."""
    last = {}
    for position, record in enumerate(records):
        if isinstance(record, dict):
            ident = record.get('id')
            if isinstance(ident, str) and ident:
                last[ident] = position
    keep_positions = set(last.values())
    kept = []
    for position, record in enumerate(records):
        if isinstance(record, dict):
            ident = record.get('id')
            if isinstance(ident, str) and ident:
                if position in keep_positions:
                    kept.append(record)
                continue
        kept.append(record)
    return kept


def solve(value):
    if not isinstance(value, list):
        return []

    records = _dedupe(value)

    groups = {}
    for record in records:
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if 'priority' in record:
            number = record['priority']
            if isinstance(number, str):
                text = number.strip()
                if not _PRIORITY_RE.fullmatch(text):
                    continue
                number = int(text, 10)
            elif type(number) is not int:
                continue
        else:
            number = 1

        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [
        {'category': name, 'maximum': groups[name]}
        for name in sorted(groups)
    ]
