import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    try:
        records = list(value)
    except TypeError:
        return []
    last = {}
    for pos, rec in enumerate(records):
        if isinstance(rec, dict):
            ident = rec.get('id')
            if isinstance(ident, str) and ident:
                last[ident] = pos
    groups = {}
    for pos, rec in enumerate(records):
        if not isinstance(rec, dict):
            continue
        ident = rec.get('id')
        if isinstance(ident, str) and ident and last.get(ident) != pos:
            continue
        if rec.get('active') is False:
            continue
        name = rec['sku'] if 'sku' in rec else rec.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'delta' in rec:
            number = rec['delta']
            if isinstance(number, str):
                text = number.strip()
                if not _INT_RE.fullmatch(text):
                    continue
                number = int(text)
            if type(number) is not int:
                continue
        else:
            number = 1
        groups[name] = groups.get(name, 0) + number
    return [{'sku': n, 'total': groups[n]} for n in sorted(groups)]
