import re

_DELTA_PATTERN = re.compile(r'[+-]?[0-9]+')


def _normalized_key(record):
    # Read 'sku', falling back to 'item' only when 'sku' is absent.
    if 'sku' in record:
        name = record['sku']
    elif 'item' in record:
        name = record['item']
    else:
        return None
    if not isinstance(name, str):
        return None
    name = name.strip()
    if not name:
        return None
    return name


def _delta_value(record):
    # Read 'delta', defaulting to 1 only when absent.
    if 'delta' not in record:
        return 1
    number = record['delta']
    if isinstance(number, bool):
        return None
    if isinstance(number, int):
        return number
    if isinstance(number, str):
        text = number.strip()
        if _DELTA_PATTERN.fullmatch(text):
            try:
                return int(text, 10)
            except ValueError:
                return None
        return None
    return None


def solve(value):
    if not isinstance(value, list):
        return []

    # Dedup before all other rules: retain only the last occurrence of
    # each nonempty string 'id'; records without such an id stay independent.
    last_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                last_position[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier:
            if last_position.get(identifier) != position:
                continue
        if record.get('active') is False:
            continue
        name = _normalized_key(record)
        if name is None:
            continue
        number = _delta_value(record)
        if number is None:
            continue
        groups[name] = groups.get(name, 0) + number

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
