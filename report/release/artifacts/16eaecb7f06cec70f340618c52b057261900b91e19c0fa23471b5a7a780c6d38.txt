import re

def solve(value):
    seen_ids = set()
    retained = []
    
    # First retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Deduplicate before all other rules.
    for record in reversed(value):
        if isinstance(record, dict) and 'id' in record:
            id_val = record['id']
            if isinstance(id_val, str) and id_val != '':
                if id_val in seen_ids:
                    continue
                seen_ids.add(id_val)
        retained.append(record)
    
    retained.reverse()
    
    groups = {}
    
    for record in retained:
        # Skip non-object records
        if not isinstance(record, dict):
            continue
            
        # skip records whose 'active' is false
        if record.get('active') is False:
            continue
            
        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            key_val = record['sku']
        elif 'item' in record:
            key_val = record['item']
        else:
            continue
            
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if not isinstance(key_val, str):
            continue
        name = key_val.strip()
        if not name:
            continue
            
        # Read 'delta', defaulting to 1 only when absent.
        if 'delta' in record:
            delta_val = record['delta']
        else:
            delta_val = 1
            
        # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace;
        # convert strings to base-10 integers. Skip all other values, including null.
        if type(delta_val) is int:
            number = delta_val
        elif isinstance(delta_val, str):
            s_delta = delta_val.strip()
            if re.fullmatch(r'[+-]?[0-9]+', s_delta):
                number = int(s_delta)
            else:
                continue
        else:
            continue
            
        groups[name] = groups.get(name, 0) + number
        
    result = []
    for name in sorted(groups.keys()):
        result.append({'sku': name, 'total': groups[name]})
        
    return result
