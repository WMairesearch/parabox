import re


_BYTES_PATTERN = re.compile(r'[+-]?[0-9]+\Z')


def _is_nonempty_string(value):
    return isinstance(value, str) and value != ''


def solve(value):
    if not isinstance(value, list):
        return []

    # First deduplicate: keep only the last occurrence of each nonempty string
    # 'id'.  Records without such an id remain independent and are all kept for
    # now; every other rule is applied after this step.
    last_id_positions = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if _is_nonempty_string(record_id):
                last_id_positions[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        if isinstance(record, dict) and _is_nonempty_string(record.get('id')):
            if last_id_positions.get(record.get('id')) != position:
                continue

        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            raw_path = record['path']
        else:
            raw_path = record.get('route')

        if not isinstance(raw_path, str) or not raw_path.strip():
            continue
        path = raw_path.strip()

        # Read 'bytes', defaulting to 1 only when absent.
        if 'bytes' not in record:
            number = 1
        else:
            raw_bytes = record['bytes']
            if type(raw_bytes) is int:
                number = raw_bytes
            elif isinstance(raw_bytes, str):
                stripped = raw_bytes.strip()
                if not _BYTES_PATTERN.match(stripped):
                    continue
                number = int(stripped)
            else:
                continue

        total, count = groups.get(path, (0, 0))
        groups[path] = (total + number, count + 1)

    result = [
        {'path': path, 'total': groups[path][0], 'count': groups[path][1]}
        for path in sorted(groups)
    ]
    return result
