"""events: group retained records by normalized category, report maximum priority."""

import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def _records_of(value):
    """Return the input as a list of candidate records, without mutating it."""
    if isinstance(value, (str, bytes, bytearray)):
        return []
    if value is None:
        return []
    try:
        return list(value)
    except TypeError:
        return []


def _record_id(record):
    """Deduplication key: the raw 'id' when it is a nonempty string, else None."""
    rid = record.get('id')
    if isinstance(rid, str) and rid != '':
        return rid
    return None


def _category_of(record):
    """Normalized category, or None when the record must be skipped."""
    if 'category' in record:
        key = record['category']
    else:
        key = record.get('kind')
    if not isinstance(key, str):
        return None
    key = key.strip()
    if key == '':
        return None
    return key


def _priority_of(record):
    """Parsed priority (default 1 when absent), or None when the record is skipped."""
    if 'priority' not in record:
        return 1
    raw = record['priority']
    if type(raw) is int:
        return raw
    if isinstance(raw, str):
        text = raw.strip()
        if _INT_RE.fullmatch(text) is not None:
            return int(text, 10)
    return None


def solve(value):
    records = _records_of(value)
    total = len(records)

    # Deduplicate first: only the last occurrence of each nonempty string id survives.
    last_index = {}
    for index in range(total):
        record = records[index]
        if isinstance(record, dict):
            rid = _record_id(record)
            if rid is not None:
                last_index[rid] = index

    groups = {}
    for index in range(total):
        record = records[index]
        if not isinstance(record, dict):
            continue
        rid = _record_id(record)
        if rid is not None and last_index[rid] != index:
            continue
        if record.get('active') is False:
            continue
        category = _category_of(record)
        if category is None:
            continue
        priority = _priority_of(record)
        if priority is None:
            continue
        if category not in groups or priority > groups[category]:
            groups[category] = priority

    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
