import re


def solve(value):
    # Deduplicate before all other rules: keep only the last occurrence of
    # each nonempty string 'id'.  Records without such an id remain independent.
    last_id_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_id_position[record_id] = position

    duplicate_positions = set()
    if last_id_position:
        for position, record in enumerate(value):
            if isinstance(record, dict):
                record_id = record.get('id')
                if (
                    isinstance(record_id, str)
                    and record_id != ''
                    and last_id_position[record_id] != position
                ):
                    duplicate_positions.add(position)

    groups = {}
    for position, record in enumerate(value):
        if position in duplicate_positions:
            continue
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')

        if not isinstance(name, str) or not name.strip():
            continue
        name = name.strip()

        if 'bytes' not in record:
            number = 1
        else:
            number = record['bytes']
            if isinstance(number, str):
                stripped = number.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped):
                    number = int(stripped)
                else:
                    continue
            if type(number) is not int:
                continue

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': total, 'count': count}
        for name, (total, count) in sorted(groups.items())
    ]
