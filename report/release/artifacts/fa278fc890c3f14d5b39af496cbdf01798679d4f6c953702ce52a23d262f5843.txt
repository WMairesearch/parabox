import re


_PRIORITY_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Deduplicate first: for each nonempty string id, keep only its last
    # occurrence.  Records without such an id are not affected.
    last_indices = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_indices[record_id] = index

    groups = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if last_indices[record_id] != index:
                    continue

        # Non-object records are skipped.
        if not isinstance(record, dict):
            continue

        # Records explicitly marked active: false are skipped.
        if record.get('active') is False:
            continue

        # Read category, falling back to kind only when category is absent.
        if 'category' in record:
            raw_category = record['category']
        else:
            raw_category = record.get('kind')

        if not isinstance(raw_category, str):
            continue
        category = raw_category.strip()
        if not category:
            continue

        # Read priority, defaulting to 1 only when the key is absent.
        if 'priority' not in record:
            priority = 1
        else:
            raw_priority = record['priority']
            if type(raw_priority) is int:
                priority = raw_priority
            elif isinstance(raw_priority, str):
                stripped_priority = raw_priority.strip()
                if _PRIORITY_RE.fullmatch(stripped_priority):
                    priority = int(stripped_priority)
                else:
                    continue
            else:
                continue

        if category in groups:
            if priority > groups[category]:
                groups[category] = priority
        else:
            groups[category] = priority

    return [
        {'category': category, 'maximum': groups[category]}
        for category in sorted(groups)
    ]
