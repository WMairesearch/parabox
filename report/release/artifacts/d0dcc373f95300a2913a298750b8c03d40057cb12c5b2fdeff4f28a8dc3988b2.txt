import re

def solve(value):
    valid_indices = [False] * len(value)
    seen_ids = set()
    for i in range(len(value) - 1, -1, -1):
        record = value[i]
        if isinstance(record, dict) and 'id' in record:
            id_val = record['id']
            if isinstance(id_val, str) and id_val != '':
                if id_val in seen_ids:
                    continue
                seen_ids.add(id_val)
        valid_indices[i] = True
    
    groups = {}
    for i, record in enumerate(value):
        if not valid_indices[i]:
            continue
        
        if not isinstance(record, dict):
            continue
            
        if record.get('active') is False:
            continue
            
        if 'sku' in record:
            key_val = record['sku']
        elif 'item' in record:
            key_val = record['item']
        else:
            continue
            
        if not isinstance(key_val, str):
            continue
        key_str = key_val.strip()
        if not key_str:
            continue
            
        if 'delta' not in record:
            delta_val = 1
        else:
            delta_val = record['delta']
            
        if type(delta_val) is int:
            delta = delta_val
        elif isinstance(delta_val, str):
            s = delta_val.strip()
            if re.fullmatch(r'[+-]?[0-9]+', s):
                delta = int(s)
            else:
                continue
        else:
            continue
            
        groups[key_str] = groups.get(key_str, 0) + delta
        
    result = [{'sku': k, 'total': groups[k]} for k in sorted(groups.keys())]
    return result
