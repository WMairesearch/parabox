import re


def solve(value):
    if not isinstance(value, list):
        return []

    # First, retain only the last occurrence of each nonempty string 'id'.
    # All other records (including records without such an id) remain independent.
    last_id_positions = {}
    independent_positions = []
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_id_positions[record_id] = position
                continue
        independent_positions.append(position)

    keep_positions = set(last_id_positions.values())
    keep_positions.update(independent_positions)

    groups = {}
    for position, record in enumerate(value):
        if position not in keep_positions:
            continue

        # Skip non-object records and records whose 'active' is false.
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            raw_path = record.get('path')
        else:
            raw_path = record.get('route')

        if not isinstance(raw_path, str):
            continue
        path = raw_path.strip()
        if not path:
            continue

        # Read 'bytes', defaulting to 1 only when absent.
        if 'bytes' in record:
            raw_bytes = record.get('bytes')
            if type(raw_bytes) is int:
                number = raw_bytes
            elif isinstance(raw_bytes, str):
                stripped_bytes = raw_bytes.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped_bytes):
                    number = int(stripped_bytes, 10)
                else:
                    continue
            else:
                continue
        else:
            number = 1

        total, count = groups.get(path, (0, 0))
        groups[path] = (total + number, count + 1)

    result = [
        {'path': path, 'total': total, 'count': count}
        for path, (total, count) in sorted(groups.items())
    ]
    return result
