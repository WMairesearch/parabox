import re

_PRIORITY_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Deduplicate first: keep only the last occurrence of each nonempty
    # string 'id'; records without such an id remain independent.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if (isinstance(identifier, str) and identifier
                and latest[identifier] != position):
            continue
        if record.get('active') is False:
            continue
        # Fall back to 'kind' only when 'category' is absent.
        if 'category' in record:
            name = record['category']
        else:
            name = record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Default priority to 1 only when absent.
        if 'priority' in record:
            number = record['priority']
        else:
            number = 1
        if isinstance(number, bool):
            continue
        if isinstance(number, int):
            pass
        elif isinstance(number, str):
            text = number.strip()
            if not _PRIORITY_RE.fullmatch(text):
                continue
            number = int(text, 10)
        else:
            continue
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    return [{'category': name, 'maximum': groups[name]}
            for name in sorted(groups)]
