import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1 (applied first): keep only the last occurrence of each nonempty
    # string id.  Records without such an id stay independent.
    last_index = {}
    for index, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                last_index[identifier] = index

    groups = {}
    for index, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        identifier = record.get('id')
        if isinstance(identifier, str) and identifier != '':
            if last_index[identifier] != index:
                continue

        if record.get('active') is False:
            continue

        if 'path' in record:
            name = record['path']
        else:
            name = record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if name == '':
            continue

        if 'bytes' in record:
            number = record['bytes']
        else:
            number = 1
        if isinstance(number, str):
            candidate = number.strip()
            if _INT_RE.fullmatch(candidate):
                number = int(candidate, 10)
        if isinstance(number, bool) or not isinstance(number, int):
            continue

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': groups[name][0], 'count': groups[name][1]}
        for name in sorted(groups)
    ]
