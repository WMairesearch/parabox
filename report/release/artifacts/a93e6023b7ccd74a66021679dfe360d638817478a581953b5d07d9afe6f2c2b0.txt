"""Inventory grouping per PROBLEM.md."""

import re

_DELTA_PATTERN = re.compile(r'[+-]?[0-9]+')
_MISSING = object()


def _record_id(record):
    """Return the record's id if it is a nonempty string, else None."""
    if not isinstance(record, dict):
        return None
    candidate = record.get('id')
    if isinstance(candidate, str) and candidate != '':
        return candidate
    return None


def _dedupe(value):
    """Keep only the last record carrying each nonempty string id."""
    kept = []
    seen = set()
    for record in reversed(value):
        identifier = _record_id(record)
        if identifier is not None:
            if identifier in seen:
                continue
            seen.add(identifier)
        kept.append(record)
    kept.reverse()
    return kept


def _delta(record):
    """Return the record's numeric delta, or None when it must be skipped."""
    number = record.get('delta', _MISSING)
    if number is _MISSING:
        return 1
    if isinstance(number, str):
        text = number.strip()
        if not _DELTA_PATTERN.fullmatch(text):
            return None
        try:
            return int(text, 10)
        except ValueError:  # pragma: no cover - absurdly long digit strings
            return None
    if isinstance(number, bool):
        return None
    if isinstance(number, int):
        return number
    return None


def solve(value):
    if not isinstance(value, list):
        return []
    groups = {}
    for record in _dedupe(value):
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        name = record.get('sku', _MISSING)
        if name is _MISSING:
            name = record.get('item', _MISSING)
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        number = _delta(record)
        if number is None:
            continue
        groups[name] = groups.get(name, 0) + number
    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
