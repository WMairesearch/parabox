import re

_DELTA_PATTERN = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # Rule 1 (before all others): retain only the last occurrence of each
    # nonempty string 'id'; records without such an id remain independent.
    last_occurrence = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier != '':
                last_occurrence[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier != '':
            if last_occurrence[identifier] != position:
                continue
        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue
        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            name = record['sku']
        elif 'item' in record:
            name = record['item']
        else:
            continue
        if not isinstance(name, str):
            continue
        name = name.strip()
        if name == '':
            continue
        # Read 'delta', defaulting to 1 only when absent.
        if 'delta' in record:
            raw = record['delta']
            if isinstance(raw, bool):
                continue
            if isinstance(raw, int):
                number = raw
            elif isinstance(raw, str):
                text = raw.strip()
                if _DELTA_PATTERN.fullmatch(text) is None:
                    continue
                number = int(text, 10)
            else:
                continue
        else:
            number = 1
        groups[name] = groups.get(name, 0) + number

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
