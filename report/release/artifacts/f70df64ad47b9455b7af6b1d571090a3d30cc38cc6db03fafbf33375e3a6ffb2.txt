import re

def solve(value):
    seen_ids = set()
    retained_records = []
    
    # Deduplicate before all other rules. Retain only the last occurrence of each nonempty string 'id'.
    for record in reversed(value):
        has_valid_id = False
        if isinstance(record, dict) and 'id' in record:
            id_val = record['id']
            if isinstance(id_val, str) and id_val != '':
                has_valid_id = True
                valid_id = id_val
        
        if has_valid_id:
            if valid_id not in seen_ids:
                seen_ids.add(valid_id)
                retained_records.append(record)
        else:
            retained_records.append(record)
            
    retained_records.reverse()
    
    groups = {}
    for record in retained_records:
        if not isinstance(record, dict):
            continue
        
        if record.get('active') is False:
            continue
            
        if 'category' in record:
            cat_val = record['category']
        elif 'kind' in record:
            cat_val = record['kind']
        else:
            continue
            
        if not isinstance(cat_val, str):
            continue
        
        category = cat_val.strip()
        if not category:
            continue
            
        if 'priority' not in record:
            priority = 1
        else:
            p_val = record['priority']
            if type(p_val) is bool:
                continue
            elif type(p_val) is int:
                priority = p_val
            elif isinstance(p_val, str):
                p_val_stripped = p_val.strip()
                if re.fullmatch(r'[+-]?[0-9]+', p_val_stripped):
                    priority = int(p_val_stripped)
                else:
                    continue
            else:
                continue
                
        if category not in groups:
            groups[category] = priority
        else:
            groups[category] = max(groups[category], priority)
            
    result = [{'category': k, 'maximum': v} for k, v in sorted(groups.items())]
    return result
