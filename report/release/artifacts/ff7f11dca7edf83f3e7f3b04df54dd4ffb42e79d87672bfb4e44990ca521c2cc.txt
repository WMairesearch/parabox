import re

def solve(value):
    # Do not modify the input
    # Deduplicate: first retain only the last occurrence of each nonempty string 'id'
    last_occurrences = {}
    for i, record in enumerate(value):
        if isinstance(record, dict) and 'id' in record:
            id_val = record['id']
            if isinstance(id_val, str) and id_val != '':
                last_occurrences[id_val] = i

    groups = {}
    for i, record in enumerate(value):
        if isinstance(record, dict) and 'id' in record:
            id_val = record['id']
            if isinstance(id_val, str) and id_val != '':
                if last_occurrences[id_val] != i:
                    continue
        
        # Skip non-object records
        if not isinstance(record, dict):
            continue
            
        # Skip records whose 'active' is false
        if 'active' in record and record['active'] is False:
            continue
            
        # Read 'category', falling back to 'kind' only when absent
        if 'category' in record:
            cat_val = record['category']
        elif 'kind' in record:
            cat_val = record['kind']
        else:
            continue
            
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace
        if not isinstance(cat_val, str):
            continue
        cat_val = cat_val.strip()
        if not cat_val:
            continue
            
        # Read 'priority', defaulting to 1 only when absent
        if 'priority' not in record:
            priority = 1
        else:
            p_val = record['priority']
            if isinstance(p_val, int) and not isinstance(p_val, bool):
                priority = p_val
            elif isinstance(p_val, str):
                p_val_stripped = p_val.strip()
                if re.fullmatch(r'[+-]?[0-9]+', p_val_stripped):
                    priority = int(p_val_stripped)
                else:
                    continue
            else:
                continue
                
        # Group retained records by normalized 'category'
        if cat_val in groups:
            groups[cat_val] = max(groups[cat_val], priority)
        else:
            groups[cat_val] = priority

    # Sort results lexicographically by 'category'
    result = []
    for cat_val in sorted(groups.keys()):
        result.append({'category': cat_val, 'maximum': groups[cat_val]})
        
    return result
