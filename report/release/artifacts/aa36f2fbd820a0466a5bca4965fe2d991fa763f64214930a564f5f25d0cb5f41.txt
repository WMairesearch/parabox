import re


def solve(value):
    # 1) Deduplicate before all other rules: keep only the last occurrence
    #    of each nonempty string 'id'.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        # Skip non-object records.
        if not isinstance(record, dict):
            continue
        # Keep only the retained (last) record for each id.
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != position:
            continue
        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue
        # Read 'category', falling back to 'kind' only when 'category' is absent.
        if 'category' in record:
            name = record.get('category')
        else:
            name = record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Read 'priority', defaulting to 1 only when absent.
        if 'priority' in record:
            number = record.get('priority')
        else:
            number = 1
        if isinstance(number, str):
            stripped = number.strip()
            if re.fullmatch(r'[+-]?[0-9]+', stripped):
                number = int(stripped, 10)
            else:
                continue
        elif type(number) is not int:  # accepts ints, rejects booleans
            continue
        if name in groups:
            if number > groups[name]:
                groups[name] = number
        else:
            groups[name] = number

    # Sort results lexicographically by 'category'.
    return [{'category': name, 'maximum': groups[name]} for name in sorted(groups)]
