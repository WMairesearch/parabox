import re


def solve(value):
    # Step 1: deduplicate by id before any other rules.
    # Keep the last occurrence of each nonempty string id; records without
    # such an id are independent and are all kept at this stage.
    seen_ids = set()
    deduped = []
    for record in reversed(value):
        if isinstance(record, dict):
            id_value = record.get('id')
            if isinstance(id_value, str) and id_value != '':
                if id_value in seen_ids:
                    continue
                seen_ids.add(id_value)
        deduped.append(record)
    deduped.reverse()

    groups = {}

    for record in deduped:
        # Step 2: skip non-object records and inactive records.
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue

        # Step 3: resolve the normalized category key.
        if 'category' in record:
            raw_category = record['category']
        else:
            raw_category = record.get('kind')
        if not isinstance(raw_category, str):
            continue
        category = raw_category.strip()
        if not category:
            continue

        # Step 4: resolve priority.
        if 'priority' not in record:
            priority = 1
        else:
            raw_priority = record['priority']
            if type(raw_priority) is int:
                priority = raw_priority
            elif isinstance(raw_priority, str):
                stripped_priority = raw_priority.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped_priority):
                    priority = int(stripped_priority)
                else:
                    continue
            else:
                continue

        # Step 5: group by normalized category, keeping the maximum priority.
        if category not in groups or priority > groups[category]:
            groups[category] = priority

    return [
        {'category': category, 'maximum': groups[category]}
        for category in sorted(groups)
    ]
