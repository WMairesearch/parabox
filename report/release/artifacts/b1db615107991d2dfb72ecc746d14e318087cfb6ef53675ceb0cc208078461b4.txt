"""access_logs: group retained access-log records by normalized path.

Spec (PROBLEM.md):
  1. Dedup FIRST: keep only the last occurrence of each nonempty string 'id';
     records without such an id stay independent.
  2. Skip non-object records and records whose 'active' is false.
  3. Key: 'path', falling back to 'route' only when 'path' is ABSENT.
     Must be a string, non-empty after stripping; strip surrounding whitespace.
  4. 'bytes': default 1 only when ABSENT. Accept ints (not bools) and strings
     matching [+-]?[0-9]+ after stripping (base 10). Skip every other value.
  5. Group by normalized path -> {'path', 'total' (sum), 'count'}; sort by path.
  6. [] when nothing remains; never mutate the input.
"""

import re

_BYTES_RE = re.compile(r'[+-]?[0-9]+')


def _surviving_indices(records):
    """Indices kept after id-dedup (last occurrence of each nonempty string id)."""
    last = {}
    for index, record in enumerate(records):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last[rid] = index
    kept = []
    for index, record in enumerate(records):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '' and last[rid] != index:
                continue
        kept.append(index)
    return kept


def _bytes_of(record):
    """Valid byte count for a record, or None when the record must be skipped."""
    if 'bytes' not in record:
        return 1
    number = record['bytes']
    if isinstance(number, bool):
        return None
    if isinstance(number, int):
        return number
    if isinstance(number, str):
        text = number.strip()
        if _BYTES_RE.fullmatch(text) is None:
            return None
        return int(text, 10)
    return None


def solve(value):
    if not isinstance(value, list):
        return []
    groups = {}
    for index in _surviving_indices(value):
        record = value[index]
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        name = record['path'] if 'path' in record else record.get('route')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if name == '':
            continue
        number = _bytes_of(record)
        if number is None:
            continue
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)
    return [{'path': key, 'total': groups[key][0], 'count': groups[key][1]}
            for key in sorted(groups)]
