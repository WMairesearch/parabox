import re


def solve(value):
    # Deduplicate first: retain only the last occurrence of each nonempty string id.
    latest = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            identifier = record.get('id')
            if isinstance(identifier, str) and identifier:
                latest[identifier] = position
    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier and latest[identifier] != position:
            continue
        if record.get('active') is False:
            continue
        if 'sku' in record:
            name = record.get('sku')
        else:
            name = record.get('item')
        if not isinstance(name, str) or not name.strip():
            continue
        name = name.strip()
        if 'delta' in record:
            number = record.get('delta')
        else:
            number = 1
        if isinstance(number, bool):
            continue
        if isinstance(number, str):
            stripped = number.strip()
            if re.fullmatch(r'[+-]?[0-9]+', stripped):
                number = int(stripped)
            else:
                continue
        if type(number) is not int:
            continue
        groups[name] = groups.get(name, 0) + number
    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
