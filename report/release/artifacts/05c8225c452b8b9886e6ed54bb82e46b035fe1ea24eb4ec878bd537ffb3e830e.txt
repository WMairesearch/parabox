import re


def solve(value):
    """Group JSON access log records by normalized path.

    The input is a JSON list of records.  Deduplication happens first: only
    the last occurrence of each nonempty-string ``id`` is kept, while records
    without such an id are always kept.
    """
    if not isinstance(value, list):
        return []

    # --- Deduplicate before all other rules. ---
    last_pos_by_id = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_pos_by_id[record_id] = position

    keep_last_positions = set(last_pos_by_id.values())
    retained = []
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            has_nonempty_string_id = isinstance(record_id, str) and record_id != ''
            if has_nonempty_string_id and position not in keep_last_positions:
                continue
        retained.append(record)

    groups = {}
    for record in retained:
        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records whose 'active' is JSON false.
        if record.get('active') is False:
            continue

        # Normalize the path.  'route' is used only when 'path' is absent.
        if 'path' in record:
            path = record['path']
        else:
            path = record.get('route')

        if not isinstance(path, str):
            continue
        path = path.strip()
        if not path:
            continue

        # Read 'bytes', defaulting to 1 only when the key is absent.
        if 'bytes' not in record:
            number = 1
        else:
            raw_bytes = record['bytes']
            if type(raw_bytes) is int:
                number = raw_bytes
            elif isinstance(raw_bytes, str):
                stripped_bytes = raw_bytes.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped_bytes):
                    number = int(stripped_bytes, 10)
                else:
                    continue
            else:
                continue

        total, count = groups.get(path, (0, 0))
        groups[path] = (total + number, count + 1)

    return [
        {'path': path, 'total': total, 'count': count}
        for path, (total, count) in sorted(groups.items())
    ]
