import re


def solve(value):
    # Deduplicate before all other rules: for each nonempty string 'id',
    # keep only the last occurrence. Records without such an id remain.
    last_id_position = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        record_id = record.get('id')
        if isinstance(record_id, str) and record_id != '':
            last_id_position[record_id] = position

    retained_records = []
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                if position != last_id_position[record_id]:
                    continue
        retained_records.append(record)

    groups = {}
    for record in retained_records:
        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records whose active is false.
        if record.get('active') is False:
            continue

        # Read path, falling back to route only when path is absent.
        if 'path' in record:
            raw_path = record.get('path')
        else:
            raw_path = record.get('route')

        # Skip non-string or empty-after-stripping path keys; strip whitespace.
        if not isinstance(raw_path, str):
            continue
        name = raw_path.strip()
        if not name:
            continue

        # Read bytes, defaulting to 1 only when absent.
        if 'bytes' in record:
            raw_bytes = record['bytes']
            if type(raw_bytes) is int:
                number = raw_bytes
            elif isinstance(raw_bytes, str):
                stripped_bytes = raw_bytes.strip()
                if re.fullmatch(r'[+-]?[0-9]+', stripped_bytes):
                    number = int(stripped_bytes)
                else:
                    continue
            else:
                continue
        else:
            number = 1

        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    result = [
        {'path': name, 'total': total, 'count': count}
        for name, (total, count) in sorted(groups.items())
    ]
    return result
