import re

def solve(input_records):
    seen_ids = set()
    deduped_records = []
    
    # First retain only the last occurrence of each nonempty string 'id'
    for record in reversed(input_records):
        has_id = False
        if isinstance(record, dict) and 'id' in record:
            record_id = record['id']
            if isinstance(record_id, str) and record_id != "":
                has_id = True
                if record_id in seen_ids:
                    continue
                seen_ids.add(record_id)
        deduped_records.append(record)
    deduped_records.reverse()
    
    groups = {}
    for record in deduped_records:
        # Skip non-object records
        if not isinstance(record, dict):
            continue
            
        # Skip records whose 'active' is false
        if record.get('active') is False:
            continue
            
        # Read 'path', falling back to 'route' only when 'path' is absent
        if 'path' in record:
            p = record['path']
        elif 'route' in record:
            p = record['route']
        else:
            continue
            
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace
        if not isinstance(p, str):
            continue
        p = p.strip()
        if p == "":
            continue
            
        # Read 'bytes', defaulting to 1 only when absent
        if 'bytes' not in record:
            b = 1
        else:
            b_raw = record['bytes']
            if isinstance(b_raw, int) and not isinstance(b_raw, bool):
                b = b_raw
            elif isinstance(b_raw, str):
                b_str = b_raw.strip()
                if re.fullmatch(r'[+-]?[0-9]+', b_str):
                    b = int(b_str)
                else:
                    continue
            else:
                continue
                
        # Group retained records by normalized 'path'
        if p not in groups:
            groups[p] = {'total': 0, 'count': 0}
        groups[p]['total'] += b
        groups[p]['count'] += 1
        
    result = []
    for p in sorted(groups.keys()):
        result.append({
            'path': p,
            'total': groups[p]['total'],
            'count': groups[p]['count']
        })
    return result
