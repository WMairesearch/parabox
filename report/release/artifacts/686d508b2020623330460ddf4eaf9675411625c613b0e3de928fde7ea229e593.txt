import re


def solve(value):
    if not isinstance(value, list):
        return []

    # First retain only the last occurrence of each nonempty string 'id';
    # records without such an id remain independent. Deduplicate before all other rules.
    last_seen = {}
    for idx, item in enumerate(value):
        if isinstance(item, dict):
            item_id = item.get('id')
            if isinstance(item_id, str) and item_id:
                last_seen[item_id] = idx

    retained = []
    for idx, item in enumerate(value):
        if isinstance(item, dict):
            item_id = item.get('id')
            if isinstance(item_id, str) and item_id:
                if last_seen[item_id] != idx:
                    continue
        retained.append(item)

    # Process retained records
    groups = {}
    for item in retained:
        # Skip non-object records and records whose 'active' is false.
        if not isinstance(item, dict):
            continue
        if item.get('active') is False:
            continue

        # Read 'category', falling back to 'kind' only when 'category' is absent.
        # Skip non-string or empty-after-stripping keys; strip surrounding whitespace.
        if 'category' in item:
            category_val = item['category']
        elif 'kind' in item:
            category_val = item['kind']
        else:
            continue

        if not isinstance(category_val, str):
            continue
        category = category_val.strip()
        if not category:
            continue

        # Read 'priority', defaulting to 1 only when absent.
        # Accept integers (not booleans) and strings matching [+-]?[0-9]+ after stripping whitespace;
        # convert strings to base-10 integers. Skip all other values, including null.
        if 'priority' not in item:
            priority = 1
        else:
            priority_val = item['priority']
            if type(priority_val) is int:
                priority = priority_val
            elif isinstance(priority_val, str):
                p_str = priority_val.strip()
                if re.fullmatch(r'[+-]?[0-9]+', p_str):
                    priority = int(p_str, 10)
                else:
                    continue
            else:
                continue

        # Group retained records by normalized 'category' and find maximum priority
        if category not in groups:
            groups[category] = priority
        else:
            if priority > groups[category]:
                groups[category] = priority

    # Each output object contains 'category' and the maximum priority as 'maximum'.
    # Sort results lexicographically by 'category'. Return [] when no records remain.
    result = []
    for cat in sorted(groups.keys()):
        result.append({
            'category': cat,
            'maximum': groups[cat]
        })
    return result
