import re

_DELTA_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    # First retain only the last occurrence of each nonempty string 'id'.
    # Records without such an id remain independent.
    last_id_position = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        id_value = record.get('id')
        if isinstance(id_value, str) and id_value != '':
            last_id_position[id_value] = position

    groups = {}

    for position, record in enumerate(value):
        # Deduplicate by nonempty string id before applying all other rules.
        if isinstance(record, dict):
            id_value = record.get('id')
            if isinstance(id_value, str) and id_value != '':
                if position != last_id_position[id_value]:
                    continue

        # Skip non-object records.
        if not isinstance(record, dict):
            continue

        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue

        # Read 'sku', falling back to 'item' only when 'sku' is absent.
        if 'sku' in record:
            key = record['sku']
        elif 'item' in record:
            key = record['item']
        else:
            continue

        if not isinstance(key, str):
            continue
        sku = key.strip()
        if not sku:
            continue

        # Read 'delta', defaulting to 1 only when absent.
        if 'delta' not in record:
            delta = 1
        else:
            delta_value = record['delta']
            if type(delta_value) is int:
                delta = delta_value
            elif isinstance(delta_value, str):
                stripped_delta = delta_value.strip()
                if _DELTA_RE.fullmatch(stripped_delta):
                    delta = int(stripped_delta)
                else:
                    continue
            else:
                continue

        groups[sku] = groups.get(sku, 0) + delta

    result = [{'sku': sku, 'total': groups[sku]} for sku in sorted(groups)]
    return result
