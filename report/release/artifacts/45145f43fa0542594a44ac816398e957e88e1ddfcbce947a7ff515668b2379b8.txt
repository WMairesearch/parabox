import re

_BYTES_PATTERN = re.compile(r'[+-]?[0-9]+')
_ABSENT = object()


def solve(value):
    # Pass 1: locate the last occurrence of each nonempty string 'id'.
    last_position = {}
    for position, record in enumerate(value):
        if isinstance(record, dict):
            record_id = record.get('id')
            if isinstance(record_id, str) and record_id != '':
                last_position[record_id] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        # Deduplicate: keep only the last record for each nonempty string id.
        record_id = record.get('id')
        if (isinstance(record_id, str) and record_id != ''
                and last_position[record_id] != position):
            continue
        # Skip records whose 'active' is false.
        if record.get('active') is False:
            continue
        # Read 'path', falling back to 'route' only when 'path' is absent.
        if 'path' in record:
            name = record['path']
        elif 'route' in record:
            name = record['route']
        else:
            continue
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Read 'bytes', defaulting to 1 only when absent.
        number = record.get('bytes', _ABSENT)
        if number is _ABSENT:
            number = 1
        elif isinstance(number, bool):
            continue
        elif isinstance(number, int):
            pass
        elif isinstance(number, str):
            text = number.strip()
            if _BYTES_PATTERN.fullmatch(text) is None:
                continue
            number = int(text, 10)
        else:
            continue
        total, count = groups.get(name, (0, 0))
        groups[name] = (total + number, count + 1)

    result = [{'path': path, 'total': total, 'count': count}
              for path, (total, count) in groups.items()]
    result.sort(key=lambda item: item['path'])
    return result
