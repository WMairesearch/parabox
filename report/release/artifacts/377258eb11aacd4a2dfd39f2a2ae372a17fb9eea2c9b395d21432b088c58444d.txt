"""access_logs: group retained access records by normalized path."""

import re

_INT_RE = re.compile(r'[+-]?[0-9]+')


def _has_id(record):
    """Return the dedup key for a record, or None when it has no usable id."""
    if not isinstance(record, dict):
        return None
    ident = record.get('id')
    if isinstance(ident, str) and ident != '':
        return ident
    return None


def _bytes_value(record):
    """Return the byte count for a record, or None when the value is invalid."""
    if 'bytes' not in record:
        return 1
    value = record['bytes']
    if isinstance(value, bool):
        return None
    if isinstance(value, int):
        return value
    if isinstance(value, str):
        text = value.strip()
        if _INT_RE.fullmatch(text):
            return int(text, 10)
    return None


def _path_value(record):
    """Return the normalized path for a record, or None when unusable."""
    if 'path' in record:
        value = record['path']
    else:
        value = record.get('route')
    if not isinstance(value, str):
        return None
    text = value.strip()
    if not text:
        return None
    return text


def solve(value):
    if not isinstance(value, (list, tuple)):
        return []

    records = list(value)

    # Deduplicate first: keep only the last record for each nonempty string id.
    last_index = {}
    for position, record in enumerate(records):
        key = _has_id(record)
        if key is not None:
            last_index[key] = position

    totals = {}
    for position, record in enumerate(records):
        if not isinstance(record, dict):
            continue
        key = _has_id(record)
        if key is not None and last_index[key] != position:
            continue
        if record.get('active') is False:
            continue
        name = _path_value(record)
        if name is None:
            continue
        number = _bytes_value(record)
        if number is None:
            continue
        total, count = totals.get(name, (0, 0))
        totals[name] = (total + number, count + 1)

    return [
        {'path': name, 'total': totals[name][0], 'count': totals[name][1]}
        for name in sorted(totals)
    ]
