import re

_DELTA_RE = re.compile(r'[+-]?[0-9]+')


def _retained(records):
    """Return records after id-based last-occurrence deduplication."""
    kept = [None] * len(records)
    seen = set()
    for position in range(len(records) - 1, -1, -1):
        record = records[position]
        if isinstance(record, dict):
            key = record.get('id')
            if isinstance(key, str) and key != '':
                if key in seen:
                    continue
                seen.add(key)
        kept[position] = record
    return [r for r in kept if r is not None]


def solve(value):
    if not isinstance(value, list):
        return []
    groups = {}
    for record in _retained(value):
        if not isinstance(record, dict):
            continue
        if record.get('active') is False:
            continue
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        if 'delta' in record:
            number = record['delta']
        else:
            number = 1
        if isinstance(number, str):
            text = number.strip()
            if _DELTA_RE.fullmatch(text) is None:
                continue
            number = int(text, 10)
        elif type(number) is not int:
            continue
        groups[name] = groups.get(name, 0) + number
    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
