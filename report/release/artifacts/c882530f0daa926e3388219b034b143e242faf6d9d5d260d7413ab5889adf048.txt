import re

def solve(value):
    # 1. Deduplication
    valid_id_last_idx = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rec_id = record.get('id')
            if type(rec_id) is str and rec_id != "":
                valid_id_last_idx[rec_id] = i
                
    groups = {}
    for i, record in enumerate(value):
        # Apply deduplication filter
        if isinstance(record, dict):
            rec_id = record.get('id')
            if type(rec_id) is str and rec_id != "":
                if valid_id_last_idx[rec_id] != i:
                    continue
                    
        # 2. Skip non-object records
        if not isinstance(record, dict):
            continue
            
        # 3. Skip records whose 'active' is false
        if record.get('active') is False:
            continue
            
        # 4. Read 'sku', falling back to 'item' only when 'sku' is absent
        if 'sku' in record:
            key_val = record['sku']
        elif 'item' in record:
            key_val = record['item']
        else:
            continue
            
        if type(key_val) is not str:
            continue
            
        sku_stripped = key_val.strip()
        if not sku_stripped:
            continue
            
        # 5. Read 'delta', defaulting to 1 only when absent
        if 'delta' not in record:
            delta = 1
        else:
            delta_val = record['delta']
            if type(delta_val) is int:
                delta = delta_val
            elif type(delta_val) is str:
                d_str = delta_val.strip()
                if re.fullmatch(r'[+-]?[0-9]+', d_str):
                    delta = int(d_str)
                else:
                    continue
            else:
                continue
                
        # Group retained records by normalized 'sku'
        groups[sku_stripped] = groups.get(sku_stripped, 0) + delta
        
    result = []
    for sku in sorted(groups.keys()):
        result.append({'sku': sku, 'total': groups[sku]})
    return result
