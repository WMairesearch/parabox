import re


_INTEGER = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # First: find the last occurrence of every nonempty-string id.
    latest = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue
        identifier = record.get('id')
        if isinstance(identifier, str) and identifier:
            latest[identifier] = position

    groups = {}
    for position, record in enumerate(value):
        if not isinstance(record, dict):
            continue

        identifier = record.get('id')
        if isinstance(identifier, str) and identifier:
            if latest.get(identifier) != position:
                # An earlier occurrence of this id; only the last survives.
                continue

        if record.get('active') is False:
            continue

        # 'sku' wins whenever the key is present, even if unusable.
        if 'sku' in record:
            name = record['sku']
        else:
            name = record.get('item')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue

        if 'delta' in record:
            number = record['delta']
        else:
            number = 1
        if isinstance(number, str):
            stripped = number.strip()
            if _INTEGER.fullmatch(stripped):
                number = int(stripped, 10)
        if type(number) is not int:
            continue

        groups[name] = groups.get(name, 0) + number

    return [{'sku': name, 'total': groups[name]} for name in sorted(groups)]
