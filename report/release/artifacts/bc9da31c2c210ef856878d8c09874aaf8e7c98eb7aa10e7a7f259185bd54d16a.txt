import re

_PRIORITY_RE = re.compile(r'[+-]?[0-9]+')


def solve(value):
    if not isinstance(value, list):
        return []

    # Rule 1: deduplicate FIRST, before all other rules.
    # Among records that are objects with a nonempty string 'id',
    # keep only the last occurrence (by original position).
    # Records without such an id stay independent. id compared raw.
    last_index = {}
    for i, record in enumerate(value):
        if isinstance(record, dict):
            rid = record.get('id')
            if isinstance(rid, str) and rid != '':
                last_index[rid] = i

    groups = {}
    for i, record in enumerate(value):
        # Rule 2: skip non-object records.
        if not isinstance(record, dict):
            continue
        rid = record.get('id')
        if isinstance(rid, str) and rid != '' and last_index[rid] != i:
            continue
        # Skip records whose 'active' is false (strict boolean identity).
        if record.get('active') is False:
            continue
        # Rule 3: category, falling back to 'kind' only when 'category' absent.
        name = record['category'] if 'category' in record else record.get('kind')
        if not isinstance(name, str):
            continue
        name = name.strip()
        if not name:
            continue
        # Rule 4: priority, defaulting to 1 only when absent.
        if 'priority' in record:
            pr = record['priority']
            if type(pr) is int:  # integers, not booleans
                num = pr
            elif isinstance(pr, str):
                text = pr.strip()
                if _PRIORITY_RE.fullmatch(text):
                    num = int(text, 10)
                else:
                    continue
            else:
                continue
        else:
            num = 1
        # Rule 5: group by normalized category, keep maximum.
        if name not in groups or num > groups[name]:
            groups[name] = num

    # Rule 6: sorted lexicographically by category; [] when nothing remains.
    return [{'category': c, 'maximum': groups[c]} for c in sorted(groups)]
